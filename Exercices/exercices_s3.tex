\section{Questions à choix multiples}
% ============================================================================

\setcounter{qcmcounter}{0}

\qcm{Si des travailleurs découragés cessent de chercher un emploi et quittent la population active, l'effet immédiat est :}
\choix{
   \item Le taux de chômage augmente
   \item Le taux de chômage diminue
   \item Le taux de participation augmente
   \item Le taux d'emploi augmente
}

\begin{reponse}
\textbf{(b)} Les travailleurs découragés passent de la catégorie \guillemotleft~chômeurs~\guillemotright\ ($U$) à la catégorie \guillemotleft~inactifs~\guillemotright\ ($I$). Le numérateur du taux de chômage ($U$) diminue, et le dénominateur ($E + U$) diminue aussi. Puisque le taux de chômage est inférieur à 1 (le numérateur est plus petit que le dénominateur), retirer une unité des deux fait \textbf{diminuer} le ratio. C'est pourquoi le taux de chômage seul peut être trompeur : une baisse peut refléter des retraits du marché plutôt que des embauches. Le taux de participation ($= (E+U)/\text{Pop.}$) diminue aussi.
\end{reponse}

\qcm{Un progrès technologique qui augmente la productivité des travailleurs dans le secteur manufacturier a pour effet, dans un modèle simple d'offre et de demande de travail :}
\choix{
   \item De déplacer la courbe d'offre de travail vers la droite
   \item De déplacer la courbe de demande de travail vers la droite
   \item De déplacer la courbe de demande de travail vers la gauche
   \item De n'avoir aucun effet sur le marché du travail
}

\begin{reponse}
\textbf{(b)} La demande de travail des entreprises dépend de la \textbf{productivité marginale du travail}. Si un progrès technologique augmente la productivité des travailleurs, chaque travailleur produit davantage de valeur $\to$ les entreprises sont prêtes à embaucher plus de travailleurs au même salaire, ou à payer plus pour le même nombre. Cela déplace la courbe de demande vers la \textbf{droite} (hausse), ce qui augmente à la fois l'emploi et le salaire d'équilibre.
\end{reponse}

\qcm{Selon le cadre d'Acemoglu et Restrepo, une technologie d'automatisation \guillemotleft~so-so~\guillemotright\ est particulièrement préoccupante pour les travailleurs parce que :}
\choix{
   \item Elle est trop coûteuse à mettre en \oe{}uvre
   \item Elle remplace des travailleurs sans créer de gains de productivité substantiels ni de nouvelles tâches
   \item Elle ne concerne que les travailleurs hautement qualifiés
   \item Elle augmente la demande de travail dans tous les secteurs
}

\begin{reponse}
\textbf{(b)} Une technologie \guillemotleft~so-so~\guillemotright\ est une automatisation qui remplace des travailleurs dans certaines tâches (\textbf{effet de déplacement}) mais sans générer de gains de productivité suffisants pour créer de nouvelles tâches ou élargir le marché (\textbf{effet de réinstauration} faible). Contrairement aux technologies transformatrices (machine à vapeur, ordinateur), l'automatisation \guillemotleft~so-so~\guillemotright\ n'a qu'un effet négatif net sur l'emploi et les salaires. Exemple : les caisses automatiques en supermarché remplacent des caissiers mais ne créent pas de nouvelles activités comparables.
\end{reponse}

\qcm{La \guillemotleft~prime de qualification~\guillemotright\ (\textit{skill premium}) désigne :}
\choix{
   \item La différence de taux de chômage entre travailleurs qualifiés et non qualifiés
   \item L'écart de salaire entre les travailleurs ayant un diplôme universitaire et ceux sans diplôme
   \item Le coût des études universitaires
   \item Le taux de rendement des investissements en capital humain
}

\begin{reponse}
\textbf{(b)} La prime de qualification mesure l'\textbf{écart de salaire} entre les travailleurs qualifiés (typiquement : diplôme universitaire) et les travailleurs non qualifiés. Cet écart a augmenté considérablement depuis les années 1980 dans les économies avancées, reflétant une demande accrue pour les compétences liées aux nouvelles technologies (progrès technologique biaisé en faveur des qualifications).
\end{reponse}

\qcm{Selon la courbe de Gatsby, les pays avec une forte inégalité de revenus tendent à avoir :}
\choix{
   \item Une mobilité sociale plus élevée
   \item Une mobilité sociale plus faible
   \item Un taux de croissance plus élevé
   \item Un taux de chômage plus faible
}

\begin{reponse}
\textbf{(b)} La courbe de Gatsby montre une corrélation positive entre l'inégalité des revenus (mesurée par le coefficient de Gini) et la persistance intergénérationnelle des revenus (mesurée par l'élasticité intergénérationnelle). Autrement dit, dans les pays plus inégaux (comme les États-Unis), les enfants de familles pauvres ont \textbf{moins de chances} de s'élever dans l'échelle des revenus $\to$ la mobilité sociale est plus \textbf{faible}.
\end{reponse}

\qcm{Un retraité qui ne cherche pas d'emploi est classé comme :}
\choix{
   \item Employé
   \item Chômeur
   \item Inactif
   \item Hors de la population en âge de travailler
}

\begin{reponse}
\textbf{(c)} La population en âge de travailler se divise en trois catégories mutuellement exclusives : \textbf{employés} ($E$), \textbf{chômeurs} ($U$, sans emploi mais cherchent activement) et \textbf{inactifs} ($I$, ne cherchent pas d'emploi). Un retraité qui ne cherche pas d'emploi est \textbf{inactif}. Il fait partie de la population en âge de travailler (15+ ans), mais pas de la \textbf{population active} ($E + U$). (d) serait correct uniquement s'il avait moins de 15 ans.
\end{reponse}

\qcm{Le vieillissement de la population tend à :}
\choix{
   \item Augmenter le taux de participation
   \item Réduire le taux de participation, car les retraités quittent la population active
   \item Augmenter le taux de chômage
   \item N'avoir aucun effet sur les indicateurs du marché du travail
}

\begin{reponse}
\textbf{(b)} Le vieillissement de la population augmente la proportion de personnes âgées (retraités) dans la population en âge de travailler. Les retraités sont \textbf{inactifs} (ne cherchent pas d'emploi), ce qui réduit le ratio population active / population en âge de travailler, soit le \textbf{taux de participation}. C'est un défi majeur pour les économies avancées : moins de travailleurs par rapport à la population totale $\to$ pression sur les systèmes de retraite et la croissance du PIB par habitant.
\end{reponse}

\qcm{En récession, si les salaires nominaux sont rigides à la baisse, le résultat est :}
\choix{
   \item Une hausse des profits des entreprises
   \item Du chômage, car la quantité demandée de travail est inférieure à la quantité offerte au salaire en vigueur
   \item Une baisse du niveau des prix
   \item Une hausse de la productivité
}

\begin{reponse}
\textbf{(b)} Quand la demande de travail diminue (récession), le salaire d'équilibre devrait baisser pour rétablir le plein emploi. Mais si les salaires nominaux sont \textbf{rigides à la baisse} (conventions collectives, salaire minimum, normes sociales), le salaire reste au-dessus du nouvel équilibre. Au salaire en vigueur, la quantité demandée ($L^D$) est inférieure à la quantité offerte ($L^S$) $\to$ l'écart $L^S - L^D$ représente le \textbf{chômage involontaire}.
\end{reponse}

\qcm{La \guillemotleft~courbe de l'éléphant~\guillemotright\ (Lakner-Milanovic) montre qu'entre 1988 et 2008, les plus grands perdants de la mondialisation ont été :}
\choix{
   \item Les plus pauvres du monde
   \item La classe moyenne des pays émergents
   \item La classe moyenne-inférieure des pays riches
   \item Le 1\,\% le plus riche mondial
}

\begin{reponse}
\textbf{(c)} La courbe de l'éléphant montre la croissance cumulative du revenu réel par percentile de la distribution mondiale. Les grands gagnants sont (1) la \textbf{classe moyenne des pays émergents} (percentiles 30--60, surtout Chine et Inde) et (2) le \textbf{top 1\,\% mondial}. Les grands \textbf{perdants} sont la \textbf{classe moyenne-inférieure des pays riches} (percentiles 75--90 de la distribution mondiale), dont les revenus ont stagné. Ce sont les ouvriers industriels des pays développés, touchés par la concurrence des importations et la désindustrialisation.
\end{reponse}

\qcm{Le fait que les guichets automatiques (ATM) n'ont pas éliminé les caissiers de banque illustre :}
\choix{
   \item L'effet de déplacement de l'automatisation
   \item L'effet de réinstauration (\textit{reinstatement effect})
   \item Le paradoxe de Jevons
   \item Le progrès technologique biaisé en faveur des qualifications
}

\begin{reponse}
\textbf{(b)} C'est un exemple classique de l'\textbf{effet de réinstauration} dans le cadre d'Acemoglu et Restrepo. Les ATM ont réduit le nombre de caissiers \textbf{par succursale} (effet de déplacement), mais en réduisant le coût d'exploitation d'une succursale, ils ont permis aux banques d'ouvrir \textbf{plus de succursales}. Le nombre total de caissiers a en fait \textbf{augmenté}, et leur travail s'est réorienté vers le conseil et la vente de produits financiers. L'automatisation d'une tâche peut \textbf{créer} de l'emploi si elle réduit les coûts et élargit le marché.
\end{reponse}

\shortq{Considérez un marché du travail simple avec une offre et une demande de travail. Le gouvernement impose un salaire minimum supérieur au salaire d'équilibre.}

\begin{enumerate}[label=(\alph*)]
   \item Illustrez graphiquement l'effet du salaire minimum sur l'emploi et le chômage.
   \item Identifiez les travailleurs qui bénéficient du salaire minimum et ceux qui en souffrent.
   \item Expliquez pourquoi l'effet du salaire minimum sur l'emploi total est ambigu dans la réalité.
\end{enumerate}

\begin{reponse}
\textbf{(a)} Le graphique devrait montrer :
\begin{itemize}
   \item Axe vertical : salaire ($w$); axe horizontal : quantité de travail ($L$)
   \item Courbe d'offre croissante, courbe de demande décroissante
   \item Salaire d'équilibre $w^*$ à l'intersection
   \item Salaire minimum $\bar{w} > w^*$ : ligne horizontale au-dessus de l'équilibre
   \item Au salaire $\bar{w}$ : la quantité demandée ($L^D$) est inférieure à la quantité offerte ($L^S$)
   \item L'écart $L^S - L^D$ représente le chômage créé par le salaire minimum
\end{itemize}

\textbf{(b)} \textbf{Gagnants :} les travailleurs qui conservent leur emploi au salaire minimum --- ils reçoivent un salaire plus élevé ($\bar{w} > w^*$). \textbf{Perdants :} les travailleurs qui perdent leur emploi ou ne sont pas embauchés à cause du salaire minimum ($L^D < L^*$). Ce sont généralement les travailleurs les moins productifs ou les moins expérimentés.

\textbf{(c)} L'effet est ambigu parce que le modèle simple suppose une concurrence parfaite. En réalité :
\begin{itemize}
   \item Les entreprises peuvent avoir un \textbf{pouvoir de monopsone} (un seul employeur dominant), auquel cas le salaire minimum peut \textit{augmenter} l'emploi en corrigeant le sous-emploi
   \item Le salaire minimum peut réduire le \textbf{roulement} de personnel (turnover) et augmenter la productivité
   \item Les études empiriques (Card \& Krueger, 1994) montrent des effets sur l'emploi souvent faibles, voire nuls, pour des hausses modérées du salaire minimum
\end{itemize}
\end{reponse}

\vfi{La hausse de la participation des femmes au marché du travail depuis les années 1950 est principalement due à une augmentation des salaires offerts aux femmes.}

\begin{reponse}
\textbf{Incertain.} La hausse des salaires offerts aux femmes (augmentation du coût d'opportunité de ne pas travailler) est \textit{un} facteur important, mais pas le seul. Plusieurs autres facteurs ont joué un rôle majeur :
\begin{itemize}
   \item \textbf{Innovations domestiques} : machine à laver, aspirateur, réfrigérateur $\to$ réduction du temps consacré aux tâches ménagères
   \item \textbf{Contraception} : meilleur contrôle de la fécondité $\to$ possibilité de planifier une carrière
   \item \textbf{Normes sociales} : évolution des attentes culturelles sur le rôle des femmes
   \item \textbf{Accès à l'éducation} : hausse du capital humain des femmes $\to$ meilleure employabilité
   \item \textbf{Changements structurels} : passage d'une économie industrielle à une économie de services (moins d'exigences physiques)
\end{itemize}
Il s'agit donc d'une combinaison de facteurs d'offre (les femmes veulent et peuvent travailler) et de demande (les employeurs les embauchent).
\end{reponse}

\vfi{L'immigration fait toujours baisser les salaires.}

\begin{reponse}
\textbf{Faux.} L'analyse vue en cours distingue deux horizons :
\begin{itemize}
   \item \textbf{Court terme :} l'offre de travail augmente (courbe d'offre se déplace vers la droite) $\to$ pression à la baisse sur les salaires. C'est l'effet mécanique immédiat.
   \item \textbf{Long terme :} la baisse du ratio $K/L$ augmente le rendement marginal du capital $\to$ les entreprises investissent davantage $\to$ la demande de travail se déplace vers la droite $\to$ les salaires remontent à leur niveau initial.
\end{itemize}
Empiriquement, la plupart des études trouvent un effet \textbf{proche de zéro} sur les salaires à long terme. Le mécanisme d'ajustement du capital (inspiration du modèle de Solow) est la clé : le capital s'adapte à la main-d'{\oe}uvre, pas l'inverse.
\end{reponse}

\vfi{Le progrès technologique augmente toujours les inégalités.}

\begin{reponse}
\textbf{Faux / Incertain.} L'effet du progrès technologique sur les inégalités dépend du \textbf{type} de technologie :
\begin{itemize}
   \item \textbf{Technologie biaisée en faveur des qualifications (SBTC)} : augmente la demande de travailleurs qualifiés $\to$ hausse de la prime de qualification $\to$ inégalités $\uparrow$ (ex. : informatisation des années 1980--2000)
   \item \textbf{Effet de réinstauration} : certaines technologies créent de \textbf{nouvelles tâches} accessibles aux travailleurs peu qualifiés $\to$ inégalités $\downarrow$ (ex. : les tableurs ont démocratisé l'analyse financière)
   \item \textbf{Automatisation \guillemotleft~so-so~\guillemotright} : remplace des travailleurs sans créer assez de nouvelles tâches ni de gains de productivité substantiels $\to$ pire scénario pour les inégalités (ex. : caisses automatiques)
\end{itemize}
L'effet net dépend de la balance entre déplacement et réinstauration. Historiquement, la technologie a souvent augmenté les inégalités à court terme avant de les réduire (courbe en U inversé de Kuznets), mais ce n'est pas garanti.
\end{reponse}

\shortq{La population en âge de travailler d'une petite économie se décompose comme suit : 200\,000 employés, 20\,000 chômeurs, 80\,000 inactifs (population totale en âge de travailler : 300\,000).}

\begin{enumerate}[label=(\alph*)]
   \item Calculez les trois indicateurs clés du marché du travail : taux de participation, taux d'emploi et taux de chômage.
   \item Si 5\,000 travailleurs découragés cessent de chercher un emploi et deviennent inactifs, recalculez les trois indicateurs.
   \item Expliquez pourquoi le taux de chômage seul peut être trompeur comme indicateur de la santé du marché du travail.
\end{enumerate}

\begin{reponse}
\textbf{(a)} Population active $= E + U = 200\,000 + 20\,000 = 220\,000$.
\begin{itemize}
   \item Taux de participation $= \frac{E + U}{\text{Pop.}} = \frac{220\,000}{300\,000} = 73{,}3\,\%$
   \item Taux d'emploi $= \frac{E}{\text{Pop.}} = \frac{200\,000}{300\,000} = 66{,}7\,\%$
   \item Taux de chômage $= \frac{U}{E + U} = \frac{20\,000}{220\,000} = 9{,}1\,\%$
\end{itemize}

\textbf{(b)} Après le départ de 5\,000 découragés : $E = 200\,000$, $U = 15\,000$, $I = 85\,000$.
\begin{itemize}
   \item Taux de participation $= \frac{215\,000}{300\,000} = 71{,}7\,\%$ ($\downarrow$ de 1,6 pp)
   \item Taux d'emploi $= \frac{200\,000}{300\,000} = 66{,}7\,\%$ (\textbf{inchangé})
   \item Taux de chômage $= \frac{15\,000}{215\,000} = 7{,}0\,\%$ ($\downarrow$ de 2,1 pp)
\end{itemize}

\textbf{(c)} Le taux de chômage a \textbf{baissé} de 9,1\,\% à 7,0\,\%, ce qui semble positif. Mais aucun emploi n'a été créé! La baisse reflète uniquement le \textbf{retrait} de travailleurs découragés de la population active. Le taux d'emploi (inchangé) donne une image plus fidèle de la situation. C'est pourquoi les économistes regardent les \textbf{trois} indicateurs conjointement.
\end{reponse}

\shortq{Expliquez pourquoi la rigidité des salaires nominaux à la baisse crée du chômage en période de récession. Comment une inflation modérée peut-elle aider à résoudre ce problème? Utilisez le modèle d'offre et de demande de travail.}

\begin{reponse}
\textbf{Mécanisme :}
\begin{itemize}
   \item En récession, la demande de travail diminue (courbe de demande vers la gauche : baisse de la production $\to$ les entreprises ont besoin de moins de travailleurs).
   \item Le nouveau salaire d'équilibre ($w_1^*$) est inférieur à l'ancien ($w_0^*$).
   \item Mais les salaires nominaux sont \textbf{rigides à la baisse} : conventions collectives, salaire minimum, résistance psychologique des travailleurs à accepter une baisse de salaire.
   \item Au salaire $w_0^* > w_1^*$, la quantité demandée de travail ($L^D$) est inférieure à la quantité offerte ($L^S$) $\to$ chômage involontaire $= L^S - L^D$.
\end{itemize}

\textbf{Rôle de l'inflation modérée :}
\begin{itemize}
   \item Avec une inflation de 2--3\,\%, les entreprises peuvent réduire le \textbf{salaire réel} ($w/P$) sans baisser le salaire \textbf{nominal}.
   \item Exemple : si l'inflation est de 3\,\% et les salaires nominaux augmentent de 1\,\%, le salaire réel baisse de $\sim$2\,\%, rétablissant l'équilibre.
   \item C'est un argument central pour une cible d'inflation \textbf{légèrement positive} (2\,\%) plutôt que 0\,\% : elle \guillemotleft~lubrifie~\guillemotright\ l'ajustement du marché du travail.
\end{itemize}
\end{reponse}

\shortq{Identifiez et expliquez brièvement les trois principaux facteurs de la hausse des inégalités \textit{au sein} des pays développés depuis les années 1980. Pour chacun, donnez un exemple concret.}

\begin{reponse}
\textbf{1. La mondialisation et le choc chinois :}
\begin{itemize}
   \item L'intégration de la Chine dans le commerce mondial a créé une concurrence intense pour les travailleurs peu qualifiés du secteur manufacturier des pays riches
   \item Exemple : les travaux d'Autor, Dorn et Hanson montrent que les régions américaines les plus exposées aux importations chinoises ont subi des pertes d'emplois durables, des baisses de salaires et une hausse des problèmes sociaux
\end{itemize}

\textbf{2. Le progrès technologique biaisé en faveur des qualifications (SBTC) :}
\begin{itemize}
   \item L'informatisation et la numérisation ont augmenté la demande de travailleurs qualifiés (programmeurs, analystes, ingénieurs) et réduit la demande de tâches routinières (secrétariat, comptabilité manuelle, travail en usine)
   \item Exemple : la prime de qualification (écart de salaire université/non-université) a doublé aux États-Unis entre 1980 et 2020
\end{itemize}

\textbf{3. L'érosion des institutions :}
\begin{itemize}
   \item Déclin des syndicats, stagnation du salaire minimum réel, déréglementation du marché du travail
   \item Exemple : le taux de syndicalisation aux États-Unis est passé de $\sim$35\,\% en 1955 à $\sim$10\,\% aujourd'hui, réduisant le pouvoir de négociation des travailleurs à revenus moyens
\end{itemize}
\end{reponse}

\shortq{En vous appuyant sur le cadre d'Acemoglu et Restrepo, expliquez l'effet de déplacement et l'effet de réinstauration de l'automatisation. Donnez un exemple historique où l'effet de réinstauration a dominé. Pourquoi l'IA pourrait-elle être différente?}

\begin{reponse}
\textbf{Effet de déplacement :} L'automatisation remplace les travailleurs dans certaines tâches existantes $\to$ la demande de travail diminue pour ces tâches $\to$ pression à la baisse sur les salaires et l'emploi.

\textbf{Effet de réinstauration :} L'automatisation crée de \textbf{nouvelles tâches} dans lesquelles les humains ont un avantage comparatif. De plus, la baisse des coûts peut élargir le marché et créer de l'emploi ailleurs.

\textbf{Exemple historique --- les tableurs :} Le logiciel VisiCalc (1979) a automatisé les calculs des comptables. Effet de déplacement : moins besoin de commis-comptables pour les calculs manuels. Effet de réinstauration : la démocratisation de l'analyse financière a créé une explosion de la demande pour des analystes, consultants et managers utilisant des tableurs. L'emploi total dans la comptabilité et la finance a \textbf{augmenté}.

\textbf{Pourquoi l'IA pourrait être différente :}
\begin{itemize}
   \item L'IA touche des tâches \textbf{cognitives non routinières} (écriture, analyse, diagnostic), historiquement à l'abri de l'automatisation
   \item Le modèle de Moll, Rachel et Restrepo suggère qu'une IA suffisamment avancée pourrait augmenter la part du capital au détriment du travail
   \item Risque d'automatisation \guillemotleft~so-so~\guillemotright\ : si l'IA remplace des travailleurs sans gains de productivité substantiels, l'effet net sur l'emploi est négatif
   \item La vitesse du changement pourrait dépasser la capacité d'adaptation des travailleurs (la reconversion prend du temps)
\end{itemize}
\end{reponse}

\section{Exercices numériques}
% ============================================================================

\setcounter{shortcounter}{0}

\shortq{Les données suivantes décrivent le marché du travail d'un pays en 2020 et 2025 (en milliers de personnes) :}

\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l c c}
\toprule
& \textbf{2020} & \textbf{2025} \\
\midrule
Employés ($E$) & 18\,000 & 19\,500 \\
Chômeurs ($U$) & 2\,000 & 1\,200 \\
Inactifs ($I$) & 5\,000 & 6\,300 \\
Pop.\ en âge de travailler & 25\,000 & 27\,000 \\
\bottomrule
\end{tabular}
\end{center}

\begin{enumerate}[label=(\alph*)]
   \item Calculez le taux de participation, le taux d'emploi et le taux de chômage pour chaque année.
   \item Le marché du travail s'est-il amélioré entre 2020 et 2025? Justifiez en utilisant les \textbf{trois} indicateurs.
\end{enumerate}

\begin{reponse}
\textbf{(a)}

\textbf{2020 :}
\begin{itemize}
   \item Pop.\ active $= 18\,000 + 2\,000 = 20\,000$
   \item Taux de participation $= 20\,000 / 25\,000 = 80{,}0\,\%$
   \item Taux d'emploi $= 18\,000 / 25\,000 = 72{,}0\,\%$
   \item Taux de chômage $= 2\,000 / 20\,000 = 10{,}0\,\%$
\end{itemize}

\textbf{2025 :}
\begin{itemize}
   \item Pop.\ active $= 19\,500 + 1\,200 = 20\,700$
   \item Taux de participation $= 20\,700 / 27\,000 = 76{,}7\,\%$
   \item Taux d'emploi $= 19\,500 / 27\,000 = 72{,}2\,\%$
   \item Taux de chômage $= 1\,200 / 20\,700 = 5{,}8\,\%$
\end{itemize}

\textbf{(b)} Le bilan est \textbf{mitigé} :
\begin{itemize}
   \item \textbf{Taux de chômage :} forte amélioration (10,0\,\% $\to$ 5,8\,\%). Pris isolément, c'est très positif.
   \item \textbf{Taux d'emploi :} quasi stable (72,0\,\% $\to$ 72,2\,\%). Malgré la baisse du chômage, la proportion de la population qui travaille n'a presque pas changé.
   \item \textbf{Taux de participation :} en baisse (80,0\,\% $\to$ 76,7\,\%). La baisse du chômage s'explique en partie par le \textbf{retrait} de personnes de la population active (les inactifs sont passés de 5\,000 à 6\,300, soit +1\,300). Cela peut refléter le vieillissement (départs à la retraite) ou le découragement.
\end{itemize}
Le taux de chômage seul donne une image trop optimiste. L'emploi n'a progressé que de 1\,500 sur une croissance de la population de 2\,000, ce qui est insuffisant pour absorber les nouveaux arrivants sur le marché du travail.
\end{reponse}

\shortq{Les données suivantes montrent le salaire moyen des travailleurs avec et sans diplôme universitaire :}

\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l c c}
\toprule
& \textbf{1980} & \textbf{2020} \\
\midrule
Salaire moyen --- diplôme universitaire & 45\,000\,\$ & 75\,000\,\$ \\
Salaire moyen --- sans diplôme & 30\,000\,\$ & 35\,000\,\$ \\
\bottomrule
\end{tabular}
\end{center}

\begin{enumerate}[label=(\alph*)]
   \item Calculez la prime de qualification (en ratio et en pourcentage) pour 1980 et 2020.
   \item Comment la prime de qualification a-t-elle évolué? Calculez la variation en points de pourcentage.
   \item Expliquez cette évolution en utilisant le concept de progrès technologique biaisé en faveur des qualifications (SBTC).
\end{enumerate}

\begin{reponse}
\textbf{(a)} La prime de qualification $= \frac{w_{\text{qualifié}}}{w_{\text{non qualifié}}}$.
\begin{itemize}
   \item 1980 : $\frac{45\,000}{30\,000} = 1{,}50$, soit une prime de $50\,\%$
   \item 2020 : $\frac{75\,000}{35\,000} = 2{,}14$, soit une prime de $114\,\%$
\end{itemize}

\textbf{(b)} La prime de qualification est passée de 50\,\% à 114\,\%, soit une hausse de \textbf{64 points de pourcentage}. Les travailleurs diplômés gagnent maintenant plus du double des non-diplômés, contre 1,5 fois en 1980. De plus, les salaires des non-diplômés ont à peine augmenté ($+17\,\%$ en 40 ans, soit $\sim$0,4\,\% par an), tandis que ceux des diplômés ont bondi de $+67\,\%$.

\textbf{(c)} Le \textbf{SBTC} (progrès technologique biaisé en faveur des qualifications) explique cette divergence :
\begin{itemize}
   \item L'informatisation et la révolution numérique ont augmenté la productivité marginale des travailleurs \textbf{qualifiés} (complémentarité entre compétences et technologie) $\to$ la demande de travail qualifié augmente $\to$ hausse des salaires qualifiés
   \item Simultanément, la technologie a \textbf{substitué} les travailleurs dans les tâches routinières (secrétariat, comptabilité, travail en usine) $\to$ la demande de travail non qualifié stagne ou diminue
   \item L'offre de diplômés a augmenté (plus d'accès à l'université), mais pas assez pour compenser la hausse de la demande $\to$ l'équilibre se fait à un prix (prime) plus élevé
\end{itemize}
\end{reponse}

\qcm{Selon le modèle du marché du travail (offre et demande), le départ d'un grand nombre de résidents riches et dépensiers est le plus susceptible d'entraîner :}
\choix{
   \item Une augmentation des salaires réels et une baisse de l'emploi
   \item Une augmentation à la fois des salaires réels et de l'emploi
   \item Un impact ambigu sur les salaires réels et une baisse de l'emploi
   \item Une baisse des salaires réels, mais un impact ambigu sur l'emploi
}

\begin{reponse}
\textbf{(c)} Le départ de résidents riches et dépensiers a \textbf{deux} effets simultanés :
\begin{itemize}
   \item \textbf{Offre de travail :} les résidents qui partent réduisent l'offre de travail $\to$ courbe d'offre vers la gauche $\to$ pression à la \textbf{hausse} sur les salaires et à la \textbf{baisse} sur l'emploi
   \item \textbf{Demande de travail :} ces résidents étaient des \guillemotleft~dépensiers~\guillemotright, donc leur départ réduit la demande de biens et services $\to$ les entreprises ont besoin de moins de travailleurs $\to$ courbe de demande vers la gauche $\to$ pression à la \textbf{baisse} sur les salaires et l'emploi
\end{itemize}
L'emploi diminue dans les deux cas, mais l'effet sur les salaires est ambigu (hausse via l'offre, baisse via la demande).
\end{reponse}

\qcm{Alors qu'une profonde récession s'installait en Espagne après 2009, un grand nombre de résidents adultes sans emploi ont émigré vers d'autres pays. L'effet de cette émigration \textbf{seule} est :}
\choix{
   \item Une baisse du taux d'activité
   \item Une baisse du taux d'emploi
   \item Une hausse du taux de chômage
   \item Toutes les réponses ci-dessus
   \item Aucun des effets ci-dessus
}

\begin{reponse}
\textbf{(a)} Les émigrants sont des adultes \textbf{sans emploi} qui quittent le pays. S'ils étaient chômeurs (cherchaient un emploi), leur départ réduit à la fois le numérateur ($U$) et le dénominateur ($E+U$) du taux de chômage, et réduit la population active par rapport à la population $\to$ le taux d'activité (participation) \textbf{baisse}. Le taux de chômage baisse aussi (même mécanisme que les travailleurs découragés). Le taux d'emploi ($E/\text{Pop}$) pourrait rester stable si la population totale diminue proportionnellement.
\end{reponse}

\qcm{Dans le modèle du marché du travail, lequel des éléments suivants devrait entraîner une \textbf{baisse de l'emploi} et une \textbf{hausse du salaire réel} à l'équilibre?}
\choix{
   \item Une baisse de l'offre de main-d'{\oe}uvre
   \item Une augmentation de l'offre de main-d'{\oe}uvre
   \item Une baisse de la demande de main-d'{\oe}uvre
   \item Une augmentation de la demande de main-d'{\oe}uvre
}

\begin{reponse}
\textbf{(a)} Pour que l'emploi baisse \textbf{et} le salaire augmente simultanément, il faut un déplacement de la courbe d'\textbf{offre} vers la \textbf{gauche}. Quand l'offre de travail diminue (ex. : départ à la retraite, émigration), il y a moins de travailleurs disponibles $\to$ les entreprises doivent offrir des salaires plus élevés pour attirer les travailleurs restants, mais l'emploi total diminue. (c) donnerait emploi$\downarrow$ et salaire$\downarrow$; (b) et (d) donneraient emploi$\uparrow$.
\end{reponse}

\shortq{L'année 2020 a été marquée par des bouleversements sur les marchés du travail. Considérez un diagramme offre-demande de travail (salaire réel $w$ en ordonnée, emploi $N$ en abscisse) où l'économie est à l'équilibre en 2019. En 2020, la demande de travail se déplace vers la gauche (les entreprises ferment) et l'offre se déplace aussi vers la gauche (les travailleurs ne peuvent pas se rendre au travail).}

\begin{enumerate}[label=(\alph*)]
   \item Parmi les scénarios suivants, lequel n'est \textbf{pas} compatible avec cette évolution? (i) Les ménages réduisent leurs dépenses. (ii) Les entreprises ferment à cause des mesures sanitaires. (iii) Les travailleurs veulent faire des heures supplémentaires par peur de la pandémie. (iv) Les travailleurs âgés prennent une retraite anticipée.
   \item Le gouvernement met en place des \textbf{subventions salariales} (il paie une partie des salaires). Montrez comment cela affecte le diagramme.
   \item Le gouvernement interdit aux entreprises de baisser les salaires en dessous du niveau de 2019. Comparez cette politique de \guillemotleft~salaire minimum~\guillemotright\ avec les subventions salariales.
\end{enumerate}

\begin{reponse}
\textbf{(a)} Le scénario \textbf{(iii)} n'est pas compatible. Si les travailleurs veulent faire \textbf{plus} d'heures, l'offre de travail se déplace vers la \textbf{droite}, pas vers la gauche. Les scénarios (i) et (ii) sont des chocs de demande (déplacement de $N^D$ vers la gauche). Le scénario (iv) est un choc d'offre (déplacement de $N^S$ vers la gauche).

\textbf{(b)} Les subventions salariales réduisent le coût du travail pour les entreprises $\to$ la courbe de demande $N^D$ se déplace vers la \textbf{droite}. Si la subvention est suffisante, $N^D$ revient au niveau de 2019. Le salaire réel remonte vers $(w/P)_{2019}$, et l'emploi augmente. Cependant, comme l'offre $N^S$ a aussi bougé vers la gauche, l'emploi ne revient pas complètement à $N_{2019}$.

\textbf{(c)} Interdire la baisse des salaires en dessous de $(w/P)_{2019}$ revient à imposer un \textbf{salaire minimum} au niveau de 2019. La demande de travail au niveau de 2020 est plus faible $\to$ au salaire $(w/P)_{2019}$, la quantité demandée est bien inférieure à la quantité offerte. L'emploi est déterminé par la demande : $N < N_{2020} < N_{2019}$. Cette politique est \textbf{moins efficace} que les subventions salariales car elle maintient le salaire mais au prix d'un emploi encore plus faible. Le revenu \textbf{total} du travail ($w \times N$) est plus faible qu'avec les subventions.
\end{reponse}

\qcm{Considérons une situation dans laquelle de nombreux chômeurs se découragent dans leur recherche d'emploi et décident de quitter le marché du travail. Quel est l'impact sur le taux de chômage et le taux d'activité?}
\choix{
   \item Le taux de chômage augmente
   \item Le taux de chômage diminue
   \item Le taux d'activité diminue
   \item Les réponses (b) et (c)
}

\begin{reponse}
\textbf{(d)} Les chômeurs découragés quittent la population active. Ils ne sont plus comptés ni comme employés ni comme chômeurs. Effets :
\begin{itemize}
   \item Le taux de chômage \textbf{diminue} car le numérateur ($U$) et le dénominateur ($E+U$) baissent tous les deux, mais $U$ baisse proportionnellement plus.
   \item Le taux d'activité (participation) \textbf{diminue} car la population active diminue par rapport à la population en âge de travailler.
\end{itemize}
C'est une des limites du taux de chômage comme indicateur : il peut baisser pour de \guillemotleft~mauvaises raisons~\guillemotright.
\end{reponse}

\shortq{Les politiciens américains se sont souvent plaints des salaires bas au Mexique. Afin de conclure un accord commercial, le Mexique est contraint d'augmenter considérablement son salaire minimum. À l'aide du modèle offre-demande du marché du travail, montrez l'impact de cette politique sur les trois principaux indicateurs du marché du travail mexicain à court terme.}

\begin{reponse}
Le salaire minimum plus élevé se situe au-dessus du salaire d'équilibre. Dans le graphique offre-demande du marché du travail :
\begin{itemize}
   \item Le salaire plus élevé \textbf{incite davantage de personnes} à chercher un emploi (mouvement vers le haut le long de la courbe d'offre $N^S$) $\to$ le \textbf{taux d'activité augmente}
   \item Au salaire minimum, la quantité demandée par les entreprises est \textbf{inférieure} à la quantité offerte $\to$ le \textbf{taux d'emploi baisse}
   \item L'écart entre l'offre et la demande crée du \textbf{chômage} $\to$ le \textbf{taux de chômage augmente}
\end{itemize}
En résumé : taux de chômage $\uparrow$, taux d'emploi $\downarrow$, taux d'activité $\uparrow$.
\end{reponse}
