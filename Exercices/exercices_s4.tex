\section{Questions à choix multiples}
% ============================================================================

\setcounter{qcmcounter}{0}

\qcm{Dans le modèle AD-AS, un choc de demande \textbf{positif} (par exemple, une hausse de la confiance des consommateurs) entraîne :}
\choix{
   \item Une hausse du PIB et une baisse du niveau des prix
   \item Une hausse du PIB et une hausse du niveau des prix
   \item Une baisse du PIB et une hausse du niveau des prix
   \item Aucun changement du PIB ni du niveau des prix
}

\begin{reponse}
\textbf{(b)} Un choc de demande positif déplace la courbe AD vers la \textbf{droite}. Le long de la courbe AS (à pente positive en raison de la rigidité des salaires), le nouvel équilibre se situe à un PIB \textbf{plus élevé} et un niveau des prix \textbf{plus élevé}. Intuitivement : plus de dépenses $\to$ plus de production, mais aussi plus de pression sur les prix.
\end{reponse}

\qcm{La courbe d'offre agrégée (AS) est croissante à court terme parce que :}
\choix{
   \item Les consommateurs achètent plus quand les prix baissent
   \item Les salaires nominaux sont rigides à court terme
   \item Le gouvernement augmente ses dépenses quand les prix montent
   \item Les exportations augmentent avec le niveau des prix
}

\begin{reponse}
\textbf{(b)} La courbe AS est croissante à court terme en raison de la \textbf{rigidité des salaires nominaux}. Quand le niveau des prix ($P$) augmente mais que les salaires restent fixes, le coût réel du travail pour les entreprises diminue $\to$ elles embauchent plus et produisent davantage. C'est la raison pour laquelle une hausse de $P$ est associée à une hausse de $Y$ le long de AS.
\end{reponse}

\qcm{La pandémie de 2020 est un exemple de :}
\choix{
   \item Un choc de demande positif uniquement
   \item Un choc d'offre négatif uniquement
   \item Un choc simultané de demande et d'offre négatifs
   \item Un choc de demande négatif suivi d'un choc d'offre positif
}

\begin{reponse}
\textbf{(c)} La pandémie de 2020 est un cas rare de choc \textbf{simultané} d'offre et de demande. Du côté de l'offre : les confinements ont réduit la capacité de production (AS vers la gauche). Du côté de la demande : les consommateurs ne pouvaient pas ou ne voulaient pas dépenser (AD vers la gauche). L'effet net a été une chute massive du PIB, avec un effet ambigu sur les prix (pression désinflationniste initiale, puis forte inflation en 2021--2022 lorsque la demande a repris avant l'offre).
\end{reponse}

\qcm{Laquelle des affirmations suivantes décrit correctement le mécanisme d'\textbf{autocorrection} de l'économie dans le modèle AD-AS?}
\choix{
   \item En récession, le gouvernement réduit automatiquement les impôts
   \item En récession, la pression salariale diminue, AS se déplace vers la droite et l'économie revient vers son potentiel
   \item En surchauffe, la banque centrale augmente automatiquement le taux directeur
   \item En surchauffe, les entreprises réduisent automatiquement leur production
}

\begin{reponse}
\textbf{(b)} Le mécanisme d'autocorrection fonctionne via l'ajustement des salaires. En récession ($Y < Y^{\text{POT}}$), le chômage augmente $\to$ la pression salariale diminue $\to$ les coûts de production baissent $\to$ la courbe AS se déplace vers la droite $\to$ les prix baissent et la production remonte vers le potentiel. Ce processus est \textbf{lent}, ce qui justifie l'intervention des politiques contracycliques. (a) et (c) décrivent des interventions \textit{discrétionnaires}, pas automatiques.
\end{reponse}

\qcm{La courbe de demande agrégée (AD) a une pente négative parce que :}
\choix{
   \item Les entreprises produisent moins quand les prix augmentent
   \item Une hausse du niveau des prix réduit la richesse réelle des ménages, ce qui diminue la consommation
   \item Le gouvernement réduit ses dépenses quand les prix augmentent
   \item Les salaires augmentent avec le niveau des prix
}

\begin{reponse}
\textbf{(b)} La pente négative de la courbe AD s'explique principalement par l'\textbf{effet de richesse} : quand le niveau des prix ($P$) augmente, la valeur réelle des actifs des ménages (épargne, obligations) diminue $\to$ les ménages se sentent moins riches $\to$ ils réduisent leur consommation ($C$) $\to$ la demande agrégée diminue. D'autres canaux renforcent cet effet (effet taux d'intérêt, effet taux de change pour une économie ouverte). (a) décrit le côté offre, pas la demande.
\end{reponse}

\qcm{À long terme, la courbe d'offre agrégée (AS) est verticale parce que :}
\choix{
   \item Les prix sont fixes à long terme
   \item Les salaires s'ajustent pleinement aux variations de prix, ramenant la production à $Y^{\text{POT}}$
   \item Le gouvernement stabilise toujours la production
   \item La technologie ne change pas à long terme
}

\begin{reponse}
\textbf{(b)} À court terme, les salaires sont rigides, ce qui permet aux entreprises de produire plus quand $P$ augmente (AS croissante). Mais à \textbf{long terme}, les salaires s'ajustent pleinement à la hausse des prix : les travailleurs renégocient leurs salaires $\to$ les coûts de production augmentent proportionnellement $\to$ les entreprises n'ont plus d'incitation à produire au-delà du potentiel. La production revient à $Y^{\text{POT}}$, quel que soit le niveau des prix $\to$ AS est \textbf{verticale}.
\end{reponse}

\qcm{La \guillemotleft~stagflation~\guillemotright\ désigne une situation où :}
\choix{
   \item L'inflation et la croissance économique sont élevées simultanément
   \item L'inflation augmente et le chômage augmente simultanément
   \item L'inflation et le chômage diminuent simultanément
   \item Le niveau des prix est stable mais la production chute
}

\begin{reponse}
\textbf{(b)} La stagflation est la combinaison de \textbf{stagnation} économique (PIB en baisse, chômage en hausse) et d'\textbf{inflation} (prix en hausse). C'est typiquement le résultat d'un \textbf{choc d'offre négatif} (AS vers la gauche) : les coûts de production augmentent $\to$ les entreprises produisent moins (Y$\downarrow$) et les prix montent (P$\uparrow$). L'exemple classique est le choc pétrolier de 1973. La stagflation est particulièrement difficile à combattre car les politiques contracycliques traditionnelles ne peuvent pas résoudre les deux problèmes simultanément.
\end{reponse}

\qcm{Un choc de demande négatif entraîne $P\downarrow$ et $Y\downarrow$. Un choc d'offre négatif entraîne :}
\choix{
   \item $P\downarrow$ et $Y\downarrow$
   \item $P\uparrow$ et $Y\uparrow$
   \item $P\uparrow$ et $Y\downarrow$
   \item $P\downarrow$ et $Y\uparrow$
}

\begin{reponse}
\textbf{(c)} Les deux types de chocs ont des effets différents sur les prix :
\begin{itemize}
   \item \textbf{Choc de demande négatif} (AD vers la gauche) : la demande baisse $\to$ $Y\downarrow$ \textbf{et} $P\downarrow$ (les deux vont dans le même sens)
   \item \textbf{Choc d'offre négatif} (AS vers la gauche) : les coûts augmentent $\to$ $Y\downarrow$ \textbf{mais} $P\uparrow$ (les deux vont en sens contraire)
\end{itemize}
Cette distinction est cruciale pour la politique économique : face à un choc de demande, baisser les taux aide (les deux objectifs sont alignés); face à un choc d'offre, la banque centrale fait face à un dilemme.
\end{reponse}

\qcm{Dans le cadre en quatre étapes de l'anatomie d'une récession, l'\guillemotleft~étincelle~\guillemotright\ de la récession de 2020 était :}
\choix{
   \item Une erreur de politique monétaire
   \item Un déséquilibre structurel accumulé sur plusieurs années
   \item Un choc exogène (la pandémie de COVID-19)
   \item Un effondrement du marché immobilier
}

\begin{reponse}
\textbf{(c)} Le cadre en quatre étapes distingue : (1) l'\textbf{étincelle} (le déclencheur initial), (2) la \textbf{transmission} (les canaux de propagation), (3) le \textbf{carburant} (les amplificateurs), (4) les \textbf{pare-feu} (la réponse politique). Pour 2020, l'étincelle était un \textbf{choc exogène} pur --- la pandémie de COVID-19. Contrairement à 2008 (déséquilibre structurel : bulle immobilière + levier financier excessif), la récession de 2020 n'a pas été causée par un faux pas de politique économique.
\end{reponse}

\setcounter{shortcounter}{0}

\shortq{Analysez la récession de 2020 à l'aide du cadre en quatre étapes : (1) étincelle, (2) transmission, (3) carburant, (4) pare-feu.}

\begin{reponse}
\textbf{(1) Étincelle :} La pandémie de COVID-19 (choc exogène de type \guillemotleft~choc~\guillemotright). Ni un faux pas de politique économique, ni un déséquilibre structurel.

\textbf{(2) Transmission :} Double canal. Du côté de l'offre : les confinements ferment les entreprises et perturbent les chaînes d'approvisionnement (AS $\leftarrow$). Du côté de la demande : les consommateurs ne peuvent pas consommer de services (restaurants, voyages) et réduisent leurs dépenses par précaution (AD $\leftarrow$).

\textbf{(3) Carburant (amplificateurs) :} L'incertitude massive renforce la contraction : les entreprises reportent leurs investissements, les ménages épargnent davantage (motif de précaution), les mises à pied créent une spirale de baisse de revenus et de consommation.

\textbf{(4) Pare-feu (réponse politique) :} Réponse sans précédent. Politique monétaire : baisse des taux à $\sim$0\,\% et programmes d'achat d'actifs. Politique budgétaire : transferts directs aux ménages (PCU au Canada, chèques de relance aux É.-U.), subventions salariales (SSUC), hausse massive de $G$. Ces mesures ont limité la profondeur et la durée de la récession, mais ont contribué à la vague inflationniste de 2021--2022.
\end{reponse}

\vfi{L'inflation est toujours et partout causée par un excès de dépenses publiques.}

\begin{reponse}
\textbf{Faux.} L'inflation peut provenir de sources variées, pas uniquement des dépenses publiques :
\begin{itemize}
   \item \textbf{Choc de demande} (pas nécessairement public) : une hausse de la confiance des consommateurs, une bulle de crédit privé, ou une politique monétaire trop accommodante peuvent toutes déplacer AD vers la droite $\to$ inflation
   \item \textbf{Choc d'offre négatif} : une hausse du prix du pétrole, une pandémie, ou des perturbations des chaînes d'approvisionnement déplacent AS vers la gauche $\to$ inflation (sans aucun lien avec les dépenses publiques)
\end{itemize}
La célèbre phrase de Milton Friedman --- \guillemotleft~l'inflation est toujours et partout un phénomène \textbf{monétaire}~\guillemotright\ --- fait référence à la politique \textbf{monétaire}, pas budgétaire. Et même cette affirmation est nuancée : l'inflation de 2021--2022 a largement résulté de chocs d'offre (chaînes d'approvisionnement, énergie).
\end{reponse}

\vfi{Une récession se termine toujours d'elle-même grâce au mécanisme d'autocorrection, sans intervention de politique économique.}

\begin{reponse}
\textbf{Incertain.} Le mécanisme d'autocorrection existe : en récession ($Y < Y^{\text{POT}}$), le chômage augmente $\to$ la pression salariale diminue $\to$ les coûts de production baissent $\to$ AS se déplace vers la droite $\to$ $Y$ remonte vers $Y^{\text{POT}}$.

Cependant :
\begin{itemize}
   \item Ce processus est très \textbf{lent} (les salaires s'ajustent lentement, surtout à la baisse)
   \item Pendant l'ajustement, le chômage reste élevé et les travailleurs souffrent de pertes de capital humain durables
   \item Des \textbf{effets d'hystérèse} peuvent survenir : une récession prolongée peut réduire le potentiel productif lui-même ($Y^{\text{POT}} \downarrow$)
\end{itemize}
C'est pourquoi les politiques contracycliques (monétaire et budgétaire) sont justifiées : non pas parce que l'économie ne peut pas se corriger seule, mais parce que l'attendre passivement est coûteux.
\end{reponse}

\shortq{Décrivez, à l'aide du modèle AD-AS, les effets d'un choc de demande négatif et d'un choc d'offre négatif. Pour chacun, identifiez l'effet sur $Y$ et $P$. Expliquez pourquoi les banques centrales trouvent les chocs d'offre plus difficiles à gérer.}

\begin{reponse}
\textbf{Choc de demande négatif} (ex. : baisse de la confiance des consommateurs) :
\begin{itemize}
   \item AD se déplace vers la \textbf{gauche}
   \item Effet : $Y \downarrow$ et $P \downarrow$
   \item La banque centrale peut baisser les taux pour déplacer AD vers la droite $\to$ cela ramène $Y$ et $P$ vers leurs niveaux initiaux. Les deux objectifs (stabiliser la production et les prix) sont \textbf{alignés}.
\end{itemize}

\textbf{Choc d'offre négatif} (ex. : hausse du prix du pétrole) :
\begin{itemize}
   \item AS se déplace vers la \textbf{gauche}
   \item Effet : $Y \downarrow$ et $P \uparrow$ (stagflation)
   \item Dilemme pour la banque centrale : stimuler AD (baisser les taux) pour soutenir $Y$ \textbf{aggrave} l'inflation; combattre l'inflation (hausser les taux) \textbf{aggrave} la récession. Les deux objectifs sont en \textbf{conflit}.
\end{itemize}

C'est pourquoi les chocs d'offre sont le cauchemar des banques centrales. Les chocs pétroliers des années 1970 et la poussée inflationniste post-COVID (en partie un choc d'offre) illustrent cette difficulté.
\end{reponse}

\shortq{Expliquez pourquoi la courbe AS est verticale à long terme mais croissante à court terme. Que se passe-t-il après un choc de demande positif à court terme, puis à long terme?}

\begin{reponse}
\textbf{Court terme --- AS croissante :} Les salaires nominaux sont fixés par des contrats et des conventions. Quand $P$ augmente, les salaires restent temporairement fixes $\to$ le coût réel du travail pour les entreprises diminue $\to$ elles embauchent et produisent davantage. D'où la relation positive entre $P$ et $Y$.

\textbf{Long terme --- AS verticale :} Les travailleurs renégocient leurs salaires pour rattraper la hausse des prix. Quand les salaires nominaux s'ajustent pleinement, le coût réel du travail revient à son niveau initial $\to$ la production revient à $Y^{\text{POT}}$, indépendamment de $P$.

\textbf{Après un choc de demande positif :}
\begin{enumerate}
   \item \textbf{Court terme :} AD se déplace vers la droite $\to$ l'économie se déplace le long de AS de court terme $\to$ $Y$ augmente au-delà de $Y^{\text{POT}}$ et $P$ augmente.
   \item \textbf{Moyen terme :} Les travailleurs constatent que les prix ont augmenté et renégocient des salaires plus élevés $\to$ les coûts de production augmentent $\to$ AS de court terme se déplace vers la \textbf{gauche} (vers le haut).
   \item \textbf{Long terme :} $Y$ revient à $Y^{\text{POT}}$, mais le niveau des prix est \textbf{plus élevé} qu'avant. Seul l'effet sur les prix est permanent; l'effet sur la production est temporaire.
\end{enumerate}
\end{reponse}

\shortq{Pour chacun des événements suivants, déterminez s'il s'agit principalement d'un choc de demande ou d'un choc d'offre, et dans quelle direction (positif ou négatif).}

\begin{enumerate}[label=(\alph*)]
   \item Les tarifs de Trump sur les importations chinoises (2018)
   \item Les confinements liés à la COVID-19 (2020)
   \item Un effondrement de la confiance des consommateurs
   \item La découverte d'un important gisement pétrolier
   \item Un programme de relance budgétaire massif
\end{enumerate}

\begin{reponse}
\textbf{(a) Tarifs de Trump :} Principalement un \textbf{choc d'offre négatif}. Les tarifs augmentent le coût des intrants importés pour les entreprises américaines $\to$ AS vers la gauche. Il y a aussi un élément de choc de demande négatif (incertitude $\to$ baisse de $I$), mais l'effet principal passe par les coûts. Effet : $Y\downarrow$, $P\uparrow$.

\textbf{(b) Confinements COVID-19 :} Choc \textbf{simultané} d'offre et de demande négatifs. Offre : fermeture des entreprises, perturbation des chaînes d'approvisionnement (AS $\leftarrow$). Demande : impossibilité de consommer, épargne de précaution (AD $\leftarrow$). Effet : $Y\downarrow$ fortement, effet ambigu sur $P$.

\textbf{(c) Effondrement de la confiance :} \textbf{Choc de demande négatif}. Les ménages réduisent $C$ par précaution, les entreprises reportent $I$ $\to$ AD vers la gauche. Effet : $Y\downarrow$, $P\downarrow$.

\textbf{(d) Découverte d'un gisement pétrolier :} \textbf{Choc d'offre positif}. La hausse de la production d'énergie réduit les coûts de production $\to$ AS vers la droite. Effet : $Y\uparrow$, $P\downarrow$.

\textbf{(e) Relance budgétaire :} \textbf{Choc de demande positif}. La hausse de $G$ (ou la baisse de $T$) augmente directement la demande agrégée $\to$ AD vers la droite. Effet : $Y\uparrow$, $P\uparrow$.
\end{reponse}

\shortq{Une économie est initialement en équilibre de long terme avec $Y = Y^{\text{POT}} = 100$ et $P = 100$.}

\begin{enumerate}[label=(\alph*)]
   \item Un choc de demande négatif réduit les dépenses souhaitées de 10\,\% à chaque niveau de prix. Expliquez qualitativement ce qui arrive à $Y$ et $P$ à court terme.
   \item Expliquez le mécanisme par lequel l'économie retourne à $Y^{\text{POT}}$ à long terme, \textbf{sans} intervention de politique économique.
   \item Pourquoi ce mécanisme d'autocorrection justifie-t-il quand même l'intervention des autorités?
\end{enumerate}

\begin{reponse}
\textbf{(a)} La courbe AD se déplace vers la \textbf{gauche} (de $\text{AD}_0$ à $\text{AD}_1$). Le long de la courbe AS de court terme (croissante), le nouvel équilibre se situe à un PIB inférieur ($Y_1 < 100$) et un niveau des prix inférieur ($P_1 < 100$). L'économie est en \textbf{récession} : $Y < Y^{\text{POT}}$, le chômage augmente.

\textbf{(b)} Mécanisme d'autocorrection :
\begin{enumerate}
   \item En récession, le chômage est élevé $\to$ les travailleurs acceptent des salaires nominaux plus bas (ou des hausses plus faibles)
   \item La baisse des salaires réduit les coûts de production $\to$ AS de court terme se déplace vers la \textbf{droite}
   \item Le déplacement de AS fait baisser les prix et augmenter la production
   \item Le processus continue jusqu'à ce que $Y = Y^{\text{POT}}$ (au nouvel état stationnaire, le niveau des prix est plus bas)
\end{enumerate}

\textbf{(c)} Bien que l'économie se corrige d'elle-même, le processus est très \textbf{lent} (les salaires s'ajustent avec retard, surtout à la baisse). Pendant ce temps, le chômage reste élevé, les travailleurs perdent du capital humain, et des effets d'hystérèse peuvent s'installer. Les politiques contracycliques (baisse des taux, hausse de $G$) accélèrent le retour à $Y^{\text{POT}}$ et limitent le coût humain de la récession.
\end{reponse}

\shortq{Appliquez le cadre en quatre étapes (étincelle, transmission, carburant, pare-feu) au choc pétrolier de 1973 (embargo de l'OPEP).}

\begin{reponse}
\textbf{(1) Étincelle :} L'embargo pétrolier de l'OPEP en octobre 1973, en réponse au soutien occidental à Israël lors de la guerre du Kippour. Le prix du pétrole a \textbf{quadruplé} en quelques mois (de $\sim$3\,\$ à $\sim$12\,\$ le baril).

\textbf{(2) Transmission :} C'est un \textbf{choc d'offre négatif} massif. Le pétrole est un intrant dans presque tous les secteurs (transport, industrie, chauffage, chimie). La hausse du prix du pétrole $\to$ hausse généralisée des coûts de production $\to$ AS vers la gauche $\to$ stagflation ($Y\downarrow$, $P\uparrow$).

\textbf{(3) Carburant (amplificateurs) :}
\begin{itemize}
   \item Spirale salaires-prix : les travailleurs demandent des hausses de salaire pour compenser l'inflation $\to$ les coûts augmentent davantage $\to$ plus d'inflation
   \item Panique et incertitude : les entreprises reportent les investissements, les ménages réduisent leur consommation
   \item Politique monétaire trop accommodante dans un premier temps (la Fed a tardé à réagir)
\end{itemize}

\textbf{(4) Pare-feu (réponse politique) :}
\begin{itemize}
   \item La réponse initiale a été \textbf{insuffisante} : les gouvernements ont tenté des contrôles de prix (inefficaces) et des rationnements d'essence
   \item La Fed de Paul Volcker (à partir de 1979, après le second choc) a finalement brisé l'inflation par une hausse massive des taux d'intérêt, au prix d'une récession sévère en 1981--1982
   \item Leçon : face à un choc d'offre, il n'y a pas de solution indolore. Accepter temporairement l'inflation ou provoquer une récession pour la combattre
\end{itemize}
\end{reponse}

\section{Questions d'examens antérieurs}
% ============================================================================

\setcounter{qcmcounter}{0}
\setcounter{shortcounter}{0}

\qcm{Historiquement, les économistes considèrent que les récessions ont principalement été causées par des chocs d'offre plutôt que de demande.}
\choix{
   \item Vrai
   \item Faux
}

\begin{reponse}
\textbf{(b) Faux.} La plupart des récessions ont été causées par des chocs de \textbf{demande} (effondrement de la confiance, resserrement monétaire excessif, crise financière). Les chocs d'offre purs (comme les chocs pétroliers de 1973 et 1979) sont moins fréquents. La récession de 2020 est inhabituelle car elle combine les deux types de chocs. La récession de 2008 est un choc de demande classique (crise financière $\to$ effondrement du crédit $\to$ AD vers la gauche).
\end{reponse}

\shortq{Au printemps 2020, trois mécanismes de transmission de la récession ont été évoqués : (1) les ménages réduisent leurs dépenses; (2) les entreprises ferment à cause des mesures sanitaires; (3) les entreprises réduisent leurs investissements par incertitude.}

\begin{enumerate}[label=(\alph*)]
   \item À l'aide de trois diagrammes AD-AS, montrez l'impact de chaque scénario sur l'économie à court terme. Partez d'un équilibre initial ($Y \approx Y^{\text{POT}}$).
   \item En observant que l'inflation a \textbf{baissé} début 2020, lequel de ces scénarios peut-on écarter comme mécanisme \textbf{principal}?
\end{enumerate}

\begin{reponse}
\textbf{(a)}
\begin{itemize}
   \item \textbf{Scénario 1} (ménages réduisent $C$) : choc de demande négatif. AD vers la gauche $\to$ $Y\downarrow$ et $P\downarrow$.
   \item \textbf{Scénario 2} (fermetures sanitaires) : choc d'offre négatif. AS vers la gauche $\to$ $Y\downarrow$ et $P\uparrow$.
   \item \textbf{Scénario 3} (incertitude $\to$ $I\downarrow$) : choc de demande négatif. AD vers la gauche $\to$ $Y\downarrow$ et $P\downarrow$.
\end{itemize}

\textbf{(b)} On peut écarter le scénario \textbf{2} comme mécanisme principal. Un choc d'offre pur (AS vers la gauche) entraîne une \textbf{hausse} des prix. Or, début 2020, l'inflation a \textbf{baissé}, ce qui est cohérent avec un choc de demande (scénarios 1 et 3). En réalité, la récession de 2020 était un mélange des trois, mais les chocs de demande ont dominé dans les premiers mois.
\end{reponse}

\shortq{Entre mi-2020 et mi-2022, l'écart de production s'est presque comblé, mais l'inflation a fortement augmenté. Comment expliquer ces observations avec le modèle AD-AS?}

\begin{reponse}
La meilleure explication combine un déplacement massif de AD vers la \textbf{droite} avec un déplacement de AS vers la \textbf{gauche} (ou un retour incomplet vers la droite).

\textbf{AD vers la droite :} Les plans de relance massifs (transferts aux ménages, subventions salariales) et les taux d'intérêt à $\sim$0\,\% ont fortement stimulé la demande agrégée. AD s'est déplacée bien au-delà de sa position pré-pandémique.

\textbf{AS vers la gauche :} Les perturbations des chaînes d'approvisionnement, la guerre en Ukraine (énergie), et le retour lent des travailleurs ont maintenu AS à gauche de sa position pré-pandémique.

\textbf{Résultat :} Le nouveau point d'intersection donne un $Y$ proche de $Y^{\text{POT}}$ (écart de production comblé) mais un $P$ nettement plus élevé (inflation). C'est le cas classique où la demande a augmenté \textbf{plus} que l'offre, de sorte que $Y$ a augmenté mais les prix aussi.
\end{reponse}

\shortq{Voici un extrait : \guillemotleft~Le pire scénario pour la Fed serait que des événements géopolitiques entraînent une nouvelle hausse brutale des prix du pétrole. Cela pourrait contraindre les décideurs à choisir entre provoquer une récession ou accepter une hausse rapide des prix.~\guillemotright\ Expliquez à l'aide d'un diagramme AD-AS.}

\begin{reponse}
Une hausse brutale des prix du pétrole est un \textbf{choc d'offre négatif} : AS se déplace vers la gauche $\to$ $Y\downarrow$ et $P\uparrow$ (stagflation).

La Fed fait face à un \textbf{dilemme} :
\begin{itemize}
   \item \textbf{Option 1 --- combattre l'inflation :} hausser les taux $\to$ AD vers la gauche $\to$ $P$ redescend, mais $Y$ chute encore plus $\to$ \textbf{récession profonde}
   \item \textbf{Option 2 --- soutenir la production :} maintenir ou baisser les taux $\to$ AD reste en place ou va vers la droite $\to$ $Y$ est soutenu, mais $P$ augmente davantage $\to$ \textbf{inflation élevée}
\end{itemize}
C'est le \guillemotleft~pire scénario~\guillemotright\ car les deux objectifs de la banque centrale (stabilité des prix et plein emploi) sont en \textbf{conflit}. Lors d'un choc de demande, les deux objectifs sont alignés; lors d'un choc d'offre, ils s'opposent.
\end{reponse}

\vfi{Une récession correspond à deux trimestres de croissance négative du PIB réel.}

\begin{reponse}
\textbf{Faux.} Il n'y a pas de définition stricte d'une récession au-delà d'une faiblesse généralisée de l'économie. Aux États-Unis, une récession est déterminée par un groupe d'économistes du NBER, sur la base de plusieurs indicateurs : revenus, emploi, production industrielle, ventes, etc. La \guillemotleft~règle des deux trimestres~\guillemotright\ est une approximation populaire mais pas une définition officielle.
\end{reponse}

\shortq{Le FMI a abaissé ses prévisions de croissance mondiale à 2.2\,\% pour 2009, bien en dessous du seuil de 3\,\% traditionnellement considéré comme le seuil d'une récession mondiale. L'économiste en chef du FMI, Olivier Blanchard, a réitéré qu'il ne croit pas que ce seuil de 3\,\% soit une définition adéquate d'une récession, considérant plutôt une croissance négative comme un indicateur plus approprié.}

\begin{enumerate}[label=(\alph*)]
   \item Expliquez brièvement pourquoi Blanchard considère une croissance négative comme une définition plus adéquate.
   \item Expliquez pourquoi une croissance mondiale sous le seuil de 3\,\% pourrait être associée à un ralentissement économique mondial important, alors qu'un taux de croissance de 3\,\% au Canada serait considéré comme une expansion vigoureuse.
\end{enumerate}

\begin{reponse}
\textbf{(a)} Conceptuellement, une récession représente une réduction du niveau de l'activité économique réelle. Donc une récession devrait être associée à une baisse de l'output. Blanchard ne fait que répéter la définition conceptuelle.

\textbf{(b)} Les économies émergentes représentent une \guillemotleft~baignoire avec moins d'eau~\guillemotright\ (éloignées de l'état stationnaire). On s'attend à ce que leur croissance soit naturellement plus élevée (rattrapage). Même au milieu d'une récession, elles ne souffriraient pas nécessairement de décroissance. Puisque les économies émergentes contribuent positivement à la croissance moyenne mondiale, que la croissance mondiale baisse sous 3\,\% serait cohérent avec une croissance négative dans les pays développés et une croissance plus faible que le potentiel pour les pays en voie de développement.
\end{reponse}

\shortq{La situation aux États-Unis, politique fiscale et marchés financiers.}

\begin{enumerate}[label=(\alph*)]
   \item En 2005, le PIB réel américain a cru de 3.5\,\% et le taux de chômage était de 5.1\,\%. L'inflation (IPC) était de 2.4\,\% et l'inflation de référence (sans énergie ni nourriture) de 2.2\,\%. Dans le diagramme AD-AS, indiquez la position probable de l'économie en 2005. Justifiez.
   \item À la fin 2008, le consensus des prévisionnistes pour la croissance du PIB au Q4 était de $-4.3\,\%$. L'inflation était de 1.7\,\% et l'inflation de référence de 0\,\%. Le taux de chômage était de 6.7\,\%. Indiquez la position probable de l'économie en décembre 2008.
   \item Décrivez quelques facteurs probables qui ont amené l'économie de la situation (a) à (b). Interprétez en termes de déplacements de AD ou AS.
   \item Sur la base de votre analyse, expliquez les raisons derrière la décision de la Fed de baisser les taux.
   \item La Fed a indiqué qu'elle continuerait d'injecter des liquidités (achats d'obligations et de titres hypothécaires). Expliquez pourquoi baisser le taux à un jour n'est plus suffisant et pourquoi une injection de liquidités est nécessaire.
   \item Après son élection en 2008, Obama a lancé un important stimulus fiscal (routes, infrastructures). Considérons deux plans : Plan A (dépenses sans effet sur $K$, $A$ ou $N^S$) et Plan B (investissements productifs qui augmentent $A$ de manière permanente). En partant de la situation (b), montrez l'impact du Plan A sur $Y$ et $P$ à court terme.
   \item Montrez l'impact du Plan B sur $Y$ et $P$ à court terme.
   \item Expliquez lequel des deux plans (A ou B) est le plus désirable économiquement et pourquoi.
\end{enumerate}

\begin{reponse}
\textbf{(a)} L'économie était proche de son potentiel : faible taux de chômage, inflation modérée. Dans le diagramme AD-AS, l'intersection de AD et AS est proche de la ligne verticale $Y^{\text{POT}}$.

\textbf{(b)} Croissance fortement négative, inflation très faible (référence à 0\,\%), taux de chômage en hausse $\to$ l'économie est nettement en dessous de son potentiel. Le point d'intersection se situe à gauche de $Y^{\text{POT}}$.

\textbf{(c)} Tous ces facteurs ont entraîné un mouvement de AD vers la gauche :
\begin{itemize}
   \item Chute des prix des maisons $\to$ effet richesse négatif sur la consommation
   \item Chute de la confiance des consommateurs
   \item Chute du multiplicateur monétaire $\to$ crise de liquidité $\to$ chute du crédit
   \item Chute de la confiance des firmes $\to$ chute de l'investissement
\end{itemize}

\textbf{(d)} L'économie est sous son potentiel $\to$ la Fed veut stimuler $C$ et $I$. La baisse très agressive des taux (de 1\,\%) vise à rassurer consommateurs et firmes. But : pousser AD vers la droite.

\textbf{(e)} La chute du multiplicateur monétaire (les banques gardent les liquidités au lieu de prêter) implique qu'une injection de liquidité par la Fed a un effet plus faible sur la masse monétaire. Les banques centrales doivent compenser en injectant encore plus de liquidités. C'est l'assouplissement quantitatif (QE) : acheter massivement des obligations pour faire baisser les taux longs.

\textbf{(f)} Plan A : $G$ augmente $\to$ AD vers la droite $\to$ $Y\uparrow$ et $P\uparrow$ à court terme. Aucun effet sur le PIB potentiel car $A$ n'est pas affecté.

\textbf{(g)} Plan B : AD vers la droite (effet de $G$) ET AS vers la droite (car $A$ augmente $\to$ les firmes produisent plus à chaque niveau de prix) ET la ligne $Y^{\text{POT}}$ se déplace vers la droite. $Y$ augmente plus que sous le Plan A. L'impact sur $P$ est ambigu (AD et AS bougent dans le même sens pour $Y$ mais en sens opposé pour $P$).

\textbf{(h)} Clairement le Plan B. Le Plan A permet un stimulus à court terme, mais à moyen-long terme il n'est pas clair que son impact sera positif (le PIB potentiel est inchangé). En plus d'un stimulus plus fort à court terme, le Plan B a un impact positif permanent grâce à l'augmentation de la PTF.
\end{reponse}

\shortq{L'économie mexicaine traverse une période difficile. Considérons les scénarios suivants :}

\begin{enumerate}[label=(\alph*)]
   \item L'incertitude commerciale pousse les clients étrangers à se tourner vers d'autres pays. Montrez comment cela affecterait l'économie mexicaine à court terme (diagramme AD-AS).
   \item Beaucoup craignent qu'une augmentation drastique du salaire minimum entraîne une hausse significative des coûts de production. Montrez l'impact sur l'économie mexicaine à court terme.
   \item Imaginez que les effets des scénarios (a) et (b) sur le PIB soient à peu près équivalents. Si vous étiez le gouverneur de la banque centrale mexicaine, lequel des deux préféreriez-vous affronter? Pourquoi?
\end{enumerate}

\begin{reponse}
\textbf{(a)} La baisse des exportations est un choc de demande négatif : AD se déplace vers la gauche $\to$ récession ($Y\downarrow$) et baisse des prix/inflation ($P\downarrow$).

\textbf{(b)} La hausse des coûts de production (salaires plus élevés) est un choc d'offre négatif : AS se déplace vers la gauche (ou vers le haut) $\to$ récession ($Y\downarrow$) et hausse des prix ($P\uparrow$). C'est de la stagflation.

\textbf{(c)} Le gouverneur préférerait le scénario (a) (choc de demande). La politique monétaire peut déplacer la courbe AD vers la droite (baisse des taux $\to$ stimulation de $C$ et $I$), ce qui ramène $Y$ et $P$ vers leurs niveaux initiaux. Les deux objectifs (stabilité des prix et plein emploi) sont \textbf{alignés}. En revanche, face au scénario (b) (choc d'offre), la banque centrale fait face à un \textbf{dilemme} : stimuler pour soutenir $Y$ aggraverait l'inflation, et combattre l'inflation aggraverait la récession.
\end{reponse}
