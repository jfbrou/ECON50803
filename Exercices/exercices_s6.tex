\section{Questions à choix multiples}
% ============================================================================

\setcounter{qcmcounter}{0}

\qcm{Le compte courant (CA) d'un pays est égal à :}
\choix{
   \item Exportations $+$ importations
   \item $S - I$ (épargne nationale moins investissement)
   \item Le solde budgétaire du gouvernement
   \item Le taux de change multiplié par les exportations nettes
}

\begin{reponse}
\textbf{(b)} L'identité fondamentale du compte courant est $\text{CA} = S - I$, où $S = S_P + S_G$ (épargne privée + épargne publique). (a) est faux : les exportations \textit{plus} les importations mesurent le volume du commerce, pas le solde. Cette identité signifie qu'un pays en surplus courant épargne plus qu'il n'investit domestiquement (l'excédent d'épargne est prêté au reste du monde), tandis qu'un pays en déficit investit plus qu'il n'épargne (il emprunte au reste du monde).
\end{reponse}

\qcm{Dans une \textbf{petite économie ouverte}, une hausse des dépenses gouvernementales ($G$) entraîne :}
\choix{
   \item Une hausse du taux d'intérêt domestique
   \item Une détérioration du compte courant (CA diminue) sans changement du taux d'intérêt
   \item Une hausse du PIB proportionnelle à la hausse de $G$
   \item Aucun effet sur l'économie
}

\begin{reponse}
\textbf{(b)} Dans une petite économie ouverte, le taux d'intérêt est déterminé par le marché mondial ($r^w$) et ne change pas. La hausse de $G$ réduit l'épargne publique ($S_G = T - G$ diminue) $\to$ l'épargne nationale $S$ diminue $\to$ à $r^w$ inchangé, $S < I$ $\to$ le compte courant se détériore (CA $= S - I$ diminue). C'est le phénomène des \guillemotleft~déficits jumeaux~\guillemotright\ : le déficit budgétaire s'accompagne d'un déficit du compte courant.
\end{reponse}

\qcm{Les tarifs de Trump en 2018 n'ont \textbf{pas} réduit le déficit commercial américain. L'explication via le modèle $S$-$I$ est que :}
\choix{
   \item Les tarifs n'ont aucun effet sur les échanges commerciaux
   \item Les tarifs modifient la composition du commerce, mais pas $S$ ni $I$, donc pas CA $= S - I$
   \item Les tarifs ont augmenté l'épargne américaine
   \item Les tarifs ont réduit l'investissement américain
}

\begin{reponse}
\textbf{(b)} Le compte courant est déterminé par l'écart entre l'épargne nationale et l'investissement ($\text{CA} = S - I$). En première approximation, les tarifs n'affectent pas directement les décisions d'épargne ni d'investissement des Américains. Ils modifient la \textbf{composition} du commerce (moins d'imports de Chine, plus du Vietnam ou d'autres pays --- c'est la \guillemotleft~déviation commerciale~\guillemotright), mais le solde global reste inchangé. Pour réduire le déficit courant, il faudrait augmenter $S$ ou réduire $I$.
\end{reponse}

\qcm{Selon l'hypothèse de la \guillemotleft~surabondance d'épargne mondiale~\guillemotright\ (\textit{global saving glut}) de Bernanke, le déficit courant américain des années 2000 s'explique par :}
\choix{
   \item Un manque d'épargne aux États-Unis
   \item Un excès d'épargne dans les pays asiatiques et exportateurs de pétrole, qui a fait baisser $r^w$
   \item Des taux d'intérêt trop élevés aux États-Unis
   \item Une hausse de la productivité américaine
}

\begin{reponse}
\textbf{(b)} Bernanke (2005) argue que le déficit américain n'est pas principalement dû à un comportement domestique, mais à un excès d'épargne \textbf{mondial}. Après les crises asiatiques de 1997--98, les pays d'Asie ont massivement accumulé des réserves (épargne de précaution). Combiné aux pétrodollars du Moyen-Orient, cet excès d'épargne mondiale a fait baisser le taux d'intérêt mondial ($r^w \downarrow$). À un $r^w$ plus faible, l'épargne américaine diminue et l'investissement augmente $\to$ $S - I$ devient plus négatif $\to$ déficit courant.
\end{reponse}

\qcm{Lequel des énoncés suivants est \textbf{vrai} pour une économie fermée?}
\choix{
   \item Le compte courant est positif
   \item $S = I$ à l'équilibre
   \item Le taux d'intérêt est déterminé par le marché mondial
   \item Les exportations nettes sont positives
}

\begin{reponse}
\textbf{(b)} Dans une économie fermée, il n'y a pas de commerce international ($NX = 0$, $\text{CA} = 0$). Toute l'épargne nationale finance l'investissement domestique : $S = I$. Le taux d'intérêt d'équilibre ($r^*$) est déterminé par l'intersection des courbes $S(r)$ et $I(r)$ domestiques, \textbf{sans} influence du reste du monde.
\end{reponse}

\qcm{Le taux d'ouverture commerciale d'un pays se mesure par :}
\choix{
   \item Les exportations divisées par le PIB ($X/Y$)
   \item Les importations divisées par le PIB ($M/Y$)
   \item La somme des exportations et importations divisée par le PIB ($(X+M)/Y$)
   \item Le solde commercial divisé par le PIB ($NX/Y$)
}

\begin{reponse}
\textbf{(c)} Le taux d'ouverture commerciale est $(X + M) / Y$, soit la somme des flux commerciaux (exportations + importations) rapportée au PIB. Ce ratio mesure l'importance du commerce international dans l'économie. Un pays comme le Canada a un taux d'ouverture d'environ 65\,\%, tandis que les États-Unis, en raison de la taille de leur marché intérieur, sont à environ 25\,\%. (a) et (b) ne mesurent qu'un seul côté du commerce; (d) mesure le \textbf{solde}, pas le \textbf{volume}.
\end{reponse}

\qcm{Un pays en déficit du compte courant a nécessairement :}
\choix{
   \item Un surplus du compte de capital (entrées nettes de capitaux)
   \item Un déficit du compte de capital (sorties nettes de capitaux)
   \item Un excédent budgétaire
   \item Une monnaie sous-évaluée
}

\begin{reponse}
\textbf{(a)} La balance des paiements impose que $\text{CA} + \text{Compte de capital} = 0$. Si $\text{CA} < 0$ (déficit courant), alors le compte de capital est en \textbf{surplus} : le pays reçoit des entrées nettes de capitaux étrangers. Intuitivement : un pays qui dépense plus qu'il ne produit (importe plus qu'il n'exporte) doit financer cet écart par des emprunts ou des ventes d'actifs au reste du monde. Chaque dollar de déficit courant a sa contrepartie en dollar de financement étranger.
\end{reponse}

\qcm{La différence clé entre une petite et une grande économie ouverte est que :}
\choix{
   \item La petite économie a un PIB plus faible
   \item La petite économie prend le taux d'intérêt mondial ($r^w$) comme donné, tandis que la grande économie influence $r^w$
   \item La grande économie ne commerce pas avec l'extérieur
   \item La petite économie ne peut pas avoir de déficit courant
}

\begin{reponse}
\textbf{(b)} Le critère n'est pas la taille absolue du PIB, mais l'\textbf{influence sur le taux d'intérêt mondial}. Une \textbf{petite} économie ouverte (ex. : Canada, Suisse) est trop petite pour affecter $r^w$ : elle prend $r^w$ comme une donnée exogène. Une \textbf{grande} économie ouverte (ex. : États-Unis, zone euro) représente une part suffisante des marchés financiers mondiaux pour influencer $r^w$ par ses décisions d'épargne et d'investissement.
\end{reponse}

\qcm{L'épargne nationale se décompose en $S = S_P + S_G$ où $S_P = Y - T - C$ et $S_G = T - G$. Une baisse d'impôts ($T\downarrow$) sans changement de $G$ :}
\choix{
   \item Augmente $S_G$ et $S_P$ dans les mêmes proportions
   \item Réduit $S_G$, mais peut augmenter $S_P$ partiellement si les ménages épargnent une partie du gain
   \item N'a aucun effet sur $S$ car les effets sur $S_P$ et $S_G$ s'annulent exactement
   \item Augmente toujours l'épargne nationale
}

\begin{reponse}
\textbf{(b)} Une baisse de $T$ réduit directement $S_G = T - G$ (le déficit budgétaire se creuse). Quant à $S_P = Y - T - C$, le revenu disponible ($Y - T$) augmente, mais les ménages consomment une partie de ce gain ($C\uparrow$). Si les ménages épargnent une fraction de la baisse d'impôts, $S_P$ augmente partiellement. L'effet net sur $S = S_P + S_G$ est généralement \textbf{négatif} : la hausse de $S_P$ ne compense pas la baisse de $S_G$. C'est pourquoi on dit que \guillemotleft~les déficits jumeaux~\guillemotright\ sont probables : $S_G\downarrow \to S\downarrow \to CA\downarrow$.
\end{reponse}

\qcm{Après les tarifs américains de 2018 sur la Chine, les exportations chinoises vers les É.-U.\ ont chuté mais les exportations chinoises \textbf{totales} n'ont pas diminué. Cela s'explique par :}
\choix{
   \item Une hausse de la productivité chinoise
   \item La déviation commerciale : la Chine a redirigé ses exportations via des pays tiers (Vietnam, Mexique)
   \item Une baisse du yuan qui a rendu les exportations chinoises moins chères
   \item Une hausse de la demande intérieure chinoise
}

\begin{reponse}
\textbf{(b)} C'est le phénomène de \textbf{déviation commerciale} (\textit{trade diversion}). Les tarifs de 2018 ont augmenté le coût des importations directes depuis la Chine, mais les entreprises ont contourné les tarifs en réacheminant les produits via des pays tiers (Vietnam, Mexique, Taïwan). La composition géographique du commerce a changé, mais le volume total d'exportations chinoises est resté similaire. C'est cohérent avec le modèle $S$-$I$ : les tarifs ne modifient pas $S$ ni $I$, donc pas le compte courant $= S - I$, mais seulement la \textbf{composition} du commerce.
\end{reponse}

\shortq{Un pays a un déficit du compte courant de 100 milliards \$. Le gouvernement envisage trois politiques pour réduire ce déficit. Pour chacune, expliquez l'effet sur $S$, $I$ et CA $= S - I$ :}

\begin{enumerate}[label=(\alph*)]
   \item Réduire les dépenses gouvernementales ($G$) de 50 milliards \$ sans changer les impôts.
   \item Imposer des tarifs de 25\,\% sur toutes les importations.
   \item Augmenter les impôts ($T$) de 30 milliards \$ pour réduire le déficit budgétaire.
\end{enumerate}

\begin{reponse}
\textbf{(a)} Réduction de $G$ : $S_G = T - G$ augmente de 50 G\$ $\to$ $S = S_P + S_G$ augmente de 50 G\$ (si $S_P$ ne change pas). $I$ reste inchangé. Donc $CA = S - I$ s'améliore de 50 G\$ (le déficit passe de $-100$ à $-50$ G\$). C'est une politique \textbf{efficace} pour réduire le déficit courant via la réduction du déficit budgétaire (fin des ``déficits jumeaux'').

\textbf{(b)} Tarifs sur les importations : en première approximation, les tarifs ne modifient ni $S$ ni $I$. Ils changent la \textbf{composition} du commerce (déviation commerciale vers d'autres pays), mais pas le solde $CA = S - I$. Le déficit courant reste à $-100$ G\$. C'est la leçon des tarifs de 2018 : le déficit commercial américain n'a pas diminué.

\textbf{(c)} Hausse de $T$ : $S_G = T - G$ augmente de 30 G\$. Cependant, la hausse de $T$ réduit le revenu disponible des ménages ($Y - T \downarrow$), ce qui réduit à la fois $C$ et $S_P$. Si la propension marginale à consommer est de 0.7, alors $C$ baisse de 21 G\$ et $S_P$ baisse de 9 G\$. L'effet net sur $S$ est : $+30 - 9 = +21$ G\$. Le CA s'améliore de 21 G\$ (moins que la hausse de $T$, en raison de l'effet sur l'épargne privée).
\end{reponse}

\shortq{Expliquez, à l'aide du modèle $S$-$I$ à trois panneaux (pays domestique, reste du monde, monde), comment une hausse anticipée de la productivité ($A$) dans un pays affecte son compte courant. Ce déficit courant est-il \guillemotleft~bon~\guillemotright\ ou \guillemotleft~mauvais~\guillemotright?}

\begin{reponse}
\textbf{Mécanisme :} Une hausse anticipée de $A$ augmente le rendement futur attendu du capital $\to$ les entreprises veulent investir davantage dès maintenant pour en profiter. La courbe $I(r)$ se déplace vers la \textbf{droite} dans le panneau domestique.

\textbf{Effet :}
\begin{itemize}
   \item À $r^w$ inchangé (petite économie ouverte) : $I$ augmente mais $S$ reste stable $\to$ $\text{CA} = S - I$ diminue $\to$ le pays passe en \textbf{déficit courant}.
   \item Si c'est une grande économie : l'augmentation de $I$ domestique fait pression à la hausse sur $r^w$, ce qui modère partiellement l'effet (mais le CA se détériore quand même).
\end{itemize}

\textbf{Bon ou mauvais?} C'est un déficit \guillemotleft~\textbf{bon}~\guillemotright. Le pays emprunte au reste du monde pour financer des investissements \textbf{productifs} qui généreront des revenus futurs. C'est exactement comme une entreprise qui s'endette pour financer un projet rentable. C'est l'explication classique du déficit américain des années 1990, lorsque le boom des technologies de l'information attirait des capitaux du monde entier.

Par contraste, un déficit causé par un excès de dépenses courantes du gouvernement ($G \uparrow \to S \downarrow$) est généralement un déficit \guillemotleft~mauvais~\guillemotright, car il ne finance pas un investissement productif mais une consommation non financée. (Nuance : si la hausse de $G$ finance des investissements publics productifs --- infrastructures, éducation --- le déficit peut aussi être \guillemotleft~bon~\guillemotright.)
\end{reponse}

\vfi{Un déficit du compte courant signifie qu'un pays vit au-dessus de ses moyens.}

\begin{reponse}
\textbf{Incertain.} Un déficit du compte courant signifie que $S < I$ (le pays investit plus qu'il n'épargne et emprunte au reste du monde). Mais il faut distinguer la \textbf{cause} du déficit :
\begin{itemize}
   \item \textbf{Déficit \guillemotleft~bon~\guillemotright} : causé par une hausse de l'investissement ($I\uparrow$) financé par des capitaux étrangers. Le pays emprunte pour financer des projets \textbf{productifs} qui généreront des revenus futurs. Exemple : les États-Unis dans les années 1990 (boom technologique) ou un pays émergent en phase de développement.
   \item \textbf{Déficit \guillemotleft~mauvais~\guillemotright} : causé par une baisse de l'épargne publique ($S_G\downarrow$ à cause de $G\uparrow$ non financé) ou une surconsommation ($C\uparrow$). Le pays emprunte pour consommer, pas pour investir. Exemple : la Grèce avant 2010.
\end{itemize}
Le signe du CA ne suffit pas pour juger : c'est la \textbf{source} du déséquilibre qui détermine s'il est sain ou dangereux.
\end{reponse}

\vfi{Le libre-échange bénéficie à tous les travailleurs d'un pays.}

\begin{reponse}
\textbf{Faux.} Le libre-échange crée des gains \textbf{agrégés} (le pays dans son ensemble est plus riche), mais ces gains sont inégalement répartis :
\begin{itemize}
   \item \textbf{Gagnants :} les consommateurs (prix plus bas, plus de variété), les travailleurs dans les secteurs exportateurs, les entreprises compétitives à l'international
   \item \textbf{Perdants :} les travailleurs dans les secteurs \textbf{importateurs concurrencés} (manufacturier), surtout les moins qualifiés et les moins mobiles géographiquement
\end{itemize}
Les travaux d'Autor, Dorn et Hanson sur le \guillemotleft~choc chinois~\guillemotright\ montrent que les régions américaines les plus exposées aux importations chinoises ont subi des pertes d'emploi durables, des baisses de salaires et une hausse des problèmes sociaux. Les gains du commerce sont \textbf{diffus} (un peu pour chaque consommateur) tandis que les pertes sont \textbf{concentrées} (beaucoup pour quelques travailleurs/régions). Une politique commerciale responsable doit combiner ouverture commerciale et mécanismes de compensation pour les perdants.
\end{reponse}

\shortq{En utilisant le diagramme $S$-$I$ à trois panneaux (pays domestique, reste du monde, monde), analysez l'effet d'une expansion budgétaire ($G\uparrow$) dans une \textbf{grande} économie ouverte. Que se passe-t-il pour $r^w$, le compte courant domestique et le compte courant du reste du monde?}

\begin{reponse}
\textbf{Mécanisme :}
\begin{itemize}
   \item $G\uparrow$ avec $T$ inchangé $\to$ $S_G = T - G$ diminue $\to$ l'épargne nationale $S$ diminue $\to$ la courbe $S$ du pays domestique se déplace vers la \textbf{gauche}.
   \item Dans le panneau du marché mondial des fonds prêtables : l'offre mondiale de fonds diminue (car le pays domestique est \textbf{grand}) $\to$ $r^w$ \textbf{augmente}.
\end{itemize}

\textbf{Effets :}
\begin{itemize}
   \item \textbf{Pays domestique :} au nouveau $r^w$ plus élevé, $S$ est un peu plus élevé (les ménages épargnent plus à taux plus élevé) et $I$ est plus faible (investissement plus coûteux). Mais l'effet net est que $S < I$ $\to$ le CA se \textbf{détériore} (déficit courant).
   \item \textbf{Reste du monde :} le $r^w$ plus élevé augmente l'épargne et réduit l'investissement dans le reste du monde $\to$ $S > I$ $\to$ le CA du reste du monde \textbf{s'améliore} (surplus courant).
   \item \textbf{Vérification :} le déficit du pays domestique $=$ le surplus du reste du monde (les flux de capitaux mondiaux s'équilibrent).
\end{itemize}

C'est le scénario des \guillemotleft~déficits jumeaux~\guillemotright\ américains des années 1980 (Reagan) et 2000 (Bush).
\end{reponse}

\shortq{Expliquez l'hypothèse de la \guillemotleft~surabondance d'épargne mondiale~\guillemotright\ (\textit{global saving glut}) de Bernanke. En utilisant le diagramme $S$-$I$, montrez comment l'excès d'épargne en Asie et chez les exportateurs de pétrole peut expliquer le déficit courant américain des années 2000.}

\begin{reponse}
\textbf{Contexte :} Après les crises asiatiques de 1997--98, les pays d'Asie (Chine, Japon, Corée, Taïwan) et les exportateurs de pétrole (Arabie Saoudite, Russie) ont accumulé massivement des réserves de change, reflétant un excès d'épargne par rapport à l'investissement domestique.

\textbf{Analyse $S$-$I$ à trois panneaux :}
\begin{itemize}
   \item \textbf{Panneau \guillemotleft~reste du monde~\guillemotright\ (Asie + pétrole) :} $S$ se déplace massivement vers la droite (épargne de précaution post-crise, revenus pétroliers). À $r^w$ initial, $S \gg I$ $\to$ excédent courant.
   \item \textbf{Panneau mondial :} l'excès d'offre de fonds fait baisser $r^w$ (le prix d'équilibre des fonds prêtables diminue).
   \item \textbf{Panneau États-Unis :} au $r^w$ plus faible, l'épargne américaine diminue ($S\downarrow$) et l'investissement augmente ($I\uparrow$) $\to$ $S < I$ $\to$ le CA américain se détériore (déficit).
\end{itemize}

\textbf{Conclusion de Bernanke :} Le déficit américain n'est pas principalement un problème \textbf{américain} (surconsommation), mais le reflet d'un problème \textbf{mondial} (excès d'épargne). Les capitaux affluent vers les É.-U.\ parce que le reste du monde n'a pas assez d'opportunités d'investissement par rapport à son épargne.
\end{reponse}

\shortq{Un même choc se produit --- une explosion de l'investissement due à un boom de l'IA. Comparez les effets dans (i) une petite économie ouverte et (ii) une grande économie ouverte, en termes de $r$, CA et investissement total.}

\begin{reponse}
\textbf{(i) Petite économie ouverte :}
\begin{itemize}
   \item Le pays prend $r^w$ comme donné (il est trop petit pour l'influencer)
   \item $I$ se déplace vers la droite : au $r^w$ inchangé, l'investissement augmente fortement
   \item $S$ reste au même niveau (à $r^w$ inchangé)
   \item $CA = S - I$ se \textbf{détériore} fortement (le pays emprunte massivement au reste du monde)
   \item $r$ domestique : \textbf{inchangé} (reste égal à $r^w$)
   \item L'investissement total augmente de toute la hausse de $I$ désiré
\end{itemize}

\textbf{(ii) Grande économie ouverte :}
\begin{itemize}
   \item La hausse de $I$ domestique fait monter $r^w$ (le pays est assez grand pour affecter le marché mondial)
   \item La hausse de $r^w$ a deux effets modérateurs : (1) elle stimule l'épargne domestique ($S\uparrow$), et (2) elle réduit partiellement l'investissement ($I$ augmente moins qu'en petite économie, car $r$ est plus élevé)
   \item $CA = S - I$ se détériore, mais \textbf{moins} qu'en petite économie
   \item $r$ domestique : \textbf{augmente}
   \item L'investissement total augmente moins qu'en petite économie (effet d'éviction via la hausse de $r$)
\end{itemize}

\textbf{Synthèse :} Dans une petite économie, tout l'ajustement se fait via le CA (le pays emprunte autant qu'il veut à $r^w$). Dans une grande économie, l'ajustement passe aussi par les prix ($r^w\uparrow$), ce qui modère le déficit courant mais réduit l'investissement réalisé.
\end{reponse}

\setcounter{shortcounter}{0}

\shortq{Voici les comptes nationaux d'un pays fictif en 2026 (en milliards \$) :}

\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l r}
\toprule
\textbf{Variable} & \textbf{Valeur} \\
\midrule
PIB ($Y$) & 500 \\
Consommation ($C$) & 300 \\
Investissement ($I$) & 120 \\
Dépenses gouvernementales ($G$) & 100 \\
Impôts ($T$) & 90 \\
\bottomrule
\end{tabular}
\end{center}

\begin{enumerate}[label=(\alph*)]
   \item Calculez les exportations nettes ($NX$).
   \item Calculez l'épargne privée ($S_P$), l'épargne publique ($S_G$) et l'épargne nationale ($S$).
   \item Calculez le compte courant ($CA$) et vérifiez que $CA = NX = S - I$.
   \item Ce pays est-il un prêteur ou un emprunteur net? Expliquez.
\end{enumerate}

\begin{reponse}
\textbf{(a)} $NX = Y - C - I - G = 500 - 300 - 120 - 100 = -20$ milliards \$. Le pays a un \textbf{déficit commercial} de 20 G\$.

\textbf{(b)}
\begin{itemize}
   \item $S_P = Y - T - C = 500 - 90 - 300 = 110$ milliards \$ (épargne privée)
   \item $S_G = T - G = 90 - 100 = -10$ milliards \$ (déficit budgétaire de 10 G\$)
   \item $S = S_P + S_G = 110 + (-10) = 100$ milliards \$ (épargne nationale)
\end{itemize}

\textbf{(c)} $CA = S - I = 100 - 120 = -20$ milliards \$. On vérifie : $CA = NX = -20$ G\$. $\surd$

L'identité $CA = NX = S - I$ est toujours respectée. Le déficit courant ($-20$) signifie que le pays investit plus qu'il n'épargne.

\textbf{(d)} Le pays est un \textbf{emprunteur net} : il emprunte 20 G\$ au reste du monde pour combler l'écart entre son épargne (100 G\$) et son investissement (120 G\$). Ces entrées de capitaux apparaissent comme un surplus du compte de capital ($+20$ G\$).
\end{reponse}

\shortq{Dans une économie fermée, l'épargne et l'investissement sont donnés par :
$$S(r) = 100 + 200r \qquad I(r) = 300 - 200r$$
où $r$ est le taux d'intérêt réel (exprimé en décimal, ex. : $r = 0.05$ pour 5\,\%).}

\begin{enumerate}[label=(\alph*)]
   \item Trouvez le taux d'intérêt d'autarcie ($r^*$) et les niveaux de $S$ et $I$ en économie fermée.
   \item L'économie s'ouvre au commerce international. Le taux d'intérêt mondial est $r^w = 0.75$ (75\,\%). Calculez $S$, $I$ et $CA$ au nouveau taux. Le pays est-il prêteur ou emprunteur?
   \item Refaites le calcul avec $r^w = 0.25$ (25\,\%). Le pays est-il prêteur ou emprunteur?
\end{enumerate}

\begin{reponse}
\textbf{(a)} En économie fermée, $S = I$ :
$$100 + 200r = 300 - 200r$$
$$400r = 200$$
$$r^* = 0.50 \quad (50\,\%)$$
$S = I = 100 + 200(0.50) = 200$.

\textbf{(b)} Avec $r^w = 0.75$ :
\begin{itemize}
   \item $S = 100 + 200(0.75) = 250$
   \item $I = 300 - 200(0.75) = 150$
   \item $CA = S - I = 250 - 150 = +100$
\end{itemize}
Le pays est un \textbf{prêteur net} : $r^w > r^*$, donc le rendement mondial est attractif $\to$ les épargnants préfèrent prêter à l'étranger. Le pays épargne plus qu'il n'investit domestiquement et exporte du capital.

\textbf{(c)} Avec $r^w = 0.25$ :
\begin{itemize}
   \item $S = 100 + 200(0.25) = 150$
   \item $I = 300 - 200(0.25) = 250$
   \item $CA = S - I = 150 - 250 = -100$
\end{itemize}
Le pays est un \textbf{emprunteur net} : $r^w < r^*$, donc le capital étranger est bon marché $\to$ les entreprises domestiques investissent davantage en empruntant au reste du monde. Le pays investit plus qu'il n'épargne et importe du capital.

\textbf{Règle générale :} si $r^w > r^*$ (autarcie), le pays est \textbf{prêteur} (CA $> 0$). Si $r^w < r^*$, le pays est \textbf{emprunteur} (CA $< 0$).
\end{reponse}

\qcm{Entre 2010 et 2015, l'Inde a connu un double déficit, à savoir un déficit budgétaire et un déficit courant. Laquelle des affirmations suivantes est correcte?}
\choix{
   \item La somme de l'épargne privée et publique était supérieure à l'investissement
   \item L'Inde exportait plus d'actifs réels et financiers qu'elle n'en importait
   \item Les recettes publiques étaient supérieures aux dépenses publiques
   \item Le compte de capital et le compte financier de l'Inde étaient négatifs
}

\begin{reponse}
\textbf{(b) --- Attention, reformulons.} Un \textbf{déficit courant} signifie $CA = S - I < 0$ : l'Inde investit plus qu'elle n'épargne et emprunte au reste du monde. Le \textbf{compte de capital} est donc en \textbf{surplus} (entrées nettes de capitaux). L'Inde \guillemotleft~importe~\guillemotright\ du capital étranger (elle vend des actifs domestiques aux étrangers). (a) est faux car $S < I$. (c) est faux car le déficit budgétaire signifie $T < G$. (d) est faux car le compte de capital est positif (surplus).

Note : \guillemotleft~exporter des actifs~\guillemotright\ signifie dans ce contexte que l'Inde \textbf{vendait} des actifs domestiques aux investisseurs étrangers (entrées de capitaux), ce qui correspond à un surplus du compte de capital.
\end{reponse}

\vfi{Les droits de douane (tarifs) sur les importations ne peuvent réduire le déficit du compte courant d'un pays que si ces droits de douane diminuent le niveau d'épargne nationale et/ou augmentent le niveau d'investissement.}

\begin{reponse}
\textbf{Vrai.} Le compte courant est déterminé par $CA = S - I$. Pour que les tarifs réduisent le déficit courant ($CA$ devient moins négatif), il faudrait que $S$ augmente ou que $I$ diminue. Or, en première approximation, les tarifs ne modifient ni les décisions d'épargne ni les décisions d'investissement. Ils changent la \textbf{composition} du commerce (déviation commerciale vers d'autres pays), mais pas le \textbf{solde} $S - I$. C'est exactement ce qu'on a observé avec les tarifs de 2018 aux États-Unis.
\end{reponse}

\vfi{L'expansion rapide des dépenses publiques aux États-Unis et en Europe pendant et après la pandémie devrait nous inquiéter du risque de détérioration des balances commerciales des économies avancées, toutes choses égales par ailleurs.}

\begin{reponse}
\textbf{Vrai.} Dans le modèle $S$-$I$, une hausse de $G$ (sans hausse de $T$) réduit l'épargne publique ($S_G = T - G \downarrow$), ce qui réduit l'épargne nationale ($S\downarrow$). À $I$ inchangé, $CA = S - I$ se détériore. Dans le diagramme à trois panneaux (pour une grande économie comme les É.-U.), le déplacement de $S$ vers la gauche fait monter $r^w$ et génère un déficit courant. C'est le mécanisme des \guillemotleft~déficits jumeaux~\guillemotright\ : déficit budgétaire $\to$ déficit courant.
\end{reponse}

\shortq{Le compte courant américain est passé de $-2\,\%$ du PIB avant la pandémie à $-3.5\,\%$ en 2022. Un commentateur écrit : \guillemotleft~L'aggravation du déficit courant s'explique par le fait que les ménages américains ont connu une hausse des prix de l'immobilier et une augmentation des soldes bancaires. Est-ce vraiment une mauvaise chose?~\guillemotright}

En utilisant le modèle $S$-$I$, expliquez comment cet effet de richesse peut aggraver le déficit courant.

\begin{reponse}
\textbf{Mécanisme :} Les ménages américains se sentent \textbf{plus riches} (hausse de la valeur immobilière + soldes bancaires plus élevés grâce aux transferts gouvernementaux et à l'épargne forcée pendant les confinements). L'effet de richesse les pousse à \textbf{consommer davantage et épargner moins}.

\textbf{Analyse $S$-$I$ :}
\begin{itemize}
   \item La courbe d'épargne privée $S_P$ se déplace vers la \textbf{gauche} (les ménages épargnent moins à chaque taux $r$)
   \item L'épargne nationale $S = S_P + S_G$ diminue
   \item Comme les É.-U.\ sont une \textbf{grande} économie, la baisse de $S$ fait monter $r^w$
   \item Au nouveau $r^w$ : $S$ est plus faible, $I$ est un peu plus faible (taux plus élevé) $\to$ $CA = S - I$ se détériore
\end{itemize}

Est-ce une \guillemotleft~mauvaise chose~\guillemotright? Cela dépend. Si l'effet de richesse reflète une hausse \textbf{réelle} et durable de la valeur des actifs, c'est un ajustement rationnel. Mais si c'est une \textbf{bulle} (comme en 2006--2007), la surconsommation est suivie d'un ajustement douloureux.
\end{reponse}

\vfi{Durant les 20 dernières années, le passif net des Américains envers l'étranger s'est détérioré, et les États-Unis sont passés d'une situation de créditeur net à une situation de dette extérieure nette. Ceci implique qu'au cours des 20 dernières années, les États-Unis ont, en moyenne, enregistré des déficits du compte courant et du compte de capital et financier.}

\begin{reponse}
\textbf{Faux.} Le compte courant (CC) et le compte de capital et financier (CKF) doivent aller dans des directions \textbf{opposées} (leur somme $= 0$). Si les É.-U.\ sont passés de créditeur à débiteur, c'est parce qu'ils ont accumulé des \textbf{déficits du compte courant} ($CA < 0$). Mais cela signifie un \textbf{surplus} du compte de capital et financier (entrées nettes de capitaux étrangers qui financent le déficit courant), pas un déficit.
\end{reponse}

\shortq{L'investissement privé au Québec a été plus faible que dans le reste du Canada depuis l'an 2000. Le Ministère des Finances s'inquiète et mandate un groupe de travail.}

\textit{Note : Pour cette question, supposez que le monde est composé uniquement du Québec et du Reste du Canada (ROC). Le Québec commence avec un déficit du compte courant.}

\begin{enumerate}[label=(\alph*)]
   \item Le Ministère a-t-il raison de s'inquiéter? Quels sont les conséquences à long terme d'un faible taux d'investissement sur la santé économique du Québec?

   \item Plusieurs économistes pointent du doigt le faible taux d'épargne québécois. En utilisant le diagramme $S$-$I$ en économie ouverte (trois panneaux : Québec, Reste du Canada, Monde), expliquez si une baisse exogène du taux d'épargne au Québec peut expliquer une réduction de l'investissement relatif.

   \item Un autre groupe croit que l'explication provient de l'augmentation plus rapide de la PTF ($A$) dans le reste du Canada, en particulier à travers l'effet sur les décisions d'investissement. Ce scénario peut-il expliquer la faible performance de l'investissement québécois?

   \item Afin de départager ces explications, vous disposez de données sur la balance commerciale du Québec avec le reste du Canada (déficits croissants : $-1\,128$M en 2001 à $-3\,060$M en 2005). Laquelle de ces théories est la plus vraisemblable?
\end{enumerate}

\begin{reponse}
\textbf{(a)} Oui. Dans le modèle de Solow, le taux d'investissement est directement relié à l'accumulation de capital et au niveau de $K/N$ et $Y/N$ à long terme. Si le taux d'investissement demeure bas, $K/N$ et $Y/N$ seront négativement affectés.

\textbf{(b)} La courbe d'épargne du Québec se déplace vers la \textbf{gauche}. Comme le Québec et le ROC forment le \guillemotleft~monde~\guillemotright\ (Canada), la courbe mondiale $S$ se déplace aussi vers la gauche. Le taux d'intérêt canadien \textbf{augmente}. L'investissement baisse au Québec \textbf{mais aussi} dans le reste du Canada (les deux font face au même $r$ plus élevé). Cet effet peut expliquer un investissement faible au Québec en général, mais techniquement il n'est pas clair qu'il explique pourquoi il serait plus faible qu'au ROC.

\textbf{(c)} Dans le ROC, la courbe $I$ se déplace vers la \textbf{droite} (la productivité marginale du capital augmente). Même chose pour le Canada au complet. Le taux d'intérêt canadien \textbf{augmente}. L'investissement au Québec \textbf{baisse} (hausse de $r$ sans déplacement de $I$ au Québec). Dans le ROC, l'investissement augmente (car la hausse de $r$ est partiellement compensée par le déplacement de $I$ vers la droite). C'est une explication potentielle.

\textbf{(d)} Les deux explications prévoient une baisse de l'investissement au Québec, mais des prédictions différentes pour le \textbf{compte courant}. La baisse de l'épargne ($S\downarrow$) implique un déficit courant plus important. L'augmentation de $A$ dans le ROC ($I$ vers la droite) implique que le déficit courant du Québec se \textbf{résorbe} (l'investissement au Québec baisse plus que l'épargne). Les données montrent un déficit croissant $\to$ la première explication (baisse de l'épargne) est plus vraisemblable.
\end{reponse}

\shortq{En réponse au ralentissement économique mondial de 2008, le premier ministre britannique Gordon Brown a appelé à des coupures d'impôts coordonnées au niveau mondial.}

\begin{enumerate}[label=(\alph*)]
   \item Tous les pays implémentent des coupures d'impôts (chèques aux ménages). À long terme, en gardant $Y$ constant, quel sera l'impact sur les taux d'intérêts mondiaux? Illustrez avec un seul graphique $S$-$I$ pour l'économie mondiale.

   \item Gordon Brown est incapable de convaincre ses homologues, et le Royaume-Uni est le seul pays à implémenter une baisse d'impôt. En utilisant le graphique à trois panneaux, montrez l'impact sur le compte courant du R.-U.\ et le taux d'intérêt mondial. (Le R.-U.\ a un déficit courant.)

   \item Le gouvernement chinois annonce un programme de dépenses d'infrastructure ($G\uparrow$) pour stimuler son économie. En faisant l'hypothèse que ces mesures augmentent la PTF ($A$) de manière permanente, quel sera l'impact sur l'emploi et les salaires réels en Chine?

   \item Quel devrait être l'impact de ce qui précède sur l'épargne privée en Chine?

   \item En vous limitant à l'impact sur $S_G$ et sur $I$, déterminez l'effet de ces mesures fiscales chinoises sur le compte courant chinois et le taux d'intérêt mondial. (La Chine a un surplus du compte courant.)
\end{enumerate}

\begin{reponse}
\textbf{(a)} La courbe $S = Y - C - G$ se déplace vers la \textbf{gauche} : une partie des coupures d'impôt sera consommée, l'épargne gouvernementale ($S_G$) chute, et l'épargne privée ($S_P$) augmente mais pas assez pour compenser. Le taux d'intérêt mondial \textbf{augmente}.

\textbf{(b)} Le R.-U.\ seul : la courbe d'épargne du R.-U.\ se déplace vers la gauche, et celle du monde aussi (car le R.-U.\ en fait partie). Le taux d'intérêt mondial augmente (légèrement, car le R.-U.\ est une grande économie). Le compte courant du R.-U.\ \textbf{se détériore} (déficit plus important) : $S\downarrow$ $\to$ $CA = S - I$ diminue.

\textbf{(c)} L'augmentation de $A$ fait augmenter la demande de travail $\to$ $N$ et $W/P$ augmentent. Les travailleurs qui anticipent des revenus futurs plus élevés pourraient aussi réduire leur offre de travail (effet revenu), ce qui pousserait $W/P$ encore plus haut. L'effet dominant est probablement celui de la demande.

\textbf{(d)} Si les travailleurs anticipent une augmentation future de $A$ (et donc de leurs revenus), ils voudront commencer à consommer davantage dès aujourd'hui $\to$ l'épargne privée \textbf{baisse}. Alternativement, si l'augmentation des salaires est perçue comme immédiate, une partie sera épargnée $\to$ épargne privée augmente. L'important est d'être cohérent.

\textbf{(e)} $G$ augmente $\to$ $S_G = T - G$ diminue $\to$ la courbe $S$ en Chine se déplace vers la gauche. $A$ augmente $\to$ la courbe $I$ se déplace vers la droite (productivité marginale du capital plus élevée). Ces deux forces font \textbf{augmenter} $r$ au niveau mondial. Pour le reste du monde, le CA augmente. Pour la Chine, le CA \textbf{diminue} (surplus plus petit ou déficit).
\end{reponse}

\shortq{Selon \textit{The Economist}, la reprise au Japon est solide, tirée par la confiance des entreprises et l'investissement.}

\begin{enumerate}[label=(\alph*)]
   \item En économie fermée : la confiance des entreprises s'améliore, augmentant l'investissement désiré. Montrez l'impact sur $r$, $S$ et $I$ dans le diagramme $S$-$I$.

   \item Si cette confiance accrue est accompagnée d'une confiance accrue des consommateurs par rapport à leurs revenus futurs, quel serait l'impact supplémentaire? Que pouvons-nous dire de $I$ au point final par rapport au point initial?

   \item Le Japon est en réalité une grande économie ouverte. Montrez graphiquement (trois panneaux) l'effet combiné de la confiance accrue des entreprises et des consommateurs. Quel est l'effet sur le compte courant japonais?

   \item Quel devrait être l'impact de cette confiance accrue des consommateurs japonais sur le compte courant \textbf{américain}?
\end{enumerate}

\begin{reponse}
\textbf{(a)} La courbe $I$ se déplace vers la droite. À l'ancien taux $r_A$, il n'y a plus assez d'épargne pour financer la demande croissante d'investissement. Le taux d'intérêt augmente à $r_B$. L'investissement et l'épargne augmentent tous les deux ($I_B = S_B > I_A = S_A$).

\textbf{(b)} Les consommateurs anticipent des revenus plus élevés $\to$ ils consomment plus aujourd'hui $\to$ l'épargne $S$ se déplace vers la \textbf{gauche}. Le taux $r$ augmente encore plus ($r_C > r_B$). L'investissement au point C par rapport au point A est \textbf{ambigu} : la hausse de $r$ déprime l'investissement, mais le déplacement de $I$ vers la droite le stimule. L'effet net dépend de l'ampleur relative des deux mouvements.

\textbf{(c)} En économie ouverte (trois panneaux) : au Japon, $I$ vers la droite et $S$ vers la gauche $\to$ pression à la hausse sur $r^w$. Le taux d'intérêt mondial augmente (mais moins qu'en économie fermée car le reste du monde absorbe une partie du choc). Le Japon passe en \textbf{déficit du compte courant} : les firmes et consommateurs japonais demandent plus de ressources que l'épargne nationale n'en fournit $\to$ le capital étranger afflue.

\textbf{(d)} La hausse de $r^w$ incite les Américains à épargner plus et investir moins. Les dépenses aux É.-U.\ diminuent (baisse de $C$ et $I$). Le déficit du compte courant américain \textbf{se réduit} ($NX_2 = S_2 - I_2 > NX_1 = S_1 - I_1$, car $S$ augmente et $I$ diminue).
\end{reponse}

\shortq{La Lettonie a récemment connu une expansion rapide qui a conduit à un déficit du compte courant non soutenable (24.2\,\% du PIB). Le FMI recommande une politique budgétaire restrictive pour réduire le déficit courant. Expliquez comment une politique budgétaire restrictive contribuerait à réduire le déficit du compte courant de la Lettonie, en utilisant le diagramme $S$-$I$ en économie ouverte. (La Lettonie est une petite économie ouverte.)}

\begin{reponse}
La politique budgétaire restrictive signifie une augmentation de $T$ ou une diminution de $G$.
\begin{itemize}
   \item De la définition de l'épargne nationale : $S = Y - C - G$ (ou $S = S_P + S_G$ avec $S_G = T - G$). Une diminution de $G$ augmente $S_G$ directement. Une augmentation de $T$ augmente $S_G$, mais l'effet sur $S$ national dépend de la réaction de $C$ : si les ménages réduisent leur consommation, $S_P$ diminue un peu, mais moins que la hausse de $S_G$ $\to$ net : $S$ augmente.
   \item La courbe $S$ se déplace vers la \textbf{droite} dans le diagramme.
   \item La Lettonie est une \textbf{petite} économie ouverte $\to$ elle n'affecte pas le taux d'intérêt mondial $r^w$. Au même $r^w$, $S$ est plus élevé mais $I$ est inchangé.
   \item $CA = S - I$ augmente $\to$ le déficit du compte courant \textbf{se réduit}.
\end{itemize}
La politique restrictive réduit les dépenses domestiques, ce qui réduit la demande d'importations et améliore la balance courante.
\end{reponse}
