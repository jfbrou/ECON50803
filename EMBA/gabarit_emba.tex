% Nom du module en surtitre, titre sur deux lignes et filet or du gabarit.
\renewcommand{\titleframe}{%
  \begin{frame}
    \begin{tikzpicture}[remember picture,overlay]
      \node[inner sep=0pt,font=\small\bfseries,text=black!70]
        at ([yshift=-.15\paperheight]current page.north) {\insertsubtitle};
      \node[align=center,inner sep=0pt,
        font=\LARGE\bfseries,text=black]
        at ([yshift=-.32\paperheight]current page.north) {\inserttitle};
      \draw[EMBAor,line width=.00556\paperheight]
        ([xshift=.02453\paperwidth,yshift=-.455\paperheight]current page.north west) --
        ([xshift=-.02453\paperwidth,yshift=-.455\paperheight]current page.north east);
      \node[align=center,inner sep=0pt,text=HECtext]
        at ([yshift=-.67\paperheight]current page.north)
        {{\large\insertauthor}\\[4pt]{\normalsize\textcolor{black!70}{\insertdate}}};
    \end{tikzpicture}
  \end{frame}%
}

% Le filet et le titre suivent la marge des diapositives de contenu.

  \setbeamercolor{background canvas}{bg=black}%
  \begin{frame}
    \begin{tikzpicture}[remember picture,overlay]
      \coordinate (repere) at ([xshift=1.2cm,yshift=-.39\paperheight]current page.north west);
      \draw[EMBAor,line width=1.5pt] (repere) -- ++(2.5cm,0) coordinate (fin);
      \ifstrempty{#1}{}{%
        \node[anchor=west,inner sep=0pt,font=\normalsize\bfseries,text=EMBAor]
          at ([xshift=.28cm]fin) {#1};}
      \node[anchor=north west,inner sep=0pt,text width=\textwidth,align=flush left,
        font=\LARGE\bfseries,text=white] at ([yshift=-.42cm]repere) {#2};
    \end{tikzpicture}
  \end{frame}%
  \endgroup
}
