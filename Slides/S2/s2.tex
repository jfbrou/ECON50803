\begin{frame}[t]{La fonction de production Cobb-Douglas}\examtag
   La forme fonctionnelle la plus utilisée en macroéconomie :
   \begin{equation*}
      Y = A \cdot K^{\alpha} \cdot L^{1-\alpha} \quad \text{avec} \quad \alpha \approx 0.3
   \end{equation*}
   \vspace{-0.5cm}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item $Y =$ quantité produite au cours d'une année (PIB réel)
      \item $K =$ stock de capital
      \item $L =$ main-d'œuvre (nombre de travailleurs)
      \item $A =$ productivité totale des facteurs (PTF) --- mesure d'efficience
   \end{itemize}
   \vspace{0.15cm}
   \begin{keyinsight}
      Les exposants $\alpha$ et $1-\alpha$ correspondent aux \textbf{parts du revenu} versées au capital ($\sim$30\,\%) et au travail ($\sim$70\,\%)
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Qu'est-ce que la PTF ($A$)\,?}\examtag
   \small
   Deux boulangeries avec les \textbf{mêmes ingrédients} (farine, beurre, œufs, employés).
   \vspace{0.25cm}
   \begin{columns}[T]
      \begin{column}{0.47\textwidth}
         \centering
         \textcolor{HECgray}{\large\textbf{Boulangerie \#1}}\\[3pt]
         \textcolor{HECgray}{PTF faible}\\[4pt]
         \footnotesize
         Recettes de base\\
         Fours mal calibrés\\
         Organisation chaotique\\[4pt]
         $\to$ 100 pains / jour
      \end{column}
      \begin{column}{0.06\textwidth}
         \centering
         \vspace{0.8cm}
         {\Large\textcolor{HECgray}{vs}}
      \end{column}
      \begin{column}{0.47\textwidth}
         \centering
         \textcolor{HECgreen}{\large\textbf{Boulangerie \#2}}\\[3pt]
         \textcolor{HECgreen}{PTF élevée}\\[4pt]
         \footnotesize
         Techniques de levain maîtrisées\\
         Fours optimisés\\
         Processus rodé\\[4pt]
         $\to$ 200 pains / jour
      \end{column}
   \end{columns}
   \vspace{0.25cm}
   \begin{keyinsight}
      La PTF ($A$) capte \textbf{tout ce qui n'est pas $K$ ou $L$} : technologie, savoir-faire, gestion, organisation\ldots\ C'est la capacité à faire \textbf{plus avec les mêmes ressources}.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Rendements d'échelle constants}\examtag
   Si on \textbf{double} tous les intrants, la production \textbf{double} :
   \vspace{0.1cm}
   \begin{equation*}
      A \cdot (2K)^{\alpha} \cdot (2L)^{1-\alpha} = 2^{\alpha} \cdot 2^{1-\alpha} \cdot A K^{\alpha} L^{1-\alpha} = 2Y
   \end{equation*}
   \\ \vspace{0.2cm}
   \textbf{L'argument de réplication :} comment doubler la production d'un McDonald\,? En construire un autre identique de l'autre côté de la rue\,!
   \vspace{0.2cm}
   \begin{keyinsight}
      La \textbf{taille} d'un pays ne détermine pas son \textbf{niveau de vie}. Le Liechtenstein, Singapour, et le Luxembourg sont tous petits, mais très riches.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Comptabilité de la croissance}\examtag
   \small
   En partant de la fonction de production :
   \begin{equation*}
      Y = A \cdot K^{\alpha} \cdot L^{1-\alpha}
   \end{equation*}
   on prend le \textbf{taux de croissance} de chaque côté :
   \begin{defbox}[Décomposition de Solow]
      \vspace{-0.1cm}
      \begin{equation*}
         \underbrace{\pctchg{Y}}_{\text{croissance du PIB}} \;\approx\; \underbrace{\pctchg{A}}_{\text{PTF}} \;+\; \underbrace{\alpha \cdot \pctchg{K}}_{\text{capital}} \;+\; \underbrace{(1 - \alpha) \cdot \pctchg{L}}_{\text{travail}}
      \end{equation*}
   \end{defbox}
   \begin{itemize}
      \setlength\itemsep{0.1cm}
      \item On observe $\pctchg{Y}$, $\pctchg{K}$ et $\pctchg{L}$ $+$ on connaît $\alpha$ (part du capital dans le revenu)
      \item On \textbf{déduit} $\pctchg{A}$ comme résidu : $\pctchg{A} = \pctchg{Y} - \alpha \cdot \pctchg{K} - (1 - \alpha) \cdot \pctchg{L}$
   \end{itemize}
\end{frame}

\begin{frame}[t]{Du PIB total au PIB par habitant}
   \small
   Diviser par la population ($N$) donne :
   \begin{equation*}
      \frac{Y}{N} = \frac{A \cdot K^{\alpha} \cdot L^{1 - \alpha}}{N} \quad \to \quad \frac{Y}{N} = A \cdot \left(\frac{K}{L}\right)^{\alpha} \cdot \frac{L}{N}
   \end{equation*}
   \textbf{Trois déterminants du niveau de vie :}
   \vspace{0.1cm}
   \begin{enumerate}
      \setlength\itemsep{0.1cm}
      \item $A =$ productivité (PTF)
      \item $K/L =$ capital par travailleur
      \item $L/N =$ taux d'emploi (démographie + participation)
   \end{enumerate}
   \vspace{0.5cm}
   Mais il y a un problème avec cette décomposition\ldots
\end{frame}

\begin{frame}[t]{Capital ou productivité\,?}\examtag
   \small
   Quand une économie devient plus efficiente ($A \uparrow$) :
   \vspace{0.15cm}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item La production augmente ($Y \uparrow$)
      \item Les entreprises investissent davantage ($I \uparrow$ et donc $K \uparrow$)
      \item $K/L$ augmente $\to$ la décomposition naïve \textbf{attribue le mérite au capital\,!} \\
   \end{itemize}
   \vspace{0.25cm}
   En réécrivant la fonction de production :
   \begin{equation*}
      \frac{Y}{N} = A \cdot \left(\frac{K}{L}\right)^{\alpha} \cdot \frac{L}{N} = A \cdot \left(\frac{K}{Y} \cdot \frac{Y}{L}\right)^{\alpha} \cdot \frac{L}{N} = A \cdot \left(\frac{K}{Y} \cdot \frac{Y}{N} \cdot \frac{N}{L}\right)^{\alpha} \cdot \frac{L}{N}
   \end{equation*}
   Encore un peu d'algèbre\ldots
   \begin{equation*}
      \frac{Y}{N} = A^{\frac{1}{1 - \alpha}} \cdot \left(\frac{K}{Y}\right)^{\frac{\alpha}{1 - \alpha}} \cdot \frac{L}{N}
   \end{equation*}
\end{frame}

\begin{frame}[t]{Rendre à César ce qui est à César}\examtag
   Fonction de \textbf{niveau de vie :}
   \begin{equation*}
      \frac{Y}{N} = A^{\frac{1}{1 - \alpha}} \cdot \left(\frac{K}{Y}\right)^{\frac{\alpha}{1 - \alpha}} \cdot \frac{L}{N}
   \end{equation*}
   \textbf{Trois déterminants du niveau de vie :}
   \vspace{0.15cm}
   \begin{enumerate}
      \setlength\itemsep{0.15cm}
      \item $A =$ productivité (PTF)
      \item $K/Y =$ profondeur du capital
      \item $L/N =$ taux d'emploi (démographie + participation) \\
   \end{enumerate}
   \vspace{0.35cm}
   On mesure $Y / N$, $K / Y$, et $L / N$ \textbf{directement}, et on en déduit $A$.
   \begin{itemize}
      \item $A$ : mesure de notre ignorance\ldots
   \end{itemize}
\end{frame}

\begin{frame}[t]{Le capital comme amplificateur de productivité}\examtag
   Décomposer la \textbf{croissance} du PIB par habitant :
   \begin{equation*}
      \underbrace{\pctchg{(Y/N)}}_{\text{PIB/hab.}} \;\approx\; \underbrace{\tfrac{1}{1 - \alpha} \cdot \pctchg{A}}_{\text{productivité}} \;+\; \underbrace{\tfrac{\alpha}{1 - \alpha} \cdot \pctchg{(K/Y)}}_{\text{capital/PIB}} \;+\; \underbrace{\pctchg{(L/N)}}_{\text{taux d'emploi}}
   \end{equation*}
   \begin{keyinsight}
      Le coefficient $\frac{1}{1 - \alpha}$ devant $A$ est un \textbf{amplificateur} : $A \uparrow \;\to\; \underbrace{Y \uparrow \;\to\; K \uparrow \;\to\; Y \uparrow}_{\text{amplification}}$. \\
      
      Quand $A$ augmente de 1\,\%, le PIB/hab. augmente de $\frac{1}{1 - \alpha} \approx 1.4$\,\%.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{D'où viennent les différences de revenus\,?}
   Qu'est-ce qui explique les massifs écarts de revenus \textbf{entre pays}\,?
   \begin{equation*}
      \frac{Y}{N} = A^{\frac{1}{1 - \alpha}} \cdot \left(\frac{K}{Y}\right)^{\frac{\alpha}{1 - \alpha}} \cdot \frac{L}{N}

   \end{equation*}
\end{frame}

\begin{frame}[t]{Le modèle de Solow}\examtag
   \textbf{Question clé :} les pays peuvent-ils croître en accumulant plus de capital\,?
   \vspace{0.25cm}

   \textbf{Ingrédients du modèle :}
   \vspace{0.1cm}
   \begin{itemize}
      \setlength\itemsep{0.1cm}
      \item Fonction de production (rendements décroissants du capital)
      \item Accumulation de capital (investissement $-$ dépréciation)
   \end{itemize}
   \vspace{0.25cm}

   \textbf{Hypothèses simplificatrices :} $Y = C + I$
   \vspace{0.1cm}
   \begin{itemize}
      \setlength\itemsep{0.1cm}
      \item Pas de gouvernement ($G = 0$) $+$ économie fermée ($XN = 0$)
   \end{itemize}
   \vspace{0.25cm}

   \textbf{Conclusions surprenantes :}
   \vspace{0.1cm}
   \begin{itemize}
      \setlength\itemsep{0.1cm}
      \item Le capital est crucial au \textbf{niveau} de vie\ldots
      \item \ldots mais l’investissement ne peut propulser la \textbf{croissance} à long terme\,!
   \end{itemize}
\end{frame}

\begin{frame}[t]{Les rendements décroissants limitent l'accumulation}\examtag
   Plus de $K$ $\to$ plus de $Y$ $\to$ plus de $I$\ldots mais \textbf{de moins en moins}.
   \vspace{0.2cm}
   \begin{enumerate}
      \setlength\itemsep{0.2cm}
      \item $K$ augmente $\to$ $Y$ augmente (mais à un rythme décroissant)
      \item $I$ augmente aussi, mais de moins en moins vite
      \item Le drain ($\delta \cdot K$) augmente \textbf{proportionnellement} à $K$
      \item À un moment : investissement $=$ dépréciation $\to$ \textbf{état stationnaire}
   \end{enumerate}
   \vspace{0.2cm}
   \begin{defbox}[État stationnaire]
      Le niveau de capital par travailleur où l'investissement compense exactement la dépréciation et la croissance de la population. Le capital $K/L$ et la production $Y/L$ par travailleur ne changent plus.
   \end{defbox}
\end{frame}

\begin{frame}[t]{Le profil de croissance selon Solow}\examtag
   \textbf{Phase de rattrapage} \textcolor{HECgray}{(capital sous son état stationnaire)}
   \vspace{0.15cm}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item Croissance \textbf{rapide} tirée par l'accumulation de $K$
      \item Ex. : Allemagne/Japon après-guerre, Chine 1980--2010
   \end{itemize}
   \vspace{0.25cm}
   \textbf{État stationnaire} \textcolor{HECgray}{(long terme)}
   \vspace{0.15cm}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item Les rendements décroissants épuisent le potentiel du capital
      \item La croissance de long terme ne peut provenir que de la \textbf{productivité} ($A$)
   \end{itemize}
   \vspace{0.25cm}
   \begin{keyinsight}
      Le capital propulse la croissance \textbf{temporairement}, mais pas \textbf{indéfiniment}.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Leçons du modèle de Solow}\examtag
   \small
   Pour des économies aux \textbf{fondamentaux identiques}, le modèle de Solow prédit que :
   \vspace{0.25cm}
   \begin{enumerate}
      \setlength\itemsep{0.25cm}
      \item \textbf{Le point de départ n'a pas d'importance à long terme}\\
      L'économie converge vers le même état stationnaire $\to$ la reprise d'après-guerre
      \item \textbf{Plus d'investissement $\to$ plus riche, mais pas de croissance permanente}\\
      Les rendements marginaux décroissants l'emportent toujours
      \item \textbf{Plus de croissance démographique $\to$ pays plus pauvre}\\
      Le drain de la baignoire coule plus vite $\to$ l'état stationnaire est plus bas
      \item \textbf{Seule la productivité ($A$) soutient la croissance de long terme}\\
      Le capital explique le \textit{niveau}, la productivité explique la \textit{croissance}
   \end{enumerate}
\end{frame}

\begin{frame}[t]{Comment relier croissance économique et émissions de CO$_2$\,?}\examtag
   \small
   \begin{defbox}[Identité de Kaya]
      \vspace{-0.2cm}
      \begin{equation*}
         \underbrace{\vphantom{\frac{Y}{N}}\text{CO}_2}_{\substack{\text{émissions}\\[1pt]\mbox{}}}
         \;=\;
         \underbrace{\vphantom{\frac{Y}{N}}N}_{\substack{\text{population}\\[1pt]\mbox{}}}
         \;\times\;
         \underbrace{\frac{Y}{N}}_{\substack{\text{niveau}\\[-1pt]\text{de vie}}}
         \;\times\;
         \underbrace{\frac{E}{Y}}_{\substack{\text{intensité}\\[-1pt]\text{énergétique}}}
         \;\times\;
         \underbrace{\frac{\text{CO}_2}{E}}_{\substack{\text{intensité}\\[-1pt]\text{carbone}}}
      \end{equation*}
   \end{defbox}
   \vspace{0.2cm}
   \textbf{Quatre leviers} pour réduire les émissions :
   \vspace{0.1cm}
   \begin{enumerate}
      \setlength\itemsep{0.1cm}
      \item Réduire la \textbf{population} (lent, éthiquement complexe)
      \item Réduire le \textbf{niveau de vie} (politiquement difficile)
      \item Réduire l'\textbf{intensité énergétique} (efficience énergétique)
      \item Réduire l'\textbf{intensité carbone} (énergies renouvelables)
   \end{enumerate}

\begin{frame}[t]{Pourquoi s'attendre au découplage\,?}
   Le découplage semble réel dans les économies avancées. Est-ce un accident ou y a-t-il des raisons \textbf{fondamentales} de s'y attendre\,?
   \vspace{0.5cm}
   \begin{enumerate}
      \setlength\itemsep{0.5cm}
      \item \textbf{L'innovation} --- la croissance repose sur la productivité ($A$), qui nous permet de faire \textbf{plus avec moins} de ressources/énergie.
      \item \textbf{La richesse libère des ressources} --- les pays riches ont les moyens d'investir dans la R\&D verte, les infrastructures propres, etc..
      \item \textbf{Les lois d'échelle} --- la production à grande échelle est plus efficiente énergétiquement (la perte de chaleur croît avec la surface, mais la capacité croît avec le volume).
   \end{enumerate}
\end{frame}

\begin{frame}[t]{Le paradoxe de Jevons}
   \small
   En 1865, William S.\ Jevons observe que les machines à vapeur deviennent de \textbf{plus en plus efficaces}\ldots\ mais que la consommation totale de charbon \textbf{augmente}\ldots
   \vspace{0.3cm}
   \begin{warning}[Effet rebond]
      Si l'efficience réduit le \textbf{coût d'usage} d'une ressource, la demande peut augmenter suffisamment pour \textbf{annuler} une partie des gains.

      \medskip
      L'ampleur de l'effet rebond dépend de l'\textbf{élasticité-prix de la demande} : plus la demande est élastique, plus le rebond est fort.

      \medskip
      \textit{Exemple :} l'éclairage LED consomme 80\,\% moins d'énergie par ampoule --- mais nous éclairons \textbf{beaucoup plus} de choses qu'avant.
   \end{warning}
\end{frame}
