\begin{frame}[t]{Long terme vs court terme}
   \small
   \textbf{Rappel :} la production d'un pays dépend de trois facteurs :
   \[
      Y = A \cdot K^{\alpha} \cdot N^{1-\alpha}
   \]

   \vspace{0.2cm}
   \begin{columns}[T]
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECnavy}{Long terme} (Séances 2--3)}
         \vspace{0.15cm}
         \begin{itemize}
            \setlength\itemsep{0.15cm}
            \item Croissance de $A$, $K$, $N$
            \item \textbf{Tendance} haussière du PIB
            \item Déterminants : institutions, innovation, capital humain
         \end{itemize}
      \end{column}
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECcoral}{Court terme} (Séance 4)}
         \vspace{0.15cm}
         \begin{itemize}
            \setlength\itemsep{0.15cm}
            \item \textbf{Fluctuations} autour de la tendance
            \item Récessions, expansions, surchauffe
            \item Déterminants : chocs, confiance, politique économique
         \end{itemize}
      \end{column}
   \end{columns}
\end{frame}

\begin{frame}[t]{Le PIB potentiel}
   \begin{defbox}[PIB potentiel ($Y^{\text{POT}}$)]
      Le niveau de production lorsque tous les facteurs (travail, capital) sont utilisés à leur \textbf{taux normal}. Ce n'est \textbf{pas} le maximum absolu --- c'est la capacité \guillemotleft~de croisière~\guillemotright.
   \end{defbox}
   \vspace{0.3cm}
   \begin{itemize}
      \setlength\itemsep{0.3cm}
      \item $Y^{\text{POT}}$ croît lentement avec $A$, $K$ et $N$ (tendance de long terme)
      \item Le PIB \textbf{réel} ($Y$) fluctue autour de $Y^{\text{POT}}$
      \item $Y > Y^{\text{POT}}$ $\to$ surchauffe \quad;\quad $Y < Y^{\text{POT}}$ $\to$ récession
   \end{itemize}
\end{frame}

\begin{frame}[t]{L'écart de production (\textit{output gap})}
   \begin{defbox}
      \[
         \text{Écart de production} = \frac{Y - Y^{\text{POT}}}{Y^{\text{POT}}} \times 100
      \]
   \end{defbox}
   \vspace{0.3cm}
   \begin{itemize}
      \setlength\itemsep{0.3cm}
      \item \textbf{Positif :} l'économie produit au-delà de sa capacité $\to$ pression inflationniste
      \item \textbf{Négatif :} l'économie sous-utilise ses ressources $\to$ chômage élevé
      \item \textbf{Zéro :} l'économie opère à son potentiel
   \end{itemize}
\end{frame}

   \begin{keyinsight}
      Les fluctuations sont \textbf{temporaires} mais \textbf{coûteuses}. Les récessions détruisent des emplois, des entreprises et de la richesse. Les expansions créent des risques de surchauffe et d'inflation.
   \end{keyinsight}

\begin{frame}[t]{Un cadre pour analyser les fluctuations}
   \small
   En microéconomie, on utilise l'offre et la demande pour analyser un marché.

   \vspace{0.2cm}
   En macroéconomie, on utilise l'\textbf{offre agrégée} (OA) et la \textbf{demande agrégée} (DA) pour analyser l'économie dans son ensemble.
   \vspace{0.3cm}
   \begin{itemize}
      \setlength\itemsep{0.25cm}
      \item \textbf{Axe horizontal :} PIB réel ($Y$) --- la production totale
      \item \textbf{Axe vertical :} Niveau des prix ($P$) --- le niveau général des prix
   \end{itemize}
   \vspace{0.2cm}
   \begin{warning}[Attention]
      $P$ est le \textbf{niveau} des prix, pas le \textit{taux} d'inflation. Une hausse de $P$ signifie de l'inflation; un $P$ stable signifie zéro inflation.
   \end{warning}
\end{frame}

\begin{frame}{La demande agrégée (DA)}
   \begin{columns}[c]
      \begin{column}{0.52\textwidth}
         \centering
         \begin{tikzpicture}[scale=1.1]
            % Axes
            \draw[-{Stealth}, thick] (0,0) -- (5.5,0) node[right] {$Y$};
            \draw[-{Stealth}, thick] (0,0) -- (0,4.5) node[above] {$P$};
            % DA curve
            \draw[HECnavy, very thick] (0.5,4) -- (5,0.5) node[right] {DA};
         \end{tikzpicture}
      \end{column}
      \begin{column}{0.48\textwidth}
         \small
         \textbf{Pente négative :} quand $P$ augmente\ldots
         \vspace{0.15cm}
         \begin{enumerate}
            \setlength\itemsep{0.2cm}
            \item \textbf{Effet de richesse :} la richesse réelle $\downarrow$ $\to$ $C \downarrow$
            \item \textbf{Effet de taux d'intérêt :} les taux $\uparrow$ $\to$ $I \downarrow$
            \item \textbf{Effet de change :} le dollar s'apprécie $\to$ $NX \downarrow$
         \end{enumerate}
      \end{column}
   \end{columns}
\end{frame}

\begin{frame}[t]{Ce qui déplace la courbe DA}
   \footnotesize
   Tout ce qui modifie les \textbf{dépenses souhaitées} à un niveau de prix donné :
   \vspace{0.15cm}
   \begin{columns}[T]
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECgreen}{DA vers la droite ($\to$)}}
         \vspace{0.1cm}
         \begin{itemize}
            \setlength\itemsep{0.1cm}
            \item Hausse de la confiance ($C \uparrow$)
            \item Optimisme des entreprises ($I \uparrow$)
            \item Hausse des dépenses publiques ($G \uparrow$)
            \item Boom chez les partenaires ($NX \uparrow$)
            \item Pol.\ monétaire expansionniste (taux $\downarrow$)
         \end{itemize}
      \end{column}
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECcoral}{DA vers la gauche ($\leftarrow$)}}
         \vspace{0.1cm}
         \begin{itemize}
            \setlength\itemsep{0.1cm}
            \item Effondrement de la confiance ($C \downarrow$)
            \item Crise financière ($I \downarrow$)
            \item Austérité budgétaire ($G \downarrow$)
            \item Récession chez les partenaires ($NX \downarrow$)
            \item Pol.\ monétaire restrictive (taux $\uparrow$)
         \end{itemize}
      \end{column}
   \end{columns}
   \vspace{0.15cm}
   \begin{bizimplication}
      \footnotesize Une hausse de la confiance des consommateurs déplace DA vers la droite. La confiance est un \textbf{indicateur avancé} suivi de près par les investisseurs.
   \end{bizimplication}
\end{frame}

\begin{frame}{L'offre agrégée de court terme (OA$_{\text{CT}}$)}
   \begin{columns}[c]
      \begin{column}{0.52\textwidth}
         \centering
         \begin{tikzpicture}[scale=1.1]
            % Axes
            \draw[-{Stealth}, thick] (0,0) -- (5.5,0) node[right] {$Y$};
            \draw[-{Stealth}, thick] (0,0) -- (0,4.5) node[above] {$P$};
            % OA_CT curve
            \draw[HECcoral, very thick] (0.5,0.5) -- (5,4) node[right] {OA$_{\text{CT}}$};
         \end{tikzpicture}
      \end{column}
      \begin{column}{0.48\textwidth}
         \small
         \textbf{Pente positive :} quand $P$ augmente\ldots
         \vspace{0.2cm}
         \begin{itemize}
            \setlength\itemsep{0.25cm}
            \item Les \textbf{salaires nominaux} sont fixés par contrat $\to$ rigides à court terme
            \item $P \uparrow$ mais $w$ fixe $\to$ le coût réel du travail $\downarrow$
            \item Les entreprises embauchent plus $\to$ $Y \uparrow$
         \end{itemize}
         \vspace{0.2cm}
         {\footnotesize\textcolor{HECgray}{Mécanisme clé : la \textbf{rigidité des salaires nominaux}}}
      \end{column}
   \end{columns}
\end{frame}

\begin{frame}[t]{Ce qui déplace la courbe OA}
   \small
   Tout ce qui modifie les \textbf{coûts de production} des entreprises :
   \vspace{0.2cm}
   \begin{columns}[T]
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECgreen}{OA vers la droite ($\to$)}}
         \vspace{0.15cm}
         \begin{itemize}
            \setlength\itemsep{0.15cm}
            \item Progrès technologique ($A \uparrow$)
            \item Baisse du prix du pétrole
            \item Baisse des salaires nominaux
            \item Amélioration des chaînes d'approvisionnement
         \end{itemize}
      \end{column}
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECcoral}{OA vers la gauche ($\leftarrow$)}}
         \vspace{0.15cm}
         \begin{itemize}
            \setlength\itemsep{0.15cm}
            \item Hausse du prix du pétrole
            \item Hausse des salaires nominaux ($w \uparrow$)
            \item Perturbation des chaînes d'approvisionnement
            \item Catastrophe naturelle
         \end{itemize}
      \end{column}
   \end{columns}
\end{frame}

         \small
         \textbf{Verticale} à $Y^{\text{POT}}$

         \vspace{0.2cm}
         Pourquoi?
         \vspace{0.15cm}
         \begin{itemize}
            \setlength\itemsep{0.2cm}
            \item À long terme, les salaires \textbf{s'ajustent pleinement} aux prix
            \item Le coût réel du travail revient à l'équilibre
            \item $Y$ revient à $Y^{\text{POT}}$, quel que soit $P$
         \end{itemize}
      \end{column}
   \end{columns}
\end{frame}

\begin{frame}[t]{Résumé : les trois courbes}
   \small
   \begin{defbox}[Les composantes du modèle OA-DA]
      \begin{itemize}
         \setlength\itemsep{0.2cm}
         \item \textbf{Demande agrégée (DA)} --- pente négative. Relie le niveau des prix à la quantité totale de biens et services demandés ($C + I + G + NX$)
         \item \textbf{Offre agrégée de court terme (OA$_{\text{CT}}$)} --- pente positive. Les salaires nominaux rigides font que les entreprises produisent \textit{plus} quand $P$ augmente
         \item \textbf{Offre agrégée de long terme (OA$_{\text{LT}}$)} --- verticale à $Y^{\text{POT}}$. Les salaires s'ajustent complètement $\to$ la production retourne au potentiel
      \end{itemize}
   \end{defbox}
\end{frame}

\begin{frame}{}
   \begin{exercise}
      Si le gouvernement augmente les dépenses publiques ($G$), que se passe-t-il à court terme?
      \vspace{0.2cm}
      \begin{enumerate}
         \setlength\itemsep{0.15cm}
         \item Quelle courbe se déplace? Dans quelle direction?
         \item Quel est l'effet sur $Y$ et $P$?
         \item L'économie est-elle en surchauffe ou en récession?
      \end{enumerate}
   \end{exercise}
\end{frame}

\begin{frame}[t]{Résumé : demande vs offre}
   \vspace{0.3cm}
   \centering
   \small
   \renewcommand{\arraystretch}{1.6}
   \begin{tabularx}{0.95\textwidth}{>{\bfseries}X c c c c}
      \toprule
      & \textbf{DA $\to$} & \textbf{DA $\leftarrow$} & \textbf{OA $\to$} & \textbf{OA $\leftarrow$} \\
      \midrule
      Production ($Y$) & $\uparrow$ & $\downarrow$ & $\uparrow$ & $\downarrow$ \\
      Prix ($P$) & $\uparrow$ & $\downarrow$ & $\downarrow$ & $\uparrow$ \\
      \midrule
      $Y$ et $P$\ldots & \multicolumn{2}{c}{même direction} & \multicolumn{2}{c}{directions opposées} \\

   \end{tabularx}

   \begin{keyinsight}
      La direction de l'inflation nous dit le type de choc. Si $Y \downarrow$ et $P \downarrow$ : choc de demande. Si $Y \downarrow$ et $P \uparrow$ : choc d'offre.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Le dilemme de la banque centrale}
   \small
   \begin{columns}[T]
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECgreen}{Choc de demande}}
         \vspace{0.15cm}
         \begin{itemize}
            \setlength\itemsep{0.15cm}
            \item $Y \downarrow$ et $P \downarrow$
            \item Stimuler DA $\to$ relance $Y$ \textbf{et} réduit le risque déflationniste
            \item Les deux objectifs sont \textbf{alignés}
         \end{itemize}
         \vspace{0.15cm}
         {\footnotesize\textcolor{HECgreen}{$\to$ Facile pour la banque centrale}}
      \end{column}
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECcoral}{Choc d'offre}}
         \vspace{0.15cm}
         \begin{itemize}
            \setlength\itemsep{0.15cm}
            \item $Y \downarrow$ et $P \uparrow$
            \item Stimuler DA $\to$ relance $Y$ mais \textbf{aggrave} l'inflation
            \item Combattre l'inflation $\to$ aggrave la récession
         \end{itemize}
         \vspace{0.15cm}
         {\footnotesize\textcolor{HECcoral}{$\to$ Dilemme pour la banque centrale}}
      \end{column}
   \end{columns}
   \vspace{0.3cm}
   \begin{warning}
      Les chocs d'offre sont le \textbf{cauchemar} des banquiers centraux : il n'y a pas de solution indolore.
   \end{warning}
\end{frame}

\begin{frame}[t]{Pourquoi c'est lent}
   \vspace{0.2cm}
   Le mécanisme d'autocorrection est \textbf{réel mais lent} :
   \vspace{0.3cm}
   \begin{itemize}
      \setlength\itemsep{0.3cm}
      \item Les salaires sont \textbf{rigides à la baisse} (contrats, salaire minimum, moral des employés)
      \item L'ajustement peut prendre \textbf{des années}
      \item Pendant ce temps : chômage élevé, faillites, perte de capital humain
      \item Risque d'\textbf{hystérèse} : une récession prolongée réduit $Y^{\text{POT}}$ lui-même
   \end{itemize}
\end{frame}

\begin{frame}[t]{1.\ Étincelle : trois types de déclencheurs}
   \vspace{0.2cm}
   \begin{enumerate}
      \setlength\itemsep{0.4cm}
      \item \textbf{Faux pas (\textit{policy error})}
      \vspace{0.1cm}
      \begin{itemize}
         \setlength\itemsep{0.1cm}
         \item Erreur de politique monétaire ou budgétaire
         \item Exemple : resserrement excessif de la Fed en 1937, 1981
      \end{itemize}
      \item \textbf{Gravité (\textit{buildup})}
      \vspace{0.1cm}
      \begin{itemize}
         \setlength\itemsep{0.1cm}
         \item Déséquilibre structurel qui finit par se corriger brutalement
         \item Exemple : bulle immobilière + levier financier $\to$ 2008
      \end{itemize}
      \item \textbf{Choc exogène}
      \vspace{0.1cm}
      \begin{itemize}
         \setlength\itemsep{0.1cm}
         \item Événement imprévisible, externe à l'économie
         \item Exemples : pandémie (2020), embargo pétrolier (1973), conflit Iran/Ormuz (2026)
      \end{itemize}
   \end{enumerate}
\end{frame}

\begin{frame}[t]{2.\ Transmission : comment le choc se propage}
   \small
   L'étincelle se transmet à l'ensemble de l'économie via les courbes OA et DA :
   \vspace{0.3cm}
   \begin{itemize}
      \setlength\itemsep{0.25cm}
      \item \textbf{Canaux de demande :} baisse de $C$ (confiance, richesse), baisse de $I$ (incertitude, crédit), baisse de $NX$ (récession chez les partenaires) $\to$ DA $\leftarrow$
      \item \textbf{Canaux d'offre :} perturbation des chaînes d'approvisionnement, hausse des coûts des intrants, fermetures forcées $\to$ OA $\leftarrow$
   \end{itemize}
\end{frame}

\begin{frame}[t]{3.\ Carburant : ce qui amplifie le feu}
   \vspace{0.2cm}
   \begin{itemize}
      \setlength\itemsep{0.35cm}
      \item \textbf{Interconnexion des chaînes d'approvisionnement} --- un maillon qui casse affecte toute la chaîne
      \item \textbf{Contagion financière} --- les pertes d'une banque fragilisent les autres (2008)
      \item \textbf{Spirale de confiance} --- pessimisme $\to$ moins de dépenses $\to$ licenciements $\to$ plus de pessimisme
      \item \textbf{Spirale salaires-prix} --- inflation $\to$ hausse des salaires $\to$ hausse des coûts $\to$ plus d'inflation (1970s)
      \item \textbf{Cycle des stocks} --- les entreprises liquident les inventaires $\to$ production chute brutalement
   \end{itemize}
\end{frame}

\begin{frame}[t]{4.\ Pare-feu : contenir le feu}
   \small
   Les outils de politique économique pour limiter les dégâts :
   \vspace{0.2cm}
   \begin{columns}[T]
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECnavy}{Politique monétaire}}
         \vspace{0.15cm}
         \begin{itemize}
            \setlength\itemsep{0.15cm}
            \item Baisser le taux directeur $\to$ crédit moins cher $\to$ $C \uparrow$, $I \uparrow$
            \item DA $\to$
            \item Achats d'actifs (QE) quand les taux sont à $\sim 0$\,\%
         \end{itemize}
      \end{column}
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECcoral}{Politique budgétaire}}
         \vspace{0.15cm}
         \begin{itemize}
            \setlength\itemsep{0.15cm}
            \item Hausse de $G$ $\to$ DA $\to$ directement
            \item Baisse d'impôts ($T \downarrow$) $\to$ revenu disponible $\uparrow$ $\to$ $C \uparrow$
            \item Transferts directs aux ménages
         \end{itemize}
      \end{column}
   \end{columns}
   \vspace{0.25cm}
   {\footnotesize\textcolor{HECgray}{$\to$ Détails en Séance 5}}
\end{frame}

\begin{frame}[t]{Pourquoi l'inflation a-t-elle tant augmenté en 2021--22?}

            \item Retour lent des travailleurs
