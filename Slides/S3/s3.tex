      \end{align*}
   \end{defbox}

\begin{frame}[t]{Pourquoi le taux de chômage ne suffit-il pas\,?}\examtag
   \footnotesize
   \vspace{-0.2cm}
   \begin{warning}[Limites du taux de chômage]
      \RaggedRight
      \vspace{-0.4cm}
      \begin{itemize}
         \setlength\itemsep{0.07cm}
         \item \textbf{Travailleurs découragés :} une baisse du chômage peut refléter des \textit{retraits} du marché, non pas des embauches
         \item \textbf{Sous-emploi :} travail à temps partiel faute de trouver un emploi à temps plein
         \item \textbf{Qualité des emplois :} un taux très faible peut signifier que les travailleurs n'ont pas les ressources pour chercher assez longtemps afin de trouver un bon \textit{match}
         \item \textbf{Démographie :} les départs à la retraite réduisent la participation; le taux de chômage ne résume pas cet effet
      \end{itemize}
   \end{warning}
   \vspace{0.05cm}
   \begin{bizimplication}
      \RaggedRight
      Pour évaluer la santé réelle du marché du travail, regardez le taux de participation, le taux d'emploi \textbf{et} les mesures de sous-emploi (U-6 aux États-Unis).
   \end{bizimplication}
\end{frame}

\begin{frame}[t]{Les conséquences du chômage}
\textbf{Pour le travailleur :}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item Perte de revenu et risque de retrouver un emploi \textbf{moins bien payé}
      \item Perte de compétences et de contacts professionnels
   \end{itemize}
   \vspace{0.3cm}
   \textbf{Pour l'entreprise :}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item Coûts de recrutement et de formation
      \item Perte de savoir organisationnel
   \end{itemize}
   \vspace{0.2cm}
   \begin{keyinsight}
      \RaggedRight\hyphenpenalty=10000
      Le chômage peut augmenter fortement même sans vague de licenciements : lorsque les \textbf{embauches ralentissent}, les personnes sans emploi le restent plus longtemps.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Deux horizons temporels}\examtag
   \begin{columns}[T]
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECcoral}{\quad Court terme ($\sim$1 an)}}
         \vspace{0.1cm}
         \begin{itemize}
            \setlength\itemsep{0.1cm}
            \item Le capital ($K$) est \textbf{fixe}
            \item Seul le travail s'ajuste
            \item Un choc d'offre sur $L^S$ (ex.\ immigration) n'affecte pas $L^D$
         \end{itemize}
      \end{column}
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECnavy}{\quad Long terme ($\sim$5+ ans)}}
         \vspace{0.1cm}
         \begin{itemize}
            \setlength\itemsep{0.1cm}
            \item Le capital \textbf{s'ajuste} (Solow\,!)
            \item Plus de travailleurs $\to$ rendement du capital $\uparrow$ $\to$ investissement $\uparrow$
            \item $K \uparrow$ $\to$ $L^D$ se déplace à droite
         \end{itemize}
      \end{column}
   \end{columns}
   \vspace{0.3cm}
   \begin{defbox}[Horizon d'ajustement]
      \RaggedRight
      \textbf{Court terme :} $K$ est fixe, seul $L$ s'ajuste. \\
      \textbf{Long terme :} $K$ s'ajuste via le mécanisme de Solow.
   \end{defbox}
\end{frame}

   Dans nos exemples, le capital finit par s'ajuster à la main-d'œuvre disponible.
   \vspace{0.4cm}
   \begin{enumerate}
      \setlength\itemsep{0.3cm}
      \item Qu'est-ce qui incite une entreprise à \textbf{investir}\,?
      \item Qu'est-ce qui détermine le \textbf{coût du financement}\,?
      \item D'où viennent les ressources pour réaliser ces projets\,?
   \end{enumerate}
   \vspace{0.35cm}
   \begin{keyinsight}
      \RaggedRight
      Il nous faut maintenant un modèle de l'\textbf{épargne et de l'investissement}.
   \end{keyinsight}

\begin{frame}[t]{Capital, investissement et financement}\examtag
\begin{defbox}[Stock vs.\ flux]
   \RaggedRight
      \textbf{Capital ($K$)} : machines, bâtiments, logiciels déjà en place\\[5pt]
      \textbf{Investissement ($I$)} : production de nouveaux actifs
   \end{defbox}
   \vspace{0.3cm}
   \textbf{Exemple :} une entreprise construit une usine et émet des obligations.
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item La construction de l'usine est un \textbf{investissement}.
      \item L'émission d'obligations est son \textbf{financement}.
   \end{itemize}
   \vspace{0.2cm}
   Acheter une action existante ne crée pas directement du nouveau capital.
\end{frame}

\begin{frame}[t]{Faut-il se limiter à l'épargne canadienne\,?}
\vspace{0.2cm}
   Une entreprise canadienne peut financer son usine auprès d'un \textbf{fonds de pension allemand} ou d'un \textbf{investisseur japonais}.
   \vspace{0.4cm}
   \begin{itemize}
      \setlength\itemsep{0.25cm}
      \item Le Canada peut investir \textbf{plus} que son épargne nationale.
      \item Les Canadiens peuvent aussi financer des projets à l'étranger.
      \item Le coût du financement dépend des conditions \textbf{mondiales}.
   \end{itemize}
   \vspace{0.3cm}
   \begin{keyinsight}
      \RaggedRight
      Ouvrir l'économie permet de dissocier l'épargne nationale et l'investissement domestique.
   \end{keyinsight}
\end{frame}

% 37
\begin{frame}[t]{Qu'est-ce que le compte courant\,?}\examtag
Le \textbf{compte courant} ($CA$) mesure ce qu'un pays reçoit du reste du monde, moins ce qu'il lui verse, en termes de commerce, de revenus et de transferts.
   \begin{equation*}
      CA = \underbrace{NX}_{\text{Export.\ nettes}} + \underbrace{\text{Revenus nets}}_{\text{Intérêts, dividendes, etc.}} + \underbrace{\text{Transferts nets}}_{\text{Aide, envois de fonds}}
   \end{equation*}
   \vspace{0.2cm}
   \begin{keyinsight}[$CA \approx NX$ : une approximation raisonnable]
      \RaggedRight
      La balance commerciale explique souvent l'essentiel du compte courant.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Les déficits jumeaux}\examtag
Puisque $S = S_P + S_G$, on peut écrire :
   \begin{equation*} CA = S_P + S_G - I \end{equation*}
   \vspace{0.15cm}
   Si $S_P$ et $I$ ne changent pas, une baisse de l'épargne publique ($S_G \downarrow$) se traduit par une détérioration du compte courant ($CA \downarrow$).
   \vspace{0.25cm}
   \begin{warning}[Déficits jumeaux]
      \RaggedRight
      Un déficit budgétaire peut creuser le déficit courant.
   \end{warning}
\end{frame}

         \textbf{\textcolor{HECcoral}{Davantage de tâches automatisées}}
         \begin{itemize}
            \setlength\itemsep{0.2cm}
            \item Les machines reprennent des tâches auparavant humaines.
            \item À efficacité donnée, la part du revenu total capturée par le travailleur diminue...
         \end{itemize}
      \end{column}
   \end{columns}
   \vspace{0.15cm}
   \begin{keyinsight}[Quelle force l'emportera\,?]
      \RaggedRight
      La part du travail augmente si la rareté du travail humain domine ; elle diminue si le transfert de tâches aux machines l'emporte.
      \par\mbox{}\hfill{\scriptsize\textcolor{HECgray}{\href{https://doi.org/10.1257/aer.20221180}{B. Jones et Liu (2024, AER)}}}
   \end{keyinsight}
\end{frame}
