\begin{frame}[t]{Pourquoi le taux de chômage ne suffit-il pas\,?}\examtag
   \vspace{0.16cm}
   \small
   \tcbset{limiteemploi/.style={
      colback=HEClightgray, colframe=HEClightgray, boxrule=0pt, arc=8pt,
      height=2.17cm, boxsep=0pt, left=10pt, right=10pt, top=7pt, bottom=7pt,
      before skip=0pt, after skip=0pt, before upper={\RaggedRight\hyphenpenalty=10000}
   }}

   \begin{columns}[T,onlytextwidth]
      \begin{column}{0.485\textwidth}
         \begin{tcolorbox}[limiteemploi]
            \textbf{\textcolor{HECnavy}{Travailleurs découragés}}\par\smallskip
            Le chômage peut baisser lorsque des personnes cessent de chercher, même sans nouvelles embauches.
         \end{tcolorbox}
      \end{column}
      \begin{column}{0.485\textwidth}
         \begin{tcolorbox}[limiteemploi]
            \textbf{\textcolor{HECnavy}{Sous-emploi}}\par\smallskip
            Certaines personnes travaillent à temps partiel faute de trouver un emploi à temps plein.
         \end{tcolorbox}
      \end{column}
   \end{columns}
   \vspace{0.13cm}
   \begin{columns}[T,onlytextwidth]
      \begin{column}{0.485\textwidth}
         \begin{tcolorbox}[limiteemploi]
            \textbf{\textcolor{HECnavy}{Qualité des emplois}}\par\smallskip
            Un taux très faible peut refléter un manque de ressources pour chercher assez longtemps un emploi adapté.
         \end{tcolorbox}
      \end{column}
      \begin{column}{0.485\textwidth}
         \begin{tcolorbox}[limiteemploi]
            \textbf{\textcolor{HECnavy}{Démographie}}\par\smallskip
            Les départs à la retraite réduisent la participation, un effet que le taux de chômage ne capture pas.
         \end{tcolorbox}
      \end{column}
   \end{columns}

   \vspace{0.08cm}
   \begin{warning}[Indicateurs à suivre conjointement]
      \footnotesize\RaggedRight
      Taux de participation, taux d'emploi et mesures de sous-emploi (U-6 aux États-Unis).
   \end{warning}
\end{frame}

\begin{frame}{Les conséquences du chômage}
   \tcbset{consequenceemploi/.style={
      colback=HECcoral!7, colframe=HECcoral!7, boxrule=0pt, arc=8pt,
      height=3.10cm, boxsep=0pt, left=9pt, right=9pt, top=7pt, bottom=7pt,
      before skip=0pt, after skip=0pt, before upper={\RaggedRight\hyphenpenalty=10000}
   }}
   \begin{columns}[T,onlytextwidth]
      \begin{column}{0.485\textwidth}
         \begin{tcolorbox}[consequenceemploi]
            \small\textbf{\textcolor{HECnavy}{Revenu perdu après un licenciement}}\par\smallskip
            {\bfseries\color{HECcoral}{\Large 12\,\%} de revenu en moins}\par\smallskip
            \footnotesize
            Perte moyenne actualisée sur 20 ans après un licenciement collectif
            (hommes, États-Unis).\par\smallskip
            {\scriptsize\hfill\href{https://www.brookings.edu/articles/recessions-and-the-costs-of-job-loss/}{Davis et von Wachter (2011)}\par}
         \end{tcolorbox}
      \end{column}
      \begin{column}{0.485\textwidth}
         \begin{tcolorbox}[consequenceemploi]
            \small\textbf{\textcolor{HECnavy}{Accès aux entretiens d'embauche}}\par\smallskip
            {\bfseries\color{HECcoral}{\Large \(\approx\) 45\,\%} de rappels en moins}\par\smallskip
            \footnotesize
            Dans une expérience américaine, des CV fictifs comparables
            indiquaient 8 mois sans emploi plutôt qu'un seul.\par\smallskip
            {\scriptsize\hfill\href{https://doi.org/10.1093/qje/qjt015}{Kroft, Lange et Notowidigdo (2013)}\par}
         \end{tcolorbox}
      \end{column}
   \end{columns}

   \vspace{0.32cm}
   {\footnotesize\RaggedRight
      \textbf{Pour les entreprises.} Savoir-faire perdu, puis coûts pour recruter et former à la reprise.\par}
   \vspace{0.10cm}
   \begin{keyinsight}[Pourquoi le chômage augmente]
      \footnotesize\RaggedRight\hyphenpenalty=10000
      Le \textbf{taux de perte d'emploi augmente} ou le \textbf{taux de retour à l'emploi baisse}.\par
      En récession, le second mécanisme domine souvent après les premiers mois (É.-U.).
   \end{keyinsight}
\end{frame}

         \end{column}
         \begin{column}{0.72\textwidth}
            \RaggedRight\small\hyphenpenalty=10000
            \textbf{Capital \(K\) fixe. À l'impact, salaire réel \(s\) rigide à la baisse.}\par\smallskip

         \end{column}

         \end{column}
         \begin{column}{0.72\textwidth}
            \RaggedRight\small\hyphenpenalty=10000
            \textbf{Le capital \(K\) s'ajuste par l'investissement (Solow).}\par\smallskip
            Plus d'emploi rend le capital plus rentable.\par
            L'investissement augmente et \(L^{D}\) se déplace vers la droite.
         \end{column}
      \end{columns}
   \end{tcolorbox}

   \par\vspace{0.25cm}
   \begin{columns}[T]

         \RaggedRight
         \textbf{\textcolor{HECgreen}{Des machines plus efficaces}}
         \begin{itemize}
            \setlength\itemsep{0.2cm}
            \item La production des machines devient abondante.
            \item Le travail humain devient le goulot d'étranglement.
            \item Le facteur rare capture la valeur\,!
         \end{itemize}
      \end{column}
      \begin{column}{0.47\textwidth}
         \RaggedRight
         \textbf{\textcolor{HECcoral}{Davantage de tâches automatisées}}
         \begin{itemize}
            \setlength\itemsep{0.2cm}
            \item Les machines reprennent des tâches auparavant humaines.
            \item À efficacité donnée, la part du revenu total capturée par le travailleur diminue...
         \end{itemize}
      \end{column}
   \end{columns}
   \vspace{0.15cm}
   \begin{keyinsight}[Quelle force l'emportera\,?]
      \RaggedRight
      La part du travail augmente si la rareté du travail humain domine ; elle diminue si le transfert de tâches aux machines l'emporte.
      \par{\scriptsize\hyperlink{s3-backup-ordinateurs}{%

% 54
\begin{frame}[t]{Qu'est-ce que le compte courant\,?}\examtag
Le \textbf{compte courant} ($CA$) mesure ce qu'un pays reçoit du reste du monde, moins ce qu'il lui verse, en termes de commerce, de revenus et de transferts.
   \begin{equation*}
      CA = \underbrace{NX}_{\text{Export.\ nettes}} + \underbrace{\text{Revenus nets}}_{\text{Intérêts, dividendes, etc.}} + \underbrace{\text{Transferts nets}}_{\text{Aide, envois de fonds}}
   \end{equation*}
   \vspace{0.2cm}
   \begin{keyinsight}[$CA \approx NX$ : une approximation raisonnable]
      \RaggedRight
      La balance commerciale explique souvent l'essentiel du compte courant.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Les déficits jumeaux}\examtag
Puisque $E = E_{P} + E_{G}$, on peut écrire :
   \begin{equation*} CA = E_{P} + E_{G} - I \end{equation*}
   \vspace{0.15cm}
   Si $E_{P}$ et $I$ ne changent pas, une baisse de l'épargne publique ($E_{G} \downarrow$) se traduit par une détérioration du compte courant ($CA \downarrow$).
   \vspace{0.25cm}
   \begin{warning}[Déficits jumeaux]
      \RaggedRight
      Un déficit budgétaire peut creuser le déficit courant.
   \end{warning}
\end{frame}

\begin{frame}[t]{Emprunter pour investir ou pour consommer\,?}
\small
Un \textbf{déficit courant} peut être bénéfique ou préoccupant selon ce que finance l'emprunt.
   \vspace{0.1cm}
   \begin{columns}[T]
      \begin{column}{0.47\textwidth}
         \textbf{\textcolor{HECgreen}{Investissements productifs}}
         \RaggedRight
         \begin{itemize}
            \setlength\itemsep{0.2cm}
            \item Usines, logiciels, infrastructures productives
            \item Les revenus futurs augmentent et peuvent permettre de rembourser.
         \end{itemize}
      \end{column}
      \begin{column}{0.47\textwidth}
         \textbf{\textcolor{HECcoral}{Consommation et gaspillage}}
         \RaggedRight
         \begin{itemize}
            \setlength\itemsep{0.2cm}
            \item Consommation excessive, dépenses publiques improductives
            \item La dette augmente sans les revenus nécessaires pour rembourser.
         \end{itemize}
      \end{column}
   \end{columns}
   \vspace{0.2cm}
   \begin{keyinsight}[Emprunter pour produire davantage demain]
      \RaggedRight
      Un déficit peut être bénéfique si les investissements financés rapportent davantage que le coût de l'emprunt.
   \end{keyinsight}
\end{frame}
