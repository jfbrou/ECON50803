\begin{frame}[t]{Comment mesure-t-on le marché du travail\,?}
   Les statistiques sur l'emploi proviennent de deux enquêtes complémentaires :
   \vspace{0.25cm}
   \begin{enumerate}
      \setlength\itemsep{0.25cm}
      \item \textbf{Enquête auprès des ménages} (Canada : \textbf{EPA}; É.-U.\ : \textbf{CPS})
      \vspace{0.15cm}
      \begin{itemize}
         \setlength\itemsep{0.15cm}
         \item $\sim$50K ménages contactés chaque mois au Canada
         \item Questions sur le statut d'emploi durant la \textit{semaine de référence}
      \end{itemize}
      \item \textbf{Enquête auprès des établissements} (Canada : \textbf{EERH}; É.-U.\ : \textbf{CES})
      \vspace{0.15cm}
      \begin{itemize}
         \setlength\itemsep{0.15cm}
         \item Compte les emplois à partir des \textbf{registres de paie} des entreprises
         \item Plus fiable pour le \textit{nombre} d'emplois, mais publiée avec délai
      \end{itemize}
   \end{enumerate}
   \vspace{0.25cm}
   Les rapports sur l'emploi font réagir les marchés financiers.
\end{frame}

\begin{frame}[t]{Trois ratios essentiels}\examtag
   \begin{defbox}[Indicateurs du marché du travail]
      \RaggedRight
      \vspace{-0.5cm}
      \begin{align*}
         \text{Taux de participation} &= \frac{E + U}{\text{Pop.\ en âge de travailler}} \times 100 \\[5pt]
         \text{Taux d'emploi} &= \frac{E}{\text{Pop.\ en âge de travailler}} \times 100 \\[5pt]
         \text{Taux de chômage} &= \frac{U}{E + U} \times 100
      \end{align*}
   \end{defbox}
\end{frame}

\begin{frame}[t]{Pourquoi le taux de chômage ne suffit-il pas\,?}\examtag
   \footnotesize
   \vspace{-0.2cm}
   \begin{warning}[Limites du taux de chômage]
      \RaggedRight
      \vspace{-0.4cm}
      \begin{itemize}
         \setlength\itemsep{0.07cm}
         \item \textbf{Travailleurs découragés :} une baisse du chômage peut refléter des \textit{retraits} du marché, non pas des embauches
         \item \textbf{Sous-emploi :} travail à temps partiel faute de trouver un emploi à temps plein
         \item \textbf{Qualité des emplois :} un taux très faible peut signifier que les travailleurs n'ont pas les ressources pour chercher assez longtemps afin de trouver un bon \textit{match}
         \item \textbf{Démographie :} les départs à la retraite réduisent la participation; le taux de chômage ne résume pas cet effet
      \end{itemize}
   \end{warning}
   \vspace{0.05cm}
   \begin{bizimplication}
      \RaggedRight
      Pour évaluer la santé réelle du marché du travail, regardez le taux de participation, le taux d'emploi \textbf{et} les mesures de sous-emploi (U-6 aux États-Unis).
   \end{bizimplication}
\end{frame}

% 10
\begin{frame}[t]{Les conséquences du chômage}
\textbf{Pour le travailleur :}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item Perte de revenu et risque de retrouver un emploi \textbf{moins bien payé}
      \item Perte de compétences et de contacts professionnels
   \end{itemize}
   \vspace{0.3cm}
   \textbf{Pour l'entreprise :}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item Coûts de recrutement et de formation
      \item Perte de savoir organisationnel
   \end{itemize}
   \vspace{0.2cm}
   \begin{keyinsight}
      \RaggedRight
      Le chômage dépend des \textbf{sorties d'emploi}, mais aussi de la vitesse à laquelle les personnes retrouvent un emploi.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Deux horizons temporels}\examtag
   \begin{columns}[T]
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECcoral}{\quad Court terme ($\sim$1 an)}}
         \vspace{0.1cm}
         \begin{itemize}
            \setlength\itemsep{0.1cm}
            \item Le capital ($K$) est \textbf{fixe}
            \item Seul le travail s'ajuste
            \item Un choc d'offre sur $L^S$ (ex.\ immigration) n'affecte pas $L^D$
         \end{itemize}
      \end{column}
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECnavy}{\quad Long terme ($\sim$5+ ans)}}
         \vspace{0.1cm}
         \begin{itemize}
            \setlength\itemsep{0.1cm}
            \item Le capital \textbf{s'ajuste} (Solow\,!)
            \item Plus de travailleurs $\to$ rendement du capital $\uparrow$ $\to$ investissement $\uparrow$
            \item $K \uparrow$ $\to$ $L^D$ se déplace à droite
         \end{itemize}
      \end{column}
   \end{columns}
   \vspace{0.3cm}
   \begin{defbox}[Horizon d'ajustement]
      \RaggedRight
      \textbf{Court terme :} $K$ est fixe, seul $L$ s'ajuste. \\
      \textbf{Long terme :} $K$ s'ajuste via le mécanisme de Solow.
   \end{defbox}
\end{frame}

\begin{frame}[t]{Qui épargne\,?}\examtag
   \begin{defbox}[Épargne nationale]
      \RaggedRight
      \vspace{-0.25cm}
      \begin{equation*} S = S_P + S_G \end{equation*}
      La part du revenu qui n'est pas consommée.
   \end{defbox}
   \vspace{0.25cm}
   \begin{itemize}
      \setlength\itemsep{0.3cm}

      \item \textbf{Épargne publique ($S_G$) :} dans le modèle, $S_G = T - G$.

   {\small $T$ : impôts nets des transferts; $G$ : consommation publique.\\[4pt]
   L'investissement public fait partie de $I$, pas de $G$.}

\begin{frame}[t]{Qu'est-ce que le compte courant\,?}\examtag
Le \textbf{compte courant} ($CA$) mesure ce qu'un pays reçoit du reste du monde, moins ce qu'il lui verse, en termes de commerce, de revenus et de transferts.
   \begin{equation*}
      CA = \underbrace{NX}_{\text{Export.\ nettes}} + \underbrace{\text{Revenus nets}}_{\text{Intérêts, dividendes, etc.}} + \underbrace{\text{Transferts nets}}_{\text{Aide, envois de fonds}}
   \end{equation*}
   \vspace{0.2cm}
   \begin{keyinsight}[$CA \approx NX$ : une approximation raisonnable]
      \RaggedRight
      La balance commerciale explique souvent l'essentiel du compte courant.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Du PIB au compte courant}\examtag
Partons de l'identité comptable du PIB (rappel séance 1) :
   \vspace{0.2cm}
   \begin{align*}
      Y &= C + I + G + NX \\[8pt]
      \underbrace{Y - C - G}_{\textcolor{HECgreen}{\boldsymbol{S}}} &= I + NX \\[10pt]
      \textcolor{HECgreen}{S - I} &= \textcolor{HECgreen}{NX = CA} \\
   \end{align*}
   \vspace{0.25cm}
   $S = Y - C - G$ est l'\textbf{épargne nationale} : ce qui reste du revenu après $C$ et $G$.
\end{frame}

\begin{frame}[t]{De l'économie fermée à l'économie ouverte}
   \small
   \vspace{0.1cm}
   \begin{columns}[T]
      \begin{column}{0.47\textwidth}
         \textbf{\textcolor{HECgray}{Économie fermée}}
         \vspace{0.2cm}
         \begin{itemize}
            \setlength\itemsep{0.2cm}
            \item $S = I$ toujours
            \item Pas d'emprunt possible à l'étranger
            \item $r$ s'ajuste pour égaliser $S$ et $I$
         \end{itemize}
      \end{column}
      \begin{column}{0.47\textwidth}
         \textbf{\textcolor{HECnavy}{Économie ouverte}}
         \vspace{0.2cm}
         \begin{itemize}
            \setlength\itemsep{0.2cm}
            \item $S \neq I$ est possible
            \item La différence est le compte courant
            \item $CA = S - I$
         \end{itemize}
      \end{column}
   \end{columns}
   \vspace{0.3cm}
   \begin{defbox}[Petite vs grande économie ouverte]
      \RaggedRight
      \footnotesize
      \textbf{Grande économie ouverte} (É.-U., Chine) : ses décisions affectent $r^W$. Un choc domestique modifie partiellement le taux mondial.\\[4pt]
      \textbf{Petite économie ouverte} (Canada, Suisse) : $r^W$ est \textbf{donné}. À $r^W$ donné, les chocs domestiques modifient surtout le compte courant.
   \end{defbox}
\end{frame}

\begin{frame}[t]{Progrès technique : qui fait mieux, qui fait quoi\,?}
   \small
   Quand les tâches humaines--machines sont \textbf{complémentaires}, deux forces s'opposent :
   \par\vspace{0.25cm}
   \begin{columns}[T]
      \begin{column}{0.47\textwidth}
         \RaggedRight
         \textbf{\textcolor{HECgreen}{Des machines plus efficaces}}
         \begin{itemize}
            \setlength\itemsep{0.2cm}
            \item La production des machines devient abondante.
            \item Le travail humain devient le goulot d'étranglement.
            \item Le facteur rare capture la valeur\,!
         \end{itemize}
      \end{column}
      \begin{column}{0.47\textwidth}
         \RaggedRight
         \textbf{\textcolor{HECcoral}{Davantage de tâches automatisées}}
         \begin{itemize}
            \setlength\itemsep{0.2cm}
            \item Les machines reprennent des tâches auparavant humaines.
            \item À efficacité donnée, la part du revenu total capturée par le travailleur diminue...
         \end{itemize}
      \end{column}
   \end{columns}
   \vspace{0.15cm}
   \begin{keyinsight}[Quelle force l'emportera\,?]
      \RaggedRight
      La part du travail augmente si la rareté du travail humain domine ; elle diminue si le transfert de tâches aux machines l'emporte.
      \par\mbox{}\hfill{\scriptsize\textcolor{HECgray}{\href{https://doi.org/10.1257/aer.20221180}{B. Jones et Liu (2024, AER)}}}
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Une infinité de chaussures droites... et alors\,?}
   Une paire exige \textbf{une chaussure gauche et une chaussure droite}.
   \par\vspace{0.2cm}
   {\small Production par jour, à capacité inchangée pour les chaussures gauches.\par}
   \par\vspace{0.2cm}
   \renewcommand{\arraystretch}{1.4}
   \centering
   \begin{tabular}{lccc}
      \toprule
      & \textbf{Gauches} & \textbf{Droites} & \textbf{Paires} \\
      \midrule
      Avant & 100 & 100 & \textbf{100} \\
      Droites automatisées & 100 & $\infty$ & \textbf{100} \\
      \bottomrule
   \end{tabular}
   \par\vspace{0.25cm}
   \begin{keyinsight}[Les complémentarités limitent les gains]
      \small\RaggedRight
      Automatiser un maillon ne suffit pas : l'autre limite la production.
      \par\mbox{}\hfill{\scriptsize\textcolor{HECgray}{Exemple pédagogique ; \href{https://web.stanford.edu/~chadj/JonesTonetti_Automation.pdf}{maillons faibles : C. Jones et Tonetti (2026).}}}
   \end{keyinsight}
   % Perfect complements: pairs = min(left shoes,right shoes).
   % Hold left-shoe capacity fixed. Reallocating workers or expanding that
   % capacity could raise pair output; the example isolates the weak-link logic.
\end{frame}
