\begin{frame}[t]{Comment mesure-t-on le marché du travail\,?}
   Les statistiques sur l'emploi proviennent de deux enquêtes complémentaires :
   \vspace{0.25cm}
   \begin{enumerate}
      \setlength\itemsep{0.25cm}
      \item \textbf{Enquête auprès des ménages} (Canada : \textbf{EPA}; É.-U.\ : \textbf{CPS})
      \vspace{0.15cm}
      \begin{itemize}
         \setlength\itemsep{0.15cm}
         \item $\sim$50K ménages contactés chaque mois au Canada
         \item Questions sur le statut d'emploi durant la \textit{semaine de référence}
      \end{itemize}
      \item \textbf{Enquête auprès des établissements} (Canada : \textbf{EERH}; É.-U.\ : \textbf{CES})
      \vspace{0.15cm}
      \begin{itemize}
         \setlength\itemsep{0.15cm}
         \item Compte les emplois à partir des \textbf{registres de paie} des entreprises
         \item Plus fiable pour le \textit{nombre} d'emplois, mais publiée avec délai
      \end{itemize}
   \end{enumerate}
   \vspace{0.25cm}
   Les marchés financiers réagissent fortement aux rapports mensuels sur l'emploi\ldots
\end{frame}

\begin{frame}[t]{Trois ratios essentiels}\examtag
   \begin{defbox}[Indicateurs du marché du travail]
      \vspace{-0.5cm}
      \begin{align*}
         \text{Taux de participation} &= \frac{E + U}{\text{Pop.\ en âge de travailler}} \times 100 \\[5pt]
         \text{Taux d'emploi} &= \frac{E}{\text{Pop.\ en âge de travailler}} \times 100 \\[5pt]
         \text{Taux de chômage} &= \frac{U}{E + U} \times 100
      \end{align*}
   \end{defbox}
\end{frame}

\begin{frame}[t]{Pourquoi le taux de chômage ne suffit-il pas\,?}\examtag
   \footnotesize
   \vspace{-0.2cm}
   \begin{warning}[Limites du taux de chômage]
      \vspace{-0.4cm}
      \begin{itemize}
         \setlength\itemsep{0.07cm}
         \item \textbf{Travailleurs découragés :} une baisse du chômage peut refléter des \textit{retraits} du marché, non pas des embauches
         \item \textbf{Sous-emploi :} travail à temps partiel faute de trouver un emploi à temps plein
         \item \textbf{Qualité des emplois :} un taux très faible peut signifier que les travailleurs n'ont pas les ressources pour chercher assez longtemps afin de trouver un bon \textit{match}
         \item \textbf{Démographie :} le vieillissement fait baisser le chômage (les retraités quittent la population active) sans amélioration réelle du marché du travail
      \end{itemize}
   \end{warning}
   \vspace{0.05cm}
   \begin{bizimplication}
      Pour évaluer la santé réelle du marché du travail, regardez le taux de participation, le taux d'emploi \textbf{et} les mesures de sous-emploi (U-6 aux États-Unis).
   \end{bizimplication}
\end{frame}

\begin{frame}[t]{Les conséquences du chômage}
   \small
   \textbf{Pour le travailleur :}
   \vspace{0.1cm}
   \begin{itemize}
      \setlength\itemsep{0.1cm}
      \item 10--20\,\% de pertes de \textbf{revenus à vie}
      \item Principalement dû à des \textbf{salaires plus bas}, moins à une baisse de l’emploi
   \end{itemize}
   \vspace{0.25cm}
   \textbf{Pour l’entreprise :}
   \vspace{0.1cm}
   \begin{itemize}
      \setlength\itemsep{0.1cm}
      \item \textbf{Coûts de recrutement} élevés
      \item Perte de \textbf{savoir organisationnel}
   \end{itemize}
   \vspace{0.1cm}
   \begin{keyinsight}[Le saviez-vous\,?]
      En récession, le taux de chômage augmente principalement parce que les entreprises \textbf{cessent d’embaucher}, pas parce qu’elles licencient davantage.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Deux horizons temporels}\examtag
   \begin{columns}[T]
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECcoral}{\quad Court terme ($\sim$1 an)}}
         \vspace{0.1cm}
         \begin{itemize}
            \setlength\itemsep{0.1cm}
            \item Le capital ($K$) est \textbf{fixe}
            \item Seul le travail s'ajuste
            \item Un choc d'offre sur $L^S$ (ex.\ immigration) n'affecte pas $L^D$
         \end{itemize}
      \end{column}
      \begin{column}{0.48\textwidth}
         \textbf{\textcolor{HECnavy}{\quad Long terme ($\sim$5+ ans)}}
         \vspace{0.1cm}
         \begin{itemize}
            \setlength\itemsep{0.1cm}
            \item Le capital \textbf{s'ajuste} (Solow\,!)
            \item Plus de travailleurs $\to$ rendement du capital $\uparrow$ $\to$ investissement $\uparrow$
            \item $K \uparrow$ $\to$ $L^D$ se déplace à droite
         \end{itemize}
      \end{column}
   \end{columns}
   \vspace{0.3cm}
   \begin{defbox}[Horizon d'ajustement]
      \textbf{Court terme :} $K$ est fixe, seul $L$ s'ajuste. \\
      \textbf{Long terme :} $K$ s'ajuste via le mécanisme de Solow.
   \end{defbox}
\end{frame}

\begin{frame}{Pourquoi les salaires ne baissent-ils pas\,?}
   \vspace{-0.5cm}
   \begin{defbox}[Rigidité nominale des salaires]
      Les salaires nominaux résistent fortement à la baisse, même en récession.
   \end{defbox}
   \vspace{0.3cm}
   Trois raisons principales :
   \vspace{0.15cm}
   \begin{itemize}
      \setlength\itemsep{0.2cm}
      \item \textbf{Conventions collectives} --- les salaires sont fixés par contrat pour 2--3 ans
      \item \textbf{Contrats implicites} --- baisser les salaires détruit le moral et la productivité
      \item \textbf{Normes sociales} --- une baisse de salaire est perçue comme une injustice
   \end{itemize}
   \vspace{0.3cm}
   Résultat : les entreprises préfèrent \textbf{licencier} que baisser les salaires.
\end{frame}

\begin{frame}{Pourquoi l'inflation aide en récession}
   Si les salaires nominaux ne peuvent pas baisser, comment rétablir l'équilibre\,?
   \vspace{0.3cm}

   \textbf{L'inflation} offre une solution élégante :
   \vspace{0.2cm}
   \begin{center}
      $\displaystyle w_{\text{réel}} = \frac{w_{\text{nominal}}}{P}$
   \end{center}
   \vspace{0.2cm}
   Si $P \uparrow$ et $w_{\text{nominal}}$ reste fixe $\to$ $w_{\text{réel}} \downarrow$ $\to$ retour vers l'équilibre.
   \vspace{0.3cm}

   Les travailleurs acceptent plus facilement une \textbf{stagnation} salariale avec 2\,\% d'inflation (baisse réelle invisible) qu'une \textbf{baisse nominale} de 2\,\%.
\end{frame}

\begin{frame}[t]{Autor, Dorn et Hanson (2013)}
   \textbf{Les états U.S. plus exposés aux importations chinoises ont connu :}
   \vspace{0.25cm}
   \begin{itemize}
      \setlength\itemsep{0.25cm}
      \item Un \textbf{chômage} plus élevé
      \item Une \textbf{participation} plus faible à la population active
      \item Des \textbf{salaires} plus bas
      \item Des transferts plus élevés (chômage, invalidité, retraite)
   \end{itemize}
   \vspace{0.2cm}
   \begin{keyinsight}
      Le commerce avec la Chine profite aux É.-U.\ dans l'ensemble, mais il y a des \textbf{perdants} concentrés géographiquement.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Délocalisation et sous-traitance}
   Aux États-Unis, \textbf{2/3} de la hausse des inégalités salariales s'est produite \textbf{entre} les entreprises, pas au sein de celles-ci.
   \vspace{0.4cm}

   \textbf{Exemple :} une banque sous-traite la conciergerie $\to$ le concierge passe d'une firme à hauts salaires à une firme spécialisée à \textbf{bas salaires}.
   \vspace{0.4cm}

   \begin{keyinsight}
      La sous-traitance trie les travailleurs en entreprises homogènes : les firmes à haute productivité gardent les emplois qualifiés et externalisent le reste. \\
      \mbox{}\hfill{\scriptsize\textcolor{HECgray}{Song et al.\ (\textit{QJE}, 2019); Bilal \& Lhuillier (\textit{JPE}, 2024)}}
   \end{keyinsight}

\begin{frame}[t]{Érosion des institutions du marché du travail}

      \item \textbf{Syndicats} --- Baisse du taux de syndicalisation, mais n'explique qu'environ 10\,\% de l'augmentation des inégalités

      \item \textbf{Fiscalité} --- Réductions d'impôts pour les hauts revenus dans les années 1980, mais depuis~:
      \begin{itemize}
         \setlength\itemsep{0.05cm}
         \item 7 économies avancées sur 10 ont \textbf{augmenté} la progressivité de leur fiscalité
         \item Les transferts sociaux sont passés de 17\,\% à 22\,\% du PIB (OCDE)
         \item Le 1\,\% le plus riche paie une part croissante de l'impôt sur le revenu
      \end{itemize}
      \vspace{-0.1cm}
      \quad {\scriptsize\color{HECgray}(\textit{The Economist}, 20 février 2026)}

\begin{frame}[t]{Automatisation et IA : deux forces opposées}
   \framesubtitle{Acemoglu \& Restrepo (2019, \textit{JEP})}
   La production requiert un ensemble de \textbf{tâches}. La technologie change quelles tâches sont effectuées par les machines vs.\ les humains.
   \vspace{0.3cm}

   \textbf{Effet de déplacement :}
   \vspace{0.1cm}
   \begin{itemize}
      \setlength\itemsep{0.1cm}
      \item L'automatisation remplace certaines tâches $\to$ réduit la demande de travail
      \item Caisses automatiques, chaînes d'assemblage robotisées, etc.
   \end{itemize}
   \vspace{0.3cm}

   \textbf{Effet de réintégration :}
   \vspace{0.1cm}
   \begin{itemize}
      \setlength\itemsep{0.1cm}
      \item Réduction des coûts $\to$ hausse de la production $\to$ nouvelles tâches
      \item Les ATMs ont \textit{réduit} le nombre de caissiers par succursale, mais les banques ont ouvert \textbf{plus de succursales} $\to$ l'emploi total a augmenté
   \end{itemize}
\end{frame}

\begin{frame}[t]{Le danger de l'automatisation \guillemotleft~so-so~\guillemotright}
   L'automatisation \guillemotleft~so-so~\guillemotright qui déplace des travailleurs \textit{sans} gains de productivité significatifs crée le pire scénario :
   \vspace{0.15cm}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item \textbf{Caisses libre-service :} déplacent des caissiers, mais n'accélèrent pas vraiment le passage en caisse
      \item \textbf{Systèmes téléphoniques automatisés :} frustrent les clients sans réduire les coûts de manière significative
   \end{itemize}
   \vspace{0.1cm}

   \begin{warning}[Automatisation \guillemotleft~so-so~\guillemotright]
      Une technologie qui remplace des travailleurs par des machines à peine plus productives augmente les inégalités sans faire croître le gâteau. \\
      \mbox{}\hfill{\scriptsize\textcolor{HECgray}{Acemoglu \& Restrepo (2019)}}
   \end{warning}
\end{frame}

\begin{frame}[t]{Leçons de l'histoire}
   \small
   \textbf{Chaque vague technologique a suscité des craintes de chômage de masse :}
   \vspace{0.2cm}
   \begin{itemize}
      \setlength\itemsep{0.2cm}
      \item \textbf{Guichets automatiques} (années 1970--90) : le nombre de caissiers par succursale a baissé, mais l'emploi total a \textit{augmenté}.
      \item \textbf{Tableurs} (années 1980) : Excel n'a pas éliminé les comptables --- il a réduit le coût des calculs et augmenté la demande d'analyse financière.
      \item \textbf{Traitements de texte} (années 1980) : les secrétaires se sont réorientées vers la coordination et la gestion de projets.
   \end{itemize}
   \vspace{0.1cm}

   \begin{keyinsight}[Le saviez-vous\,?]
      Socrate s'opposait à l'invention de l'\textit{écriture}, craignant qu'elle détruise la mémoire et la capacité de penser\ldots
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Et qu'en est-il de l'IA\,?}
   \framesubtitle{Yotzov et al.\ (NBER, 2026) --- 6\,000 dirigeants (É.-U., R.-U., Allemagne, Australie)}
   \small
   \begin{columns}[t]
      \begin{column}{0.47\textwidth}
         \quad \textbf{Aujourd'hui}
         \vspace{0.1cm}
         \begin{itemize}
            \setlength\itemsep{0.1cm}
            \item \textbf{69\,\%} des entreprises l'utilisent
            \item Les dirigeants l'utilisent en moyenne \textbf{1.5\,h/semaine}
            \item \textbf{80\,\%+} des entreprises ne rapportent \textit{aucun impact} sur l'emploi ou la productivité
         \end{itemize}
      \end{column}
      \hspace{-1cm}
      \begin{column}{0.47\textwidth}
         \quad \textbf{Anticipations (3 prochaines années)}
         \vspace{0.1cm}
         \begin{itemize}
            \setlength\itemsep{0.1cm}
            \item Productivité : \textbf{+1.4\,\%}
            \item Production : \textbf{+0.8\,\%}
            \item Emploi selon les \textit{dirigeants} : \textbf{$-$0.7\,\%}
            \item Emploi selon les \textit{employés} : \textbf{+0.5\,\%}
         \end{itemize}
      \end{column}
   \end{columns}
   \vspace{0.15cm}
   \begin{warning}
      Les dirigeants prévoient des suppressions de postes; les employés anticipent des créations nettes. Qui a raison\,? Pour l'instant, les données ne tranchent pas.
   \end{warning}
\end{frame}

\begin{frame}[t]{Pourquoi l'impact de l'IA tarde-t-il\,?}
   \framesubtitle{Le paradoxe de Solow}
   \vspace{0.2cm}
   \begin{keyinsight}[Robert Solow, 1987]
      \small
      \guillemotleft~\textit{On voit l'âge des ordinateurs partout, sauf dans les statistiques de productivité.}\guillemotright
   \end{keyinsight}
   \vspace{0.3cm}
   \begin{itemize}
      \setlength\itemsep{0.3cm}
      \item La même chose se produit avec l'IA --- les gains de productivité \textbf{agrégée} tardent à se matérialiser\ldots
      \item Pourquoi\,? La réponse tient en un mot~: \textbf{complémentarité}
   \end{itemize}
\end{frame}

\begin{frame}[t]{Complémentarité, maillons faibles et goulots d'étranglement}
   \small
   \begin{enumerate}
      \setlength\itemsep{0.2cm}
      \item \textbf{La production est une chaîne de tâches complémentaires} (théorie O-ring)
      \newline{\scriptsize\color{HECgray}Le joint O-ring de la navette Challenger; les chaînes d'approvisionnement paralysées pendant le Covid; toute l'industrie mondiale des puces dépend des machines d'une seule entreprise néerlandaise (ASML)}
      \newline{\scriptsize\color{HECgray}(Kremer, \textit{QJE}, 1993; Jones, \textit{AEJ:~Macro}, 2011)}

      \item \textbf{L'IA n'automatise (encore) que certaines tâches} (paradoxe de Moravec)
      \newline{\scriptsize\color{HECgray}Ce qui est simple pour un humain (jugement, motricité, bon sens) est difficile pour l'IA --- et vice versa}

      \item \textbf{Les tâches non automatisées deviennent les goulots d'étranglement}
      \newline{\scriptsize\color{HECgray}L'économie ne peut croître plus vite que son maillon le plus faible}
      \newline{\scriptsize\color{HECgray}(Jones \& Tonetti, 2026; Aghion, Jones \& Jones, 2018)}

      \item \textbf{Conséquence~: la maladie des coûts de Baumol}
      \newline{\scriptsize\color{HECgray}Les secteurs difficiles à automatiser ne deviennent pas plus productifs, mais doivent compétionner pour les mêmes salaires $\to$ leurs prix montent~: santé, éducation, construction, garde d'enfants}
   \end{enumerate}
\end{frame}

\begin{frame}[t]{Faut-il s'inquiéter de la montée des inégalités\,?}
   \textbf{L'argument \guillemotleft~pour~\guillemotright{} les inégalités~:}
   \vspace{0.15cm}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item Sans récompenses différenciées, qui prendrait des risques, innoverait, travaillerait 80~heures par semaine\,?
      \item L'inégalité des \textbf{résultats} n'est pas le problème en soi
   \end{itemize}
   \vspace{0.3cm}
   \textbf{Le vrai problème --- l'inégalité des \textbf{chances}~:}
   \vspace{0.15cm}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item Combien de futurs Albert Einstein et Marie Curie grandissent dans des familles sans accès à une éducation de qualité\,?
      \item Inégalité des chances $\to$ \textbf{talent gaspillé} $\to$ \textbf{croissance perdue}
      \item Empiriquement, inégalité des résultats $\to$ inégalité des chances\ldots
   \end{itemize}
\end{frame}
