\begin{frame}[t]{Pourquoi ne pas laissez les prix faire leur boulot\,?}
   \small
   \begin{columns}[T]
      \begin{column}{0.45\textwidth}
         \textbf{\textcolor{HECnavy}{Récession} ($Y < Y_{\text{POT}}$)}
         \vspace{0.1cm}
         \begin{enumerate}
            \setlength\itemsep{0.08cm}
            \item Chômage élevé
            \item Travailleurs abondants
            \item Baisse des salaires nominaux
            \item Baisse des coûts de production
            \item Hausse des profits et de l'offre
            \item[\textcolor{HECnavy}{$\to$}] \textbf{AS se déplace vers la droite}
         \end{enumerate}
      \end{column}
      \begin{column}{0.45\textwidth}
         \textbf{\textcolor{HECcoral}{Surchauffe} ($Y > Y_{\text{POT}}$)}
         \vspace{0.1cm}
         \begin{enumerate}
            \setlength\itemsep{0.08cm}
            \item Chômage faible
            \item Travailleurs rares
            \item Hausse des salaires nominaux
            \item Hausse des coûts de production
            \item Baisse des profits et de l'offre
            \item[\textcolor{HECcoral}{$\to$}] \textbf{AS se déplace vers la gauche}
         \end{enumerate}
      \end{column}
   \end{columns}
   \vspace{0.15cm}
   \begin{keyinsight}
      Après un certain temps, les salaires et coûts de production \textbf{s'ajustent naturellement} et l'économie revient vers $Y_{\text{POT}}$. Alors pourquoi combattre les récessions\,?
   \end{keyinsight}

\begin{frame}[t]{Pourquoi ne pas laisser la nature guérir d'elle-même\,?}
   \vspace{0.2cm}
   \begin{columns}[T]
      \begin{column}{0.47\textwidth}
         \textbf{\textcolor{HECnavy}{L'ajustement est lent}}
         \vspace{0.15cm}
         \begin{itemize}
            \setlength\itemsep{0.15cm}
            \item Prix : durée moyenne de 8 mois
            \item Salaires : durée moyenne d'1 an
            \item Ajustement complet : \textbf{2 à 3 ans}
         \end{itemize}
      \end{column}
      \begin{column}{0.47\textwidth}
         \textbf{\textcolor{HECcoral}{Les récessions sont coûteuses}}
         \vspace{0.15cm}
         \begin{itemize}
            \setlength\itemsep{0.15cm}
            \item Chômage massif, faillites
            \item Hystérèse : effets \textbf{permanents}
            \item Coûts sociaux (santé, criminalité)
         \end{itemize}
      \end{column}
   \end{columns}
   \vspace{0.3cm}
   \begin{keyinsight}[Diplômer en récession]
      Les diplômés qui entrent sur le marché du travail en récession subissent des revenus initiaux inférieurs de \textbf{10 à 20\,\%}, un écart qui met \textbf{8 à 15 ans} à se refermer.\\
      \mbox{}\hfill{\scriptsize\textcolor{HECgray}{Oreopoulos et al.\ (\textit{AEJ: Applied}, 2012)\,; Schwandt \& von Wachter (\textit{JLE}, 2019)}}
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{La Banque du Canada : rôles}
   \small
   \vspace{0.2cm}
   \begin{enumerate}
      \setlength\itemsep{0.25cm}
      \item \textbf{Politique monétaire} --- maintenir l'inflation basse, stable et prévisible
      \item \textbf{Stabilité financière} --- partenariat avec les organismes de réglementation
      \item \textbf{Émission de la monnaie} --- conception/émission des billets (\$ numérique\,?)
      \item \textbf{Gestion financière} --- trésorier du gouv. fédéral (services bancaires)
      \item \textbf{Systèmes de paiement} --- supervision de l'infrastructure de paiement
   \end{enumerate}
   \vspace{0.1cm}
   \begin{keyinsight}[Indépendance]
      La BdC est une institution publique \textbf{indépendante} du gouvernement sur le plan opérationnel (objectifs clairs). C'est cette indépendance qui fonde sa \textbf{crédibilité}.
   \end{keyinsight}
\end{frame}

   \end{itemize}

\begin{frame}[t]{L'instrument : le taux directeur}
   \small
   \vspace{0.15cm}
   \begin{defbox}[Taux directeur (taux à un jour)]
      Taux auquel les banques commerciales s'échangent des fonds pour \textbf{24h}.
   \end{defbox}
   \vspace{0.05cm}
   \begin{itemize}
      \setlength\itemsep{0.12cm}
      \item La BdC annonce une \textbf{cible} pour ce taux, 8 fois par année (dates fixes)
      \item Elle l'atteint en ajustant l'\textbf{offre de liquidités} dans le système bancaire
      \item Actuellement : \textbf{2.25\,\%} (mars 2026)
   \end{itemize}
   \vspace{0.05cm}
   \begin{bizimplication}
      Quand la BdC baisse le taux, votre marge de crédit coûte moins cher \textbf{le lendemain}. L'hypothèque suit en quelques semaines (propagation).
   \end{bizimplication}
\end{frame}

\begin{frame}[t]{Défi \#1 : l'incertitude}
   \small
   \vspace{0.2cm}
   \textbf{$Y_{\text{POT}}$ n'est pas directement observable.} Il faut l'\textbf{estimer}.
   \vspace{0.15cm}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item Quelle est la taille de l'\textbf{écart de production}\,?
      \item L'économie est-elle en récession ou simplement en \textbf{ralentissement}\,?
      \item Les données arrivent avec \textbf{retard} et sont souvent \textbf{révisées}
      \item Risque d'en faire trop ou pas assez\ldots
   \end{itemize}
   \vspace{0.1cm}
   \begin{warning}[Analogie]
      On blâme les économistes de ne pas avoir été en mesure de prévoir la Grande Récession de 2008--2009. Pourtant, on ne blâme pas les médecins de ne pas avoir su prévoir la pandémie de COVID-19\ldots
   \end{warning}
\end{frame}

\begin{frame}[t]{Défi \#2 : les délais de transmission}
   \small
   \vspace{0.2cm}
   \textbf{Consensus :} 6 à 8 trimestres entre la décision et l'effet maximal sur l'inflation.
   \vspace{0.2cm}
   \begin{itemize}
      \setlength\itemsep{0.2cm}
      \item La BdC doit donc agir en fonction de ses \textbf{prévisions}, pas de la situation actuelle
      \item Les prévisions sont \textbf{imparfaites} --- c'est loin d'être une science exacte
      \item Métaphore : \guillemotleft~conduire en regardant dans le rétroviseur~\guillemotright
   \end{itemize}
   \vspace{0.2cm}
   \begin{keyinsight}
      C'est pour cette raison que les banques centrales répètent que leurs décisions sont \guillemotleft~\textit{forward-looking}~\guillemotright\ et dépendent des \guillemotleft~données à venir~\guillemotright.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Défi \#4 : crédibilité et indépendance}

   \vspace{0.1cm}
   \begin{itemize}
      \setlength\itemsep{0.1cm}
      \item Capacité de la BC à lisser les cycles économiques dépend de sa crédibilité
      \item Si la BC est crédible, les gens ajustent leurs anticipations de prix en conséquence
   \end{itemize}

   \begin{keyinsight}
      \footnotesize L'indépendance est le \textbf{moyen}, la crédibilité est la \textbf{fin}. Une BC indépendante non crédible ne sert à rien\,; une BC crédible a besoin d'indépendance pour le demeurer.
   \end{keyinsight}

\begin{frame}[t]{Défi \#5 : la trappe de liquidité}
   \small
   \vspace{-0.25cm}
   \begin{defbox}[Borne inférieure zéro]
      \footnotesize Si le taux directeur descend à 0\,\%, les ménages et les banques peuvent toujours détenir de la \textbf{monnaie physique}, qui génère 0\,\% de rendement. Le taux ne peut donc pas descendre beaucoup en dessous de zéro --- sinon, tout le monde retire ses dépôts.
   \end{defbox}
   \begin{itemize}
      \setlength\itemsep{0.1cm}
      \item En 2009 et en 2020, la BdC, la Fed et la BCE ont atteint $\sim$0\,\%
      \item L'économie avait \textbf{encore besoin} de stimulation\ldots
      \item Certaines BC ont tenté des \textbf{taux négatifs} (BCE: $-$0.5\,\%, Suède, Japon) --- effet limité
   \end{itemize}
   \vspace{0.05cm}
   \begin{keyinsight}
      \footnotesize Quand le taux directeur est à zéro, l'outil principal de la politique monétaire est \textbf{paralysé} (\guillemotleft~pousser sur une corde~\guillemotright). Il faut se tourner vers des outils \textbf{non conventionnels}\ldots
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Indications prospectives (\textit{forward guidance})}
   \small
   \vspace{0.2cm}
   \textbf{Idée :} la banque centrale s'engage publiquement à garder son taux directeur bas pour \guillemotleft~aussi longtemps que nécessaire~\guillemotright.
   \vspace{0.2cm}
   \begin{itemize}
      \setlength\itemsep{0.2cm}
      \item Même si le taux à un jour est déjà à 0\,\%, les taux de \textbf{long terme} dépendent des \textbf{anticipations} du taux directeur futur
      \item En s'engageant à garder les taux bas $\to$ taux longs $\downarrow$ $\to$ stimulation de $C$ et $I$
      \item Exemple : le \textit{dot plot} de la Fed (projections des membres du FOMC)
   \end{itemize}

   \begin{warning}
      Si la banque centrale ne respecte pas sa promesse, elle perd sa crédibilité --- c'est une carte qu'on ne peut jouer qu'\textbf{une seule fois}.
   \end{warning}

\begin{frame}[t]{Assouplissement quantitatif}
   \small
   \vspace{0.2cm}
   \begin{defbox}[Assouplissement quantitatif]
      La banque centrale achète des titres à plus \textbf{longue échéance} sur les marchés secondaires, pour agir \textbf{directement} sur les taux de long terme.
   \end{defbox}
   \vspace{0.15cm}
   \textbf{Mécanisme :}
   \vspace{0.1cm}
   \begin{enumerate}
      \setlength\itemsep{0.1cm}
      \item BdC achète des titres \textbf{à longue échéance} $\to$ prix $\uparrow$
      \item Prix $\uparrow$ $\to$ taux de rendement à long terme $\downarrow$
      \item Taux longs $\downarrow$ $\to$ stimule \textbf{directement} la consommation et l'investissement
   \end{enumerate}
   \vspace{0.25cm}
   \textbf{Utilisé en :} 2008--14 (Fed, BCE), 2020--22 (Fed, BdC, BCE)
\end{frame}

\begin{frame}[t]{Taux d'intérêt négatifs}
   \small
   \begin{defbox}[Taux d'intérêt négatifs]
      La banque centrale impose un taux d'intérêt \textbf{négatif} sur les réserves des banques commerciales : au lieu de recevoir un intérêt, elles \textbf{paient} pour \guillemotleft~parker~\guillemotright~leur argent à la banque centrale.
   \end{defbox}
   \vspace{0.1cm}

   \begin{itemize}

      \item Suède : $-0.5\,\%$ (2015--2019)
      \item Suisse : $-0.75\,\%$ (2015--2022)
      \item Danemark : $-0.75\,\%$ (2012--2022)
   \end{itemize}
   \vspace{0.05cm}
   {\scriptsize\textcolor{HECgray}{Ni la BdC ni la Fed n'ont adopté de taux négatifs.}}

\begin{frame}[t]{Le plancher de 0\,\% et la monnaie numérique}
   \small
   \vspace{0.2cm}
   \textbf{Pourquoi les taux ne peuvent pas descendre beaucoup plus bas\,?}
   \vspace{0.2cm}
   \begin{itemize}
      \setlength\itemsep{0.2cm}
      \item L'argent comptant offre un rendement de \textbf{0\,\%} --- c'est un plancher naturel
      \item Taux très négatifs $\to$ les gens retirent leur argent et le stockent physiquement
      \item En pratique, le plancher se situe autour de $-0.75\,\%$
   \end{itemize}
   \vspace{0.3cm}
   \textbf{La solution radicale :} les \textbf{monnaies numériques} de banque centrale
   \vspace{0.2cm}
   \begin{itemize}
      \setlength\itemsep{0.2cm}
      \item Sans argent comptant, plus d'option de fuite vers le cash
      \item La BC pourrait imposer des taux \textbf{profondément négatifs} en récession sévère
   \end{itemize}
\end{frame}

\begin{frame}[t]{L'autre levier : le gouvernement}
   \small
   \textbf{Trois canaux pour stimuler la demande agrégée :}
   \vspace{0.2cm}
   \begin{enumerate}
      \setlength\itemsep{0.2cm}
      \item \textbf{Dépenses publiques ($G \uparrow$)} --- effet direct sur le PIB (infrastructure, défense, santé)
      \item \textbf{Transferts aux ménages} --- effet indirect via $C$ (chèques, assurance-emploi)
      \item \textbf{Baisses d'impôts ($T \downarrow$)} --- effet indirect via $C$ et $I$ (revenu disponible $\uparrow$)
   \end{enumerate}
   \vspace{0.2cm}
   Certains de ces canaux agissent comme des \textbf{stabilisateurs automatiques} : l'assurance-emploi et l'impôt progressif ajustent $T$ et les transferts \textbf{sans décision politique}.
   \vspace{0.15cm}
   \begin{keyinsight}[Différence fondamentale]
      $G$ entre \textbf{directement} dans le PIB ($Y = C + I + G + NX$). Les transferts et baisses d'impôts ne stimulent le PIB que si les ménages \textbf{dépensent} l'argent reçu.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Déficit vs dette : flux vs stock}
   \small
   \vspace{-0.25cm}
   \begin{columns}[T]
      \begin{column}{0.47\textwidth}
         \begin{defbox}[Déficit budgétaire]
            \footnotesize
            Recettes $<$ Dépenses en une année.\\[4pt]
            Le déficit est un \textbf{flux} : il mesure le déséquilibre \textit{pendant} une période.
         \end{defbox}
      \end{column}
      \begin{column}{0.47\textwidth}
         \begin{defbox}[Dette publique]
            \footnotesize
            Accumulation des déficits passés.\\[4pt]
            La dette est un \textbf{stock} : elle mesure le total \textit{à un moment donné}.
         \end{defbox}
      \end{column}
   \end{columns}
   \vspace{0.3cm}
   \centering
   Relation fondamentale\,:\\[4pt]
   {\large $\text{Dette}_{t} = \text{Dette}_{t-1} + \text{Déficit}_{t}$}
   \vspace{0.15cm}
   \begin{keyinsight}
      On mesure souvent la dette en \textbf{\,\% du PIB} pour la comparer entre pays et dans le temps. Un pays qui croît rapidement peut absorber plus de dette.
   \end{keyinsight}
\end{frame}
