\begin{frame}[t]{Qu'est-ce que la macroéconomie\,?}
   \small
   La \textbf{micro}économie étudie les décisions \textbf{individuelles} des ménages et des entreprises.
   \par\vspace{0.2cm}
   Mais ces décisions s'\textbf{agrègent} et influencent la \textbf{macro}économie dans son ensemble.
   \par\vspace{0.2cm}
   En retour, cette \textbf{agrégation} compte énormément pour les décisions individuelles.
   \vspace{0.15cm}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item Le paradoxe de l'épargne
      \item Se lever à un match de hockey
   \end{itemize}
   \par\vspace{0.3cm}
   La \textbf{macroéconomie} s'intéresse à :
   \vspace{0.15cm}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item La \textbf{vue d'ensemble} de l'économie, pas un marché isolé
      \item Les \textbf{interconnexions} entre consommateurs, entreprises et gouvernement
      \item Les tendances \textbf{long terme}, fluctuations \textbf{court terme} et \textbf{politiques économiques}
   \end{itemize}
\end{frame}

% Source: S1/s1.tex | What GDP measures

\begin{frame}{Qu'est-ce que le PIB\,?}
   \begin{defbox}[Produit intérieur brut (PIB)]
      \textbf{Valeur marchande} totale de tous les biens et services \textbf{finaux} produits \textbf{dans un pays} au cours d'une \textbf{période donnée}.
   \end{defbox}
   \vspace{0.3cm}
   \textbf{Mots clés :}
   \vspace{0.15cm}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item \textbf{Valeur marchande :} mesurée en dollars (ou dans une autre devise)
      \item \textbf{Finaux :} exclut les biens intermédiaires (évite le double comptage)
      \item \textbf{Dans un pays :} production en sol canadien
      \item \textbf{Période donnée :} un flux, pas un stock (trimestriel ou annuel)
   \end{itemize}
\end{frame}

% 11 | \begin{frame}[t]{Comparer le PIB entre pays : convertir les devises}

% ============================================================================
\begin{frame}[t,label=ess-main-prix]{Comparer le PIB entre pays : convertir les devises}
   \small

   \textbf{Comment comparer les PIB de pays utilisant des devises différentes ?}
   \vspace{0.15cm}
   \begin{enumerate}
      \setlength\itemsep{0.15cm}
      \item \textbf{Taux de change du marché :}  $\smallquad e^{N} = \mathdollar_D / \mathdollar_F$
      \vspace{0.1cm}
      \begin{itemize}
         \setlength\itemsep{0.1cm}
         \item Largement disponibles et mis à jour fréquemment
         \item Problème : très volatils, et ignorent les différences de niveaux de prix
      \end{itemize}
      \item \textbf{Parité de pouvoir d'achat (PPA) :} $\smallquad e^{PPA} = P_D / P_F$
      \vspace{0.1cm}
      \begin{itemize}
         \setlength\itemsep{0.1cm}
         \item Meilleures comparaisons de niveaux de vie
         \item Un dollar achète plus à Lagos qu'à New York
         \item Les biens/services sont-ils comparables d'un pays à l'autre\,?
         \item Mis à jour moins fréquemment
         \item Taux de change réel : $e^{R} = e^{N} / e^{PPA}$
      \end{itemize}
   \end{enumerate}
   \essbackupbutton{Prix entre pays}{ess-backup-prix}
\end{frame}

      \small Pour évaluer leur pouvoir d'achat, le \textbf{RNB par habitant en PPA} est souvent plus révélateur que le PIB par habitant.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{PIB et prix}
   \begin{defbox}
      \vspace{-0.5cm}
      \begin{itemize}
         \setlength\itemsep{0.15cm}
         \item \textbf{PIB nominal} = production évaluée aux prix \textit{courants}
         \item \textbf{PIB réel} = production évaluée à prix \textit{constants} (année de référence)
         \item \textbf{Déflateur du PIB} = $\frac{\text{PIB nominal}}{\text{PIB réel}} \cdot 100$
      \end{itemize}
   \end{defbox}
   \vspace{0.05cm}
   \begin{keyinsight}
      Le PIB nominal est contaminé par les variations de prix. Le PIB réel isole les variations de quantité, mesurant la vraie croissance de la \textit{production}.
   \end{keyinsight}
\end{frame}

% 16 | \begin{frame}[t]{Inflation : des prix qui augmentent plus ou moins vite}
% Source: S1/s1.tex | Inflation definition
% Adaptation: Make percentage units and inflation-versus-price-level distinction explicit.
% ============================================================================
\begin{frame}[t]{Inflation : des prix qui augmentent plus ou moins vite}
   \small
   \begin{defbox}[Inflation]
      Variation en pourcentage du \textbf{niveau général des prix}.
   \end{defbox}
   \begin{equation*}
      \pi_t = \frac{P_t - P_{t-1}}{P_{t-1}} \cdot 100
   \end{equation*}
   \vspace{0.1cm}
   \small Dans les médias : généralement la variation des prix \textbf{sur 12 mois}.
   \vspace{0.1cm}
   \begin{keyinsight}[Désinflation $\neq$ déflation]
      \small De 5\,\% à 2\,\% : les prix \textbf{augmentent moins vite}. La \textbf{déflation} désigne une baisse du niveau général des prix.
   \end{keyinsight}
\end{frame}

\begin{frame}[t,label=ess-main-qualite]{Comment mesure-t-on l'IPC\,?}
   \small
   \begin{enumerate}
      \setlength\itemsep{0.3cm}
      \item \textbf{Un panier représentatif} des dépenses des ménages : alimentation, logement, transport, services\ldots
      \item \textbf{Des prix comparables dans le temps}, en tenant compte des changements de quantité et de qualité.
      \item \textbf{Des pondérations selon les dépenses} : une catégorie plus importante pèse davantage dans l'indice. Le panier est actualisé régulièrement.
   \end{enumerate}
   \vspace{0.3cm}
   \begin{bizimplication}[Pour votre organisation]
      \small L'IPC mesure les prix à la \textbf{consommation}. Il ne représente pas nécessairement l'évolution de \textbf{vos coûts d'entreprise}.
   \end{bizimplication}

   \vspace{0.2cm}
   \begin{enumerate}
      \setlength\itemsep{0.2cm}
      \item \textbf{Dépenses publiques ($G \uparrow$)} --- effet direct sur le PIB ($G$)
      \item \textbf{Transferts aux ménages} --- effet indirect via $C$ (e.g., assurance-emploi)

      $G$ entre \textbf{directement} dans le PIB ($Y = C + I + G + NX$). Les transferts soutiennent la demande si les ménages \textbf{dépensent} l'argent reçu.
   \end{keyinsight}

\begin{frame}[t,label=ess-backup-qualite]{Complément : qualité et nouveaux produits}
   \small
   \textbf{\textcolor{HECnavy}{1. Comparer une qualité équivalente}}
   \par\vspace{0.2cm}
   Une voiture plus chère mais mieux équipée : quelle part du prix reflète de l'inflation vs.\ un \textbf{meilleur produit} ? Les agences corrigent pour la qualité, mais imparfaitement.
   \par\vspace{0.4cm}
   \textbf{\textcolor{HECgreen}{2. Intégrer les nouveaux produits}}
   \par\vspace{0.2cm}
   Un nouveau produit n'a pas de prix antérieur. Comparer des bougies et des ampoules exige de mesurer le \textbf{service rendu : les lumens}, pas seulement le prix de chaque produit.
   \par\vspace{0.25cm}
   \begin{keyinsight}[Une mesure utile, mais imparfaite]
      \small Mal mesurer les gains de qualité ou les quantités derrières les nouveaux produits peut conduire à \textbf{surestimer l'inflation} et à \textbf{sous-estimer les gains réels}.
   \end{keyinsight}

\end{frame}
