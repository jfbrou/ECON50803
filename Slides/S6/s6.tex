\begin{frame}[t]{Deux mesures des échanges}
   \footnotesize
   \vspace{-0.25cm}
   \begin{defbox}
      \small
      \textbf{Balance commerciale :} $NX = X - M$ (exportations nettes de biens et services)\\[4pt]
      \textbf{Compte courant :} $CA = NX + \text{revenus nets de facteurs} + \text{transferts nets}$
   \end{defbox}
   \vspace{0.2cm}
   \begin{columns}[T]
      \begin{column}{0.47\textwidth}
         \textbf{Revenus nets de facteurs :}
         \begin{itemize}
            \setlength\itemsep{0.05cm}
            \item Dividendes d'une filiale à l'étranger
            \item Intérêts sur des oblig.\ étrangères
            \item Salaire d'un Canadien aux É.-U.\
         \end{itemize}
      \end{column}
      \begin{column}{0.47\textwidth}
         \textbf{Transferts nets :}
         \begin{itemize}
            \setlength\itemsep{0.05cm}
            \item Aide au développement
            \item Envois de fonds des immigrants
            \item Contributions aux org.\ intl.
         \end{itemize}
      \end{column}
   \end{columns}
   \vspace{0.2cm}
   \begin{keyinsight}
      \footnotesize
      En pratique, $CA \approx NX$. Les autres composantes sont petites par rapport à $NX$.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Du PIB au compte courant}
   \vspace{0.3cm}
   Partons de l'identité comptable du PIB (rappel séance 1) :
   \vspace{0.5cm}
   \begin{align*}
      Y &= C + I + G + NX \\[10pt]
      \underbrace{Y - C - G}_{\textcolor{HECgreen}{\boldsymbol{S}}} &= I + NX \\[14pt]
      \textcolor{HECgreen}{S} - I &= NX \approx CA
   \end{align*}
   \vspace{0.2cm}
   \centering
   où $\textcolor{HECgreen}{S = Y - C - G}$ est l'\textbf{épargne nationale} : ce qui reste du revenu\\
   après la consommation privée ($C$) et publique ($G$).
\end{frame}

\begin{frame}[t]{L'identité fondamentale}
   \vspace{0.3cm}
   \centering
   {\LARGE $\mathbf{CA = S - I}$}
   \vspace{0.5cm}

   \begin{columns}[T]
      \begin{column}{0.47\textwidth}
         \centering
         \textcolor{HECgreen}{\large\bfseries $S > I$ $\to$ $CA > 0$}\\[8pt]
         \small
         Le pays épargne plus qu'il n'investit.\\[4pt]
         Il \textbf{prête} l'excédent au reste du monde.\\[4pt]
         \textcolor{HECgray}{Ex.\ : Allemagne, Chine, Japon}
      \end{column}
      \begin{column}{0.47\textwidth}
         \centering
         \textcolor{HECcoral}{\large\bfseries $S < I$ $\to$ $CA < 0$}\\[8pt]
         \small
         Le pays investit plus qu'il n'épargne.\\[4pt]
         Il \textbf{emprunte} au reste du monde.\\[4pt]
         \textcolor{HECgray}{Ex.\ : É.-U., Canada, Australie}
      \end{column}
   \end{columns}
   \vspace{0.3cm}
   \begin{keyinsight}
      Le compte courant n'est pas une question de compétitivité internationale --- c'est l'\textbf{écart} entre épargne et investissement domestiques.
   \end{keyinsight}
\end{frame}

\begin{frame}[t]{Deux composantes de l'épargne : $S = S_P + S_G$}
   \vspace{0.1cm}
   \begin{columns}[T]
      \begin{column}{0.47\textwidth}
         \begin{defbox}[Épargne privée ($S_P$)]
            \footnotesize
            Ménages + entreprises.\\[4pt]
            \textbf{Déterminants :}
            \begin{itemize}
               \setlength\itemsep{0.08cm}
               \item Taux d'intérêt ($r$)
               \item Revenu courant
               \item Richesse et incertitude
               \item Accès au crédit
            \end{itemize}
         \end{defbox}
      \end{column}
      \begin{column}{0.47\textwidth}
         \begin{defbox}[Épargne publique ($S_G$)]
            \footnotesize
            $S_G = T - G$\\[4pt]
            \textbf{Déterminants :}
            \begin{itemize}
               \setlength\itemsep{0.08cm}
               \item Recettes fiscales ($T$)
               \item Dépenses publiques ($G$)
               \item $S_G > 0$ : surplus budgétaire
               \item $S_G < 0$ : déficit budgétaire
            \end{itemize}
         \end{defbox}
      \end{column}
   \end{columns}
\end{frame}

\begin{frame}[t]{Les déficits jumeaux}
   \small
   Puisque $S = S_P + S_G$, on peut écrire :
   \vspace{0.1cm}
   \[CA = S_P + S_G - I\]
   \vspace{0.1cm}
   Si $S_P$ et $I$ ne changent pas, un creusement du déficit budgétaire ($S_G \downarrow$) se traduit par un creusement du déficit commercial ($CA \downarrow$).
   \vspace{0.25cm}
   \begin{warning}[Déficits jumeaux]
      Un pays qui creuse son déficit \textbf{budgétaire} creuse aussi son déficit \textbf{commercial}.
   \end{warning}
   \vspace{0.25cm}
   \textbf{Exemple :} les É.-U.\ sous Reagan (années 1980). Baisses d'impôts massives + hausse des dépenses militaires $\to$ $S_G \downarrow$ $\to$ déficit budgétaire \textbf{et} déficit commercial records.
\end{frame}

      En économie fermée, $r^*$ est le taux qui égalise l'épargne et l'investissement : $S(r^*) = I(r^*)$.
   \end{defbox}

         \vspace{0.15cm}

\begin{frame}[t]{Lecture du diagramme}
   \small
   \vspace{0.1cm}
   \begin{itemize}
      \setlength\itemsep{0.3cm}
      \item \textbf{Panneau 3 (Monde) :} le taux mondial $r^w$ s'ajuste pour que $S^w(r^w) = I^w(r^w)$
      \item \textbf{Panneau 1 (Domestique) :} au taux $r^w$, le pays investit plus qu'il n'épargne $\to$ $CA < 0$
      \item \textbf{Panneau 2 (Reste du monde) :} au taux $r^w$, le reste du monde épargne plus qu'il n'investit $\to$ $CA > 0$
      \item Les déséquilibres sont parfaitement \textbf{symétriques} : le déficit de l'un est le surplus de l'autre
   \end{itemize}
   \vspace{0.15cm}
   \begin{keyinsight}
      Le taux mondial $r^w$ s'ajuste pour que l'épargne mondiale égale l'investissement mondial : $S^w(r^w) = I^w(r^w)$. Les déséquilibres entre pays sont le miroir les uns des autres.
   \end{keyinsight}
\end{frame}

         \end{keyinsight}
      \end{column}
   \end{columns}
\end{frame}

\begin{frame}[t]{Petite vs grande économie ouverte}
   \small
   \vfill
   \renewcommand{\arraystretch}{1.6}
   \centering
   \begin{tabularx}{\textwidth}{>{\bfseries\small}l >{\small}X >{\small}X >{\small\raggedright\arraybackslash}X}
      \toprule
      & \textbf{Écon. fermée} & \textbf{Grande écon. ouverte} & \textbf{Petite écon. ouverte} \\
      \midrule
      Taux d'intérêt & $r^*$ s'ajuste pour $S = I$ & $r^w$ partiellement affecté & $r^w$ donné (exogène) \\
      Effet d'un choc domestique sur $r$ & Plein ajustement & Ajustement partiel & Aucun effet \\
      Compte courant & $CA = 0$ toujours & $CA \neq 0$ & $CA = S(r^w) - I(r^w)$ \\
      Exemple & --- & É.-U., Chine & Canada, Suisse \\
      \bottomrule
   \end{tabularx}
   \vfill
\end{frame}

\begin{frame}[t]{Le Canada : une petite économie ouverte}
   \vspace{0.2cm}
   \begin{bizimplication}
      \footnotesize
      Le Canada est une petite économie ouverte : le taux d'intérêt mondial est essentiellement \textbf{donné}. Les politiques domestiques (budgétaire, fiscale) affectent principalement le \textbf{compte courant}, pas le taux d'intérêt.
   \end{bizimplication}
   \vspace{0.2cm}
   \small
   \textbf{Conséquences pratiques :}
   \vspace{0.1cm}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item Un déficit budgétaire canadien se traduit directement en \textbf{déficit commercial} (pas en hausse de $r$)
      \item Un boom d'investissement au Canada attire des capitaux étrangers (emprunts), pas une hausse des taux
      \item La BdC ne peut pas s'éloigner trop longtemps des taux américains sans provoquer des mouvements de capitaux
   \end{itemize}
\end{frame}

\begin{frame}[t]{Le puzzle du déficit américain}
   \small
   \vspace{0.2cm}
   Depuis les années 1980, les É.-U.\ ont un \textbf{déficit persistant} du compte courant.
   \vspace{0.3cm}

   \textbf{Les explications domestiques} que nous avons vues :
   \vspace{0.1cm}
   \begin{itemize}
      \setlength\itemsep{0.15cm}
      \item Déficits budgétaires ($S_G \downarrow$) $\to$ prédisent $CA \downarrow$ et $r \uparrow$
      \item Boom technologique des années 1990 ($I \uparrow$) $\to$ prédisent $CA \downarrow$ et $r \uparrow$
      \item Consommation élevée ($S_P \downarrow$) $\to$ prédisent $CA \downarrow$ et $r \uparrow$
   \end{itemize}
   \vspace{0.2cm}
   \begin{warning}[Le puzzle]
      Toutes ces explications prédisent $r \uparrow$. Mais les taux d'intérêt réels ont \textbf{baissé} pendant cette période. Quelque chose d'autre doit être à l'œuvre.
   \end{warning}
\end{frame}

\begin{frame}[t]{L'excès d'épargne mondiale}
   \framesubtitle{Bernanke (\textit{Fed Speech}, 2005)}
   \small
   \vspace{0.1cm}
   La réponse est dans le \textbf{reste du monde} : trois sources d'épargne excédentaire.
   \vspace{0.15cm}
   \begin{enumerate}
      \setlength\itemsep{0.15cm}
      \item \textbf{Asie de l'Est après la crise de 1997} --- épargne de précaution, accumulation massive de réserves en dollars
      \item \textbf{Moyen-Orient} --- revenus pétroliers recyclés en actifs américains (obligations du Trésor)
      \item \textbf{Chine} --- taux d'épargne extraordinairement élevé ($\sim$45\,\% du PIB)
   \end{enumerate}
   \vspace{0.1cm}
   \begin{keyinsight}
      \footnotesize
      L'excès d'épargne mondial explique à la fois le déficit commercial américain \textbf{et} les faibles taux d'intérêt. Les deux phénomènes ont la \textbf{même cause} : $S^w_R \uparrow$ $\to$ $r^w \downarrow$ $\to$ $CA_{\text{É.-U.}} \downarrow$.
   \end{keyinsight}
\end{frame}

   \end{figure}
\end{frame}

\begin{frame}[t]{Le modèle S-I en action}
   \small
   \vspace{0.1cm}
   \textbf{Relecture du puzzle avec le modèle :}
   \vspace{0.15cm}
   \begin{enumerate}
      \setlength\itemsep{0.2cm}
      \item L'excès d'épargne dans le reste du monde ($S_R \uparrow$) déplace $S^w$ vers la droite
      \item $r^w$ \textbf{baisse}
      \item Aux É.-U., au nouveau $r^w$ plus bas : $I_H \uparrow$ et $S_H \downarrow$
      \item Résultat : $CA_{\text{É.-U.}} = S_H - I_H$ se creuse davantage
   \end{enumerate}
   \vspace{0.15cm}
   \begin{bizimplication}
      \footnotesize
      Les déséquilibres mondiaux ne sont pas le résultat de \guillemotleft~mauvais comportements~\guillemotright\ d'un seul pays. Ils reflètent des forces macroéconomiques profondes (démographie, développement financier, politique industrielle) dans \textbf{plusieurs} pays simultanément.
   \end{bizimplication}
\end{frame}


\begin{frame}[t]{Les tarifs dans l'actualité}
   \small
   \vspace{0.2cm}
   En séance 4, nous avons vu l'effet des tarifs comme \textbf{choc d'offre} dans le modèle AD-AS.
   \vspace{0.2cm}
   \begin{itemize}
      \setlength\itemsep{0.2cm}
      \item Les tarifs augmentent les coûts de production $\to$ AS se déplace vers la gauche
      \item Résultat : $Y \downarrow$ et $P \uparrow$ (stagflation)
   \end{itemize}
   \vspace{0.3cm}
   Maintenant, analysons leur effet sur la \textbf{balance commerciale}.
   \vspace{0.2cm}

   {\large La question centrale : les droits de douane peuvent-ils réduire le déficit commercial\,?}
\end{frame}

\begin{frame}[t]{Les tarifs et le modèle S-I}
   \small
   \vspace{0.1cm}
   \begin{warning}[Le point analytique clé]
      \footnotesize
      Puisque $CA = S - I$, les droits de douane ne peuvent réduire le déficit commercial \textbf{que s'ils modifient $S$ ou $I$}.
   \end{warning}
   \vspace{0.15cm}
   \begin{columns}[T]
      \begin{column}{0.47\textwidth}
         \textbf{Que font les tarifs au marché des biens\,?}
         \begin{itemize}
            \setlength\itemsep{0.1cm}
            \item Réduisent les importations de certains pays
            \item Augmentent les prix domestiques
            \item Provoquent des représailles
         \end{itemize}
      \end{column}
      \begin{column}{0.47\textwidth}
         \textbf{Mais que font-ils à $S$ et $I$\,?}
         \begin{itemize}
            \setlength\itemsep{0.1cm}
            \item Ne changent pas l'épargne nationale
            \item Ne changent pas l'investissement
            \item Donc $CA = S - I$ est \textbf{inchangé}
         \end{itemize}
      \end{column}
   \end{columns}
\end{frame}

\begin{frame}[t]{L'évidence empirique}
   \small
   \vspace{0.1cm}
   \textbf{Les tarifs de 2018 sur la Chine :}
   \vspace{0.15cm}
   \begin{itemize}
      \setlength\itemsep{0.2cm}
      \item Le déficit commercial des É.-U.\ \textbf{avec la Chine} a diminué
      \item Mais le déficit commercial \textbf{total} des É.-U.\ est resté essentiellement inchangé
      \item Les importations se sont simplement \textbf{redirigées} : du Vietnam, du Mexique, de l'ASEAN
   \end{itemize}
   \vspace{0.15cm}
   \begin{tcolorbox}[colback=HEClightgray, colframe=HECgray, boxrule=0.5pt, arc=8pt,
      width=0.9\textwidth, height=3.5cm, valign=center]
      \centering
      \textcolor{HECgray}{\small [Figure : Déficit commercial É.-U.\ : total vs avec la Chine (2017--2025)]}\\
      \textcolor{HECgray}{\small Source : U.S.\ Census Bureau}
   \end{tcolorbox}
\end{frame}

\begin{frame}[t]{Le grand enseignement}
   \vspace{0.2cm}
   \begin{keyinsight}[Le message central]
      Les droits de douane ne corrigent pas les déficits commerciaux --- ils ne font que \textbf{rediriger} les flux commerciaux. Pour réduire le déficit, il faudrait augmenter l'épargne ou réduire l'investissement.
   \end{keyinsight}
   \vspace{0.2cm}
   \begin{bizimplication}
      Pour les entreprises, les tarifs créent de l'incertitude et des réorganisations coûteuses des chaînes d'approvisionnement, sans résoudre le problème sous-jacent.
   \end{bizimplication}
   \vspace{0.15cm}
   {\small\textcolor{HECgray}{La vraie question n'est pas \guillemotleft~comment réduire les importations~\guillemotright\ mais \guillemotleft~pourquoi l'Amérique épargne-t-elle si peu\,?~\guillemotright}}
\end{frame}

\begin{frame}[t]{Ce que vous devriez retenir}
   \small
   \vspace{0.1cm}
   \begin{enumerate}
      \setlength\itemsep{0.3cm}
      \item \textbf{La macro, c'est concret} --- chaque concept se traduit en décisions d'affaires
      \item \textbf{Les identités comptables disciplinent la réflexion} --- $Y = C + I + G + NX$ et $CA = S - I$ sont des garde-fous contre les raisonnements fallacieux
      \item \textbf{Le long terme et le court terme sont différents} --- ce qui est bon pour la croissance (épargne, investissement) n'est pas toujours bon pour la stabilisation

      \item \textbf{Tout est interconnecté} --- dans un monde ouvert, les décisions d'un pays affectent tous les autres
   \end{enumerate}
